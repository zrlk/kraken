/-!
# CheckPremises — freeze the MEANING of the finals' premises (mechanism sketch)

For each user-facing final, the constants in its statement are expanded transitively through
`def`/`structure`/`inductive` bodies (theorem bodies are never entered: proofs carry no meaning),
stopping at the machine/language layer, which is listed but not expanded.  The closure is
pretty-printed deterministically to `OUT/<final>.txt`; `guard.sh` compares md5s with the
frozen `premises.md5` and fails unless `WM_APPROVE_PREMISES=<reason>` is set, in which case it
re-freezes and prints the diff for the gate log.  Run:
  LEAN_PATH=<oleans> lean CheckPremises.lean   (≈15 s)
-/
open Lean Meta

namespace CheckPremises

def stopModules : List Name :=
  [`Promises, `WeakMem, `CoreLang, `CoreTerm, `CoreTyping, `CoreUtils, `TermRun, `Caps, `Freeze,
   `Scope, `SubWord, `Syntax, `Typestate, `ArrayLib, `HashMap, `HashMapLF, `HashMapOwn, `FlatHashMap]

/-- The module of `n`, independent of the layout: when the library is vendored with its modules named
`WeakCore.<M>`, the leading `WeakCore` component is dropped, so the stop rule and the `Init`/`Std`/`Lean`
test see `<M>` in both layouts (in the source tree no module name starts with `WeakCore`; in a vendored copy
every module name does, and the prefix is dropped). -/
def moduleOf (env : Environment) (n : Name) : Name :=
  match env.getModuleIdxFor? n with
  | some i =>
      let m := env.header.moduleNames[i.toNat]!
      if m.getRoot == `WeakCore && m != `WeakCore then m.replacePrefix `WeakCore .anonymous else m
  | none => `_local

/-- (per-toolchain sets) The module layout of this run, from the modules of the finals: `vendored` when every
final's module is named `WeakCore.<M>` (the prefix `moduleOf` drops), `source` when none is, `mixed` otherwise
(guard.sh rejects `mixed`).  The layout changes the names of derived instances (the source layout adds a
`_<module>` suffix, the vendored layout none), so the premise sets are keyed on it together with the toolchain. -/
def layoutOf (env : Environment) (finals : List Name) : String :=
  let v := finals.map fun f => match env.getModuleIdxFor? f with
    | some i => let m := env.header.moduleNames[i.toNat]!; m.getRoot == `WeakCore && m != `WeakCore
    | none => false
  if v.all id then "vendored" else if v.all (!·) then "source" else "mixed"

def isStop (m : Name) : Bool :=
  m.getRoot == `Init || m.getRoot == `Std || m.getRoot == `Lean || m == `_local || stopModules.contains m

/-- Stop rule for a constant: stop module, unless forced. -/
def stopC (env : Environment) (force : List Name) (n : Name) : Bool :=
  isStop (moduleOf env n) && !force.contains n

/-- Auxiliary constants generated by the elaborator (recursors, match/proof auxiliaries, eqns). -/
def skip (n : Name) : Bool :=
  n.isInternal || n.toString.endsWith ".mk" || n.toString.endsWith ".rec" || n.toString.endsWith ".casesOn"
  || ["match_", "proof_", "_sunfold", "_unsafe_rec", "_eq_", ".eq_", "instSizeOf", "._", ".noConfusion",
      ".below", ".brecOn", ".recOn", ".binductionOn", ".injEq", ".sizeOf_spec"].any
       (fun p => (n.toString.splitOn p).length > 1)

/-- (eqn roots) A compiled recursive body (`brecOn`/`WellFounded.fix`) hides its arms in
internal auxiliaries (`f._f`, `f._unary`) that the closure skips; such definitions are read through
their unfolding equation `f.eq_def` instead (statement text = the source-level arms). -/
def isRecBody (v : Expr) : Bool :=
  v.getUsedConstants.any fun c =>
    let s := c.toString
    (s.splitOn ".brecOn").length > 1 || (s.splitOn "._unary").length > 1 || (s.splitOn "._f").length > 1
    -- (guard fix) mutual well-founded recursion: the value is a `f._mutual (PSum.inl ..)`
    -- wrapper whose arms `skip` hides; read through `f.eq_def` instead
    || (s.splitOn "._mutual").length > 1
    || c == ``WellFounded.fix || c == ``WellFounded.Nat.fix

/-- The unfolding-equation statements of the recursive definitions among `names` (realized on demand). -/
def eqnTypes (names : Array Name) : MetaM (Std.HashMap Name Expr) := do
  let mut m : Std.HashMap Name Expr := {}
  for n in names do
    let some (.defnInfo d) := (← getEnv).find? n | continue
    unless isRecBody d.value do continue
    let some e ← getUnfoldEqnFor? n (nonRec := true) | throwError "no unfold equation for {n}"
    let some ei := (← getEnv).find? e | throwError "missing {e}"
    m := m.insert n ei.type
  return m

def usedOf (env : Environment) (ci : ConstantInfo) (eq : Std.HashMap Name Expr := {}) : Array Name := Id.run do
  let mut s := ci.type.getUsedConstants
  match ci with
  | .defnInfo d => s := s ++ (match eq.get? ci.name with | some t => t.getUsedConstants | none => d.value.getUsedConstants)
  | .opaqueInfo d => s := s ++ d.value.getUsedConstants
  | .inductInfo d => for c in d.ctors do if let some cc := env.find? c then s := s ++ cc.type.getUsedConstants
  | .ctorInfo d => s := s.push d.induct
  | .recInfo d => s := s.push d.getMajorInduct
  | _ => pure ()
  return s

/-- The meaning closure of the constants in `roots` (names of definitions, sorted). -/
def closure (roots : Array Name) (force : List Name := forceBody) : MetaM (Array Name) := do
  let mut seen : Std.HashSet Name := {}
  let mut out : Array Name := #[]
  let mut todo := roots.toList
  while h : todo ≠ [] do
    let n := todo.head h; todo := todo.tail
    if seen.contains n then continue
    seen := seen.insert n
    let env ← getEnv
    let some ci := env.find? n | continue
    let m := moduleOf env n
    if m.getRoot == `Init || m.getRoot == `Std || m.getRoot == `Lean || m == `_local then continue
    if ci matches .thmInfo _ then continue
    out := out.push n
    if isStop m && !force.contains n then continue
    let eq ← eqnTypes #[n]
    for c in usedOf env ci eq do if !skip c && !seen.contains c then todo := c :: todo
  return out.qsort (·.toString < ·.toString)

/-- (guard fix, h2) The classes of `skip`ped constants a rendered text may mention, each recognised
by the elaborator's own registry (no name sniffing).  Anything else `skip`ped that a rendered signature
or body mentions is a hidden body: the guard FAILS (`render`). -/
def skipClasses : List (String × (Environment → Name → Bool)) :=
  [("constructor", fun env c => (env.find? c) matches some (.ctorInfo _)),
   ("recursor", fun env c => (env.find? c) matches some (.recInfo _)),
   ("theorem", fun env c => (env.find? c) matches some (.thmInfo _)),
   ("matcher", fun env c => Meta.isMatcherCore env c),
   ("aux recursor", fun env c => isAuxRecursor env c),
   ("sparse casesOn", fun env c => isSparseCasesOn env c),
   ("noConfusion", fun env c => isNoConfusion env c)]

/-- (h2) Explicit per-name exceptions (none). -/
def skipAllow : List Name := []

def skipOK (env : Environment) (c : Name) : Bool :=
  skipAllow.contains c || skipClasses.any fun (_, p) => p env c

/-- (h2) The `skip`ped constants of `e` outside `skipClasses`/`skipAllow`. -/
def hidden (env : Environment) (e : Expr) : Array Name :=
  e.getUsedConstants.filter fun c => skip c && !skipOK env c

/-- Pretty-printer options for the fingerprint: FULL text, no depth/step elision (`⋯`).
Proofs stay elided (`pp.proofs false`): proofs carry no meaning, and printing them would make the
fingerprint move under proof refactors.  (guard fix, NOTE B of the R4 review: with the defaults,
`pp.deepTerms false` cut long telescopes such as `SysInvN.mk` at depth 50, so later fields' text
was not fingerprinted, only their constant names.) -/
def fullOpts (o : Options) : Options :=
  let o := Lean.pp.deepTerms.set o true
  let o := Lean.pp.deepTerms.threshold.set o 1000000
  let o := Lean.pp.maxSteps.set o 1000000000
  let o := Lean.pp.proofs.set o false
  Lean.pp.proofs.withType.set o false

/-- Deterministic text of a closure: signature of every constant; body of every expanded one. -/
def render (names : Array Name) (force : List Name := forceBody) : MetaM String :=
  withTheReader Core.Context (fun ctx => { ctx with options := fullOpts ctx.options, maxRecDepth := 100000 }) do
  let env ← getEnv
  let mut s := ""
  let mut bad : Array (Name × Name) := #[]
  for n in names do
    let some ci := env.find? n | continue
    s := s ++ s!"{n} : {← ppExpr ci.type}\n"
    for c in hidden env ci.type do bad := bad.push (n, c)
    unless stopC env force n do
      let pre := s.length
      match ci with
      | .defnInfo d =>
          match (← eqnTypes #[n]).get? n with
          | some t =>
              s := s ++ s!"  eq_def {← ppExpr t}\n"
              for c in hidden env t do bad := bad.push (n, c)
          | none =>
              s := s ++ s!"  := {← ppExpr d.value}\n"
              for c in hidden env d.value do bad := bad.push (n, c)
      | .inductInfo d => for c in d.ctors do
          if let some cc := env.find? c then
            s := s ++ s!"  | {c} : {← ppExpr cc.type}\n"
            for h in hidden env cc.type do bad := bad.push (n, h)
          -- (M0) a forced language-root structure also lists its field names (e.g. `Cfg`'s `cap`)
          if langRoots.contains n && isStructure env n then
            s := s ++ s!"  fields {getStructureFields env n}\n"
      | _ =>
          -- (check only) a forced root that is neither a definition nor an inductive (`opaque`, or a
          -- `partial def`, which compiles to an opaque constant) would be frozen by its SIGNATURE only
          if langRoots.contains n || force.contains n then
            throwError "premise guard: forced root {n} is not a definition or an inductive (opaque or partial?): \
              its body would not be printed"
      -- (check only) a forced root must print a body line (`  := `, `  eq_def `, `  | `)
      if langRoots.contains n || force.contains n then
        let out := (s.drop pre).toString
        unless (out.splitOn "  := ").length > 1 || (out.splitOn "  eq_def ").length > 1 || (out.splitOn "  | ").length > 1 do
          throwError "premise guard: forced root {n} rendered without a body line"
  -- (h2) hard check: no rendered text mentions a hidden (skipped, unclassified) constant
  unless bad.isEmpty do
    throwError "premise guard (h2): rendered text mentions skipped constants outside skipClasses/skipAllow \
      (a body the fingerprint would not show): {bad.toList.map fun (n, c) => s!"{n} -> {c}"}"
  return s

/-- (tie) The `ROOT <name> : <statement>` line of a theorem root: its OWN statement, under the
same pinned options as `render`, so a change of shape that keeps the constant set (swapped conjuncts, a
dropped premise, a weakened conclusion) moves the fingerprint.  h2 is run on the statement too. -/
def rootLine (f : Name) (ty : Expr) : MetaM String :=
  withTheReader Core.Context (fun ctx => { ctx with options := fullOpts ctx.options, maxRecDepth := 100000 }) do
  let env ← getEnv
  let bad := hidden env ty
  unless bad.isEmpty do
    throwError "premise guard (h2): the statement of root {f} mentions skipped constants outside \
      skipClasses/skipAllow: {bad.toList}"
  return s!"ROOT {f} : {← ppExpr ty}\n"

/-- One file per final: the closure of the constants of its STATEMENT (premises and conclusion); for a
theorem root the file starts with its `ROOT` statement line (`rootLine`). -/
def freeze (dir : String) (finals : List Name) : MetaM Unit := do
  let env ← getEnv
  IO.FS.createDirAll dir
  for n in forceBody do unless (env.find? n).isSome do throwError "missing forced {n}"
  for f in finals do
    let some ci := env.find? f | throwError "missing final {f}"
    let roots := if ci matches .thmInfo _ then ci.type.getUsedConstants.filter (!skip ·) else #[f]
    let names ← closure roots
    let hdr ← if ci matches .thmInfo _ then rootLine f ci.type else pure ""
    IO.FS.writeFile s!"{dir}/{f}.txt" (hdr ++ (← render names))
    logInfo m!"{f}: {names.size} constants"
  -- the language roots, each with its own text forced
  -- strict mode (set by the gate): `WM_GUARD_STRICT=1` makes an absent language root an error,
  -- except the names listed (one per line, `#` comments) in `lang_absent_ok.txt` in the working directory
  -- (the guard's tools/premises dir): M0 (pre-spawn) lists `CoreTerm.tailCheck`; the spawn-era freeze is empty.
  let strict := (← IO.getEnv "WM_GUARD_STRICT") == some "1"
  let hasOk ← System.FilePath.pathExists "lang_absent_ok.txt"
  let okText ← if strict && hasOk then IO.FS.readFile "lang_absent_ok.txt" else pure ""
  let okAbsent : List String :=
    ((okText.splitOn "\n").map (fun s => s.trimAscii.toString)).filter (fun l => l != "" && !l.startsWith "#")
  for f in langRoots do
    unless (env.find? f).isSome do
      if strict && !okAbsent.contains f.toString then
        throwError "missing language root {f} (WM_GUARD_STRICT=1; not in lang_absent_ok.txt)"
      IO.FS.writeFile s!"{dir}/lang.{f}.txt" "ABSENT\n"; logInfo m!"lang {f}: ABSENT"; continue
    let force := forceBody ++ [f]
    let names ← closure #[f] force
    -- a forced root must render non-empty and with its own text (a theorem or a skipped
    -- constant renders empty / bodiless): hard error, check only
    unless names.contains f do
      throwError "premise guard: language root {f} renders empty or without its own text (theorem or skipped constant?)"
    IO.FS.writeFile s!"{dir}/lang.{f}.txt" (← render names force)
    logInfo m!"lang {f}: {names.size} constants"
  -- (per-toolchain sets) the module layout of this run (guard.sh selects the premise set by toolchain and layout)
  IO.FS.writeFile s!"{dir}/.layout" s!"{layoutOf env finals}\n"
  -- (guard tool fix) completion marker, written last: the files of this run, one per line (guard.sh checks that
  -- each exists and is non-empty, and that no other `.txt` file is present)
  IO.FS.writeFile s!"{dir}/.expected"
    (String.join ((finals.map fun f => s!"{f}.txt\n") ++ (langRoots.map fun f => s!"lang.{f}.txt\n")))

end CheckPremises
