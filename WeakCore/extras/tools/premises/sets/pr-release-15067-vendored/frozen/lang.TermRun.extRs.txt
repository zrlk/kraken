TermRun.extRs : TermRun.Bounds → CoreTerm.ExtName → Nat → List TermRun.Word → List TermRun.Event → List (List TermRun.Word)
  := fun b n i ws tr =>
  match n with
  | CoreTerm.ExtName.devChoice w =>
    List.map (fun v => [TermRun.cst (WeakMem.Val.num v)]) (List.filter (fun x => decide (x < 2 ^ w.bits)) b.io)
  | x => n.resultsL i ws tr
