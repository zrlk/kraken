TermRun.curCfg : TermRun.Cfg → Nat → TermRun.TTh → String → CoreTerm.Expr → TermRun.Cfg
TermRun.extCfg : TermRun.Cfg →
  Nat → TermRun.TTh → String → CoreTerm.ExtName → CoreTerm.Expr → List TermRun.Word → List TermRun.Word → TermRun.Cfg
  := fun c i th x n e ws rs =>
  match n with
  | CoreTerm.ExtName.current => TermRun.curCfg c i th x e
  | x_1 =>
    { mem := c.mem, pairs := c.pairs,
      ths :=
        c.ths.set! i
          { term := e, env := (x, n.sig.snd, rs) :: th.env, σs := th.σs, stack := th.stack,
            vs := if n = CoreTerm.ExtName.park then TermRun.parkVs (Array.size c.mem) th else th.vs,
            promises := th.promises, nalloc := th.nalloc, undefs := th.undefs, tidLbl := th.tidLbl,
            permit := if n = CoreTerm.ExtName.park then none else th.permit, tys := th.tys },
      lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace ++ n.events ws rs }
