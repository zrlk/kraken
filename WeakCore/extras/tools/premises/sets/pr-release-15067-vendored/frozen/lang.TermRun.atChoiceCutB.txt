TermRun.atChoiceCutB : TermRun.Bounds → TermRun.TTh → Bool
  := fun b th =>
  match th.term with
  | CoreTerm.Expr.let_ x (CoreTerm.Op.ext (CoreTerm.ExtName.devChoice w) a) e =>
    th.promises.isEmpty && (List.filter (fun x => decide (x < 2 ^ w.bits)) b.io).isEmpty
  | x => false
