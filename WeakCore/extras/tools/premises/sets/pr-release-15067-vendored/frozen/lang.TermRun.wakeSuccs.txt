TermRun.wakeSuccs : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Bounds → TermRun.Cfg → Nat → List TermRun.Cfg
  := fun p k arch b c i =>
  match c.ths[i]? with
  | some th =>
    match th.term with
    | CoreTerm.Expr.let_ x (CoreTerm.Op.ext CoreTerm.ExtName.park a) e =>
      if th.permit.isSome = true then []
      else
        match TermRun.evalAt th.env a with
        | some (fst, ws) =>
          List.filter (fun c' => TermRun.certB p k arch b.solo c' i)
            [TermRun.extCfg c i th x CoreTerm.ExtName.park e ws []]
        | none => []
    | x => []
  | none => []
