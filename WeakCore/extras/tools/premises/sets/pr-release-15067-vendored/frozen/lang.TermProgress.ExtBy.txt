TermProgress.ExtBy : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun p k arch c i c' =>
  ∃ x n a e th rt ws rs,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e ∧
        (TermRun.ExtNeedsNoProm n th → th.promises = []) ∧
          TermRun.ExtEnabled n th ∧
            TermRun.evalAt th.env a = some (rt, ws) ∧
              n.Admits i ws c.trace rs ∧
                TermRun.Certified p k arch (TermRun.extCfg c i th x n e ws rs) i ∧
                  c' = TermRun.wake (TermRun.extCfg c i th x n e ws rs) i (TermRun.relView th.vs) n ws
