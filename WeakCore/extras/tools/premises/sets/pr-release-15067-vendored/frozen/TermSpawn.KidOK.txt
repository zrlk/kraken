  := fun x x_1 =>
  match x, x_1 with
  | CoreLang.Idx.lit a, CoreLang.Idx.lit b => CoreLang.Idx.lit (a + b)

TermClose.cl : (Nat → Nat) → (Nat → CoreLang.Ty) → Nat → Nat → CoreLang.Ty → CoreLang.Ty
  eq_def ∀ (ρ : Nat → Nat) (W : Nat → CoreLang.Ty) (c d : Nat) (x : CoreLang.Ty),
  TermClose.cl ρ W c d x =
    match x with
    | CoreLang.Ty.intS i => CoreLang.Ty.intS (TermClose.clI ρ d i)
    | a.prod b => (TermClose.cl ρ W c d a).prod (TermClose.cl ρ W c d b)
    | a.sum b => (TermClose.cl ρ W c d a).sum (TermClose.cl ρ W c d b)
    | a.code b k => (TermClose.cl ρ W c d a).code (TermClose.cl ρ W c d b) k
    | CoreLang.Ty.array i t => CoreLang.Ty.array (TermClose.clI ρ d i) (TermClose.cl ρ W c d t)
    | CoreLang.Ty.ptr μ t => CoreLang.Ty.ptr (TermClose.clM ρ d μ) (TermClose.cl ρ W c d t)
    | CoreLang.Ty.ptrOpt μ t => CoreLang.Ty.ptrOpt (TermClose.clM ρ d μ) (TermClose.cl ρ W c d t)
    | CoreLang.Ty.loc μ t => CoreLang.Ty.loc (TermClose.clM ρ d μ) (TermClose.cl ρ W c d t)
    | CoreLang.Ty.word r P => CoreLang.Ty.word (TermClose.clI ρ d r) P
    | CoreLang.Ty.tok r st =>
      CoreLang.Ty.tok (TermClose.clI ρ d r)
        (Option.map
          (fun x =>
            match x with
            | (P, i) => (P, TermClose.clI ρ d i))
          st)
    | CoreLang.Ty.word2 r P => CoreLang.Ty.word2 (TermClose.clI ρ d r) P
    | CoreLang.Ty.tok2 r P i j => CoreLang.Ty.tok2 (TermClose.clI ρ d r) P (TermClose.clI ρ d i) (TermClose.clI ρ d j)
    | CoreLang.Ty.epair r P => CoreLang.Ty.epair (TermClose.clI ρ d r) P
    | CoreLang.Ty.tokE r P e => CoreLang.Ty.tokE (TermClose.clI ρ d r) P (TermClose.clI ρ d e)
    | CoreLang.Ty.share r => CoreLang.Ty.share (TermClose.clI ρ d r)
    | CoreLang.Ty.cap s r => CoreLang.Ty.cap s (TermClose.clI ρ d r)
    | CoreLang.Ty.tvar k => if k < c then CoreLang.Ty.tvar k else W (k - c)
    | t.mu => (TermClose.cl ρ W (c + 1) d t).mu
    | CoreLang.Ty.tex κ t => CoreLang.Ty.tex κ (TermClose.cl ρ W (c + 1) d t)
    | t.iall => (TermClose.cl ρ W c (d + 1) t).iall
    | t.iex => (TermClose.cl ρ W c (d + 1) t).iex
    | t.refine ψ => (TermClose.cl ρ W c d t).refine (TermClose.clP ρ d ψ)
    | t => t
TermClose.clI : (Nat → Nat) → Nat → CoreLang.Idx → CoreLang.Idx
  eq_def ∀ (ρ : Nat → Nat) (d : Nat) (x : CoreLang.Idx),
  TermClose.clI ρ d x =
    match x with
    | CoreLang.Idx.lit n => CoreLang.Idx.lit n
    | CoreLang.Idx.var k => if k < d then CoreLang.Idx.var k else CoreLang.Idx.lit (ρ (k - d))
    | i.add j => TermClose.addF (TermClose.clI ρ d i) (TermClose.clI ρ d j)
    | i.sub j => TermClose.subF (TermClose.clI ρ d i) (TermClose.clI ρ d j)
    | CoreLang.Idx.smul n i => TermClose.smulF n (TermClose.clI ρ d i)
TermClose.clM : (Nat → Nat) → Nat → CoreLang.Mode → CoreLang.Mode
  := fun ρ d x =>
  match x with
  | CoreLang.Mode.rgn r => CoreLang.Mode.rgn (TermClose.clI ρ d r)
  | μ => μ
TermClose.clP : (Nat → Nat) → Nat → CoreLang.Pred → CoreLang.Pred
  eq_def ∀ (ρ : Nat → Nat) (d : Nat) (x : CoreLang.Pred),
  TermClose.clP ρ d x =
    match x with
    | CoreLang.Pred.tt => CoreLang.Pred.tt
    | CoreLang.Pred.le i j => CoreLang.Pred.le (TermClose.clI ρ d i) (TermClose.clI ρ d j)
    | CoreLang.Pred.eq i j => CoreLang.Pred.eq (TermClose.clI ρ d i) (TermClose.clI ρ d j)
    | p.and q => (TermClose.clP ρ d p).and (TermClose.clP ρ d q)
    | CoreLang.Pred.ini i => CoreLang.Pred.ini (TermClose.clI ρ d i)
TermClose.smulF : Nat → CoreLang.Idx → CoreLang.Idx
  := fun n x =>
  match x with
  | CoreLang.Idx.lit a => CoreLang.Idx.lit (n * a)
  | i => CoreLang.Idx.smul n i
TermClose.subF : CoreLang.Idx → CoreLang.Idx → CoreLang.Idx
  := fun x x_1 =>
  match x, x_1 with
  | CoreLang.Idx.lit a, CoreLang.Idx.lit b => CoreLang.Idx.lit (a - b)

TermClose.tb : Nat → CoreLang.Ty → Bool
  eq_def ∀ (c : Nat) (x : CoreLang.Ty),
  TermClose.tb c x =
    match x with
    | CoreLang.Ty.tvar k => decide (k < c)
    | a.prod b => TermClose.tb c a && TermClose.tb c b
    | a.sum b => TermClose.tb c a && TermClose.tb c b
    | a.code b k => TermClose.tb c a && TermClose.tb c b
    | CoreLang.Ty.array i t => TermClose.tb c t
    | CoreLang.Ty.ptr μ t => TermClose.tb c t
    | CoreLang.Ty.ptrOpt μ t => TermClose.tb c t
    | CoreLang.Ty.loc μ t => TermClose.tb c t
    | t.mu => TermClose.tb (c + 1) t
    | CoreLang.Ty.tex κ t => TermClose.tb (c + 1) t
    | t.iall => TermClose.tb c t
    | t.iex => TermClose.tb c t
    | t.refine ψ => TermClose.tb c t
    | x => true

TermSound.AtOff : CoreLang.Ty → Nat → CoreLang.Ty → Prop
  | TermSound.AtOff.here : ∀ {t : CoreLang.Ty}, TermSound.LayT t → TermSound.AtOff t 0 t
  | TermSound.AtOff.fst : ∀ {a b u : CoreLang.Ty} {o : Nat}, TermSound.AtOff a o u → TermSound.AtOff (a.prod b) o u
  | TermSound.AtOff.snd : ∀ {a b u : CoreLang.Ty} {o : Nat} {us : List TermRun.Acc},
  TermRun.accs a = some us → TermSound.AtOff b o u → TermSound.AtOff (a.prod b) (TermRun.nWords us + o) u
  | TermSound.AtOff.elem : ∀ {e u : CoreLang.Ty} {n j o : Nat} {us : List TermRun.Acc},
  TermRun.accs e = some us →
    j < n →
      TermSound.AtOff e o u → TermSound.AtOff (CoreLang.Ty.array (CoreLang.Idx.lit n) e) (j * TermRun.nWords us + o) u
  | TermSound.AtOff.mu : ∀ {ρ : Nat → Nat} {W : Nat → CoreLang.Ty} {s u : CoreLang.Ty} {o : Nat},
  (∀ (k : Nat), TermClose.Clo (W k)) →
    CoreLang.Ty.unguarded 0 s = false →
      TermSound.AtOff (TermClose.cl ρ W 0 0 (CoreTerm.unfold1 s.mu)) o u →
        TermSound.AtOff (TermClose.cl ρ W 0 0 s.mu) o u
  | TermSound.AtOff.fold : ∀ {ρ : Nat → Nat} {W : Nat → CoreLang.Ty} {s T : CoreLang.Ty} {o : Nat},
  (∀ (k : Nat), TermClose.Clo (W k)) →
    CoreLang.Ty.unguarded 0 s = false →
      TermRun.accs (TermClose.cl ρ W 0 0 s.mu) ≠ none →
        TermSound.AtOff T o (TermClose.cl ρ W 0 0 (CoreTerm.unfold1 s.mu)) →
          TermSound.AtOff T o (TermClose.cl ρ W 0 0 s.mu)

    P.clauses[c]? = some cl ∧ n0 = cl.s0 * k0 + cl.o0 ∧ n1 = cl.s1 * k1 + cl.o1 ∧ CoreTyping.Rel2Holds cl.rel k0 k1
TermSound.IniAt : (WeakMem.AllocLabel → Nat) → TermSound.Cell → WeakMem.Address → CoreLang.Ty → Prop
  := fun ini c a U =>
  ∀ (N : Nat) (e : CoreLang.Ty),
    c.ty = CoreLang.Ty.array (CoreLang.Idx.lit N) e →
      U = c.ty ∧ 0 = a.offset ∨
        ∃ us j o,
          TermRun.accs e = some us ∧ j < ini a.label ∧ j * TermRun.nWords us + o = a.offset ∧ TermSound.AtOff e o U
TermSound.IniR : (Nat → Nat) → (WeakMem.AllocLabel → Nat) → CoreLang.Pred → List TermRun.Word → Prop
  eq_def ∀ (ρ : Nat → Nat) (ini : WeakMem.AllocLabel → Nat) (x : CoreLang.Pred) (x_1 : List TermRun.Word),
  TermSound.IniR ρ ini x x_1 =
    match x, x_1 with
    | CoreLang.Pred.ini k, ws =>
      ∀ (a : WeakMem.Address), List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] → CoreTyping.ev ρ k ≤ ini a.label
    | p.and q, ws => TermSound.IniR ρ ini p ws ∧ TermSound.IniR ρ ini q ws
    | x, x_2 => True

          TermSound.etok (TermSound.lyt 0 rt) = TermSound.etok (TermSound.lyt 0 (TermRun.close σs t))
TermSound.LayT : CoreLang.Ty → Prop
  := fun t => TermRun.accs t ≠ none ∨ ∃ n e, t = CoreLang.Ty.array (CoreLang.Idx.lit n) e ∧ TermRun.accs e ≠ none
TermSound.ModeOK : Nat →
  (WeakMem.AllocLabel → Option WeakMem.Ts) →
    (WeakMem.AllocLabel → Prop) → CoreLang.Mode → WeakMem.AllocLabel → TermSound.Cell → List TermRun.Word → Prop
  := fun i P S x x_1 x_2 x_3 =>
  match x, x_1, x_2, x_3 with
  | CoreLang.Mode.own, x, c, x_4 => c.owner = some i
  | CoreLang.Mode.pub, l, x, ws => ∃ f, P l = some f ∧ (S l ∨ ∀ (w : TermRun.Word), w ∈ ws → f ≤ w.view)
  | CoreLang.Mode.rgn r, x, x_4, x_5 => True
TermSound.PayOK : CoreLang.Ty → List TermRun.Word → Prop
  := fun a p =>
  a.rawWords = some p.length ∧
    (TermRun.armK a = TermRun.ArmK.ptr → ∃ w adr, p = [w] ∧ w.val = WeakMem.Val.loc adr 0) ∧
      (TermRun.armK a = TermRun.ArmK.opt →
        ∃ w, p = [w] ∧ (w.val = WeakMem.Val.null ∨ ∃ adr, w.val = WeakMem.Val.loc adr 0))

TermSound.Shp.wc : TermSound.Shp → Option Nat
  eq_def ∀ (x : TermSound.Shp),
  x.wc =
    match x with
    | TermSound.Shp.leaf n => n
    | a.prod b => do
      let __do_lift ← a.wc
      let __do_lift_1 ← b.wc
      pure (__do_lift + __do_lift_1)
    | TermSound.Shp.iex n b => n
    | TermSound.Shp.fix sg w => some 1
    | TermSound.Shp.lanes sg w => some 1
    | TermSound.Shp.sum n R => n
    | TermSound.Shp.code A B => some 1
    | TermSound.Shp.tex n b => n
TermSound.SumV : (List TermRun.Word → Prop) → (List TermRun.Word → Prop) → CoreLang.Ty → CoreLang.Ty → List TermRun.Word → Prop
  := fun Va Vb a b ws =>
  (a.sum b).tier = CoreLang.Tier.boxed ∨
    ∃ j p arm,
      TermRun.decodeArm (a.sum b).tier (TermRun.armsOf (a.sum b)) ws = some (j, p) ∧
        (TermRun.armsOf (a.sum b))[j]? = some arm ∧
          TermSound.PayOK arm p ∧
            (∃ q, TermRun.encodeArm (a.sum b).tier (TermRun.armsOf (a.sum b)) j p = some q ∧ TermSound.VEq q ws) ∧
              (j < (TermRun.armsOf a).length →
                  ∃ q, TermRun.ofArm a j p = some q ∧ TermSound.ShapeC a q ∧ TermSound.wc a ≠ none ∧ Va q) ∧
                ((TermRun.armsOf a).length ≤ j →
                  ∃ q,
                    TermRun.ofArm b (j - (TermRun.armsOf a).length) p = some q ∧
                      TermSound.ShapeC b q ∧ TermSound.wc b ≠ none ∧ Vb q)

TermSound.VR : (Nat → Nat) → TermSound.TEnv → CoreLang.Ty → TermSound.Shp → List TermRun.Word → Prop
  eq_def ∀ (ρ : Nat → Nat) (η : TermSound.TEnv) (x : CoreLang.Ty) (x_1 : TermSound.Shp) (x_2 : List TermRun.Word),
  TermSound.VR ρ η x x_1 x_2 =
    match x, x_1, x_2 with
    | CoreLang.Ty.intS i, x, ws => List.map (fun x => x.val) ws = [WeakMem.Val.num (CoreTyping.ev ρ i)]
    | CoreLang.Ty.int, x, ws => ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n]
    | CoreLang.Ty.opq o, x, ws => ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n]
    | CoreLang.Ty.tid, x, ws =>
      ∃ l c,
        List.map (fun x => x.val) ws = [WeakMem.Val.loc { label := l, offset := 0 } 0] ∧
          η.sto l = some c ∧ c.ty = CoreLang.Ty.unit ∧ TermSound.ModeOK η.tid η.pfr η.seen CoreLang.Mode.pub l c ws
    | CoreLang.Ty.fix sg w, rt, ws =>
      rt = TermSound.Shp.fix sg w ∧ ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n] ∧ n < TermRun.pw2 w
    | CoreLang.Ty.lanes sg w, rt, ws =>
      rt = TermSound.Shp.lanes sg w ∧ ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n]
    | t.refine ψ, rt, ws => TermSound.VR ρ η t rt ws ∧ TermSound.IniR ρ (η.selIni t) ψ ws
    | b.iex, rt, ws => ∃ n rb w, rt = TermSound.Shp.iex n rb ∧ TermSound.VR (TermSound.vcons w ρ) η b rb ws
    | a.prod b, rt, ws =>
      ∃ ra rb,
        rt = ra.prod rb ∧
          ∀ (n : Nat),
            ra.wc = some n →
              n ≤ ws.length ∧ TermSound.VR ρ η a ra (List.take n ws) ∧ TermSound.VR ρ η b rb (List.drop n ws)
    | a.sum b, rt, ws =>
      ∃ n Ra Rb,
        rt = TermSound.Shp.sum n (Ra.sum Rb) ∧
          (((Ra.sum Rb).tier = CoreLang.Tier.boxed ∧
              ∃ x s B,
                List.map (fun x => x.val) ws = [WeakMem.Val.loc x 0] ∧
                  x.offset = 0 ∧
                    η.box x.label = some B ∧
                      n = some (s + 1) ∧
                        (s ≤ B.tag ∧ (TermRun.armsOf (Ra.sum Rb))[B.tag - s]?.bind TermSound.wc = some B.pay.length) ∧
                          ((η.seen x.label ∧ ∃ f, η.pfr x.label = some f ∧ B.fr ≤ f) ∨
                              ∀ (w : TermRun.Word), w ∈ ws → B.fr ≤ w.view) ∧
                            (B.tag - s < (TermRun.armsOf Ra).length →
                                (TermRun.isBoxed Ra = true →
                                    TermSound.VR ρ η a (TermSound.vsh s (TermSound.shp Ra)) ws) ∧
                                  (TermRun.isBoxed Ra = false →
                                    ∃ q,
                                      TermRun.ofArm Ra (B.tag - s) B.pay = some q ∧
                                        TermSound.ShapeC Ra q ∧
                                          TermSound.wc Ra ≠ none ∧
                                            TermSound.Valid Ra q ∧ TermSound.VR ρ η.unsee a (TermSound.shp Ra) q)) ∧
                              ((TermRun.armsOf Ra).length ≤ B.tag - s →
                                (TermRun.isBoxed Rb = true →
                                    TermSound.VR ρ η b
                                      (TermSound.vsh (s + (TermRun.armsOf Ra).length) (TermSound.shp Rb)) ws) ∧
                                  (TermRun.isBoxed Rb = false →
                                    ∃ q,
                                      TermRun.ofArm Rb (B.tag - s - (TermRun.armsOf Ra).length) B.pay = some q ∧
                                        TermSound.ShapeC Rb q ∧
                                          TermSound.wc Rb ≠ none ∧
                                            TermSound.Valid Rb q ∧ TermSound.VR ρ η.unsee b (TermSound.shp Rb) q))) ∨
            ∃ j p,
              TermRun.decodeArm (Ra.sum Rb).tier (TermRun.armsOf (Ra.sum Rb)) ws = some (j, p) ∧
                (j < (TermRun.armsOf Ra).length →
                    ∃ q, TermRun.ofArm Ra j p = some q ∧ TermSound.VR ρ η a (TermSound.shp Ra) q) ∧
                  ((TermRun.armsOf Ra).length ≤ j →
                    ∃ q,
                      TermRun.ofArm Rb (j - (TermRun.armsOf Ra).length) p = some q ∧
                        TermSound.VR ρ η b (TermSound.shp Rb) q))
    | CoreLang.Ty.tvar k, rt, ws => ∃ ws', TermSound.WLe ws' ws ∧ η.get k rt ws'
    | σ.code τ k, rt, ws =>
      ∃ i d,
        List.map (fun x => x.val) ws = [WeakMem.Val.num i] ∧
          η.prog[i]? = some d ∧
            List.find? (fun x => x.name == d.name) η.prog = some d ∧
              d.tparams = [] ∧
                d.idx = [] ∧
                  TermClose.cl ρ η.wit 0 0 σ = TermSound.clD d.argTy ∧
                    TermClose.cl ρ η.wit 0 0 τ = TermSound.clD d.retTy
    | CoreLang.Ty.ptr μ u, x, ws =>
      ∃ a c,
        List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
          a.offset = 0 ∧
            η.sto a.label = some c ∧
              TermSound.Fd c.ty (TermClose.cl ρ η.wit 0 0 u) ∧
                TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧ TermRun.accs c.ty ≠ none
    | CoreLang.Ty.loc μ u, x, ws =>
      (List.map (fun x => x.val) ws = [WeakMem.Val.null] ∧
          ∃ e, TermSound.Fd (CoreLang.Ty.array (CoreLang.Idx.lit 0) e) (TermClose.cl ρ η.wit 0 0 u)) ∨
        ∃ a c U,
          List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
            η.sto a.label = some c ∧
              TermSound.AtOff c.ty a.offset U ∧
                TermSound.Fd U (TermClose.cl ρ η.wit 0 0 u) ∧
                  TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧
                    TermSound.IniAt (TermSound.modeIni μ η.ini η.rini) c a U
    | CoreLang.Ty.ptrOpt μ u, x, ws =>
      List.map (fun x => x.val) ws = [WeakMem.Val.null] ∨
        ∃ a c,
          List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
            a.offset = 0 ∧
              η.sto a.label = some c ∧
                TermSound.Fd c.ty (TermClose.cl ρ η.wit 0 0 u) ∧
                  TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧ TermRun.accs c.ty ≠ none
    | CoreLang.Ty.void, x, x_3 => False
    | CoreLang.Ty.tex κ b, rt, ws =>
      ∃ n rb s, rt = TermSound.Shp.tex n rb ∧ TermClose.Clo s ∧ TermSound.WitOK κ s ∧ TermSound.VR ρ (η.push0 s) b rb ws
    | b.mu, rt, ws => TermSound.VR ρ (η.push0 (TermClose.cl ρ η.wit 0 0 b.mu)) b rt ws
    | b.iall, rt, ws => ∀ (n : Nat), TermSound.VR (TermSound.vcons n ρ) η b rt ws
    | CoreLang.Ty.word r P, x, ws =>
      CoreTerm.pwf P = true ∧
        (P.ghost = false → ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n] ∧ TermSound.InP P n)
    | CoreLang.Ty.word2 r P, x, ws =>
      CoreTerm.pwf2 P = true ∧
        ∃ n0 n1, List.map (fun x => x.val) ws = [WeakMem.Val.num n0, WeakMem.Val.num n1] ∧ TermSound.InP2 P n0 n1
    | CoreLang.Ty.epair r P, x, ws =>
      CoreTerm.pwfE P = true ∧ ∃ n0 n1, List.map (fun x => x.val) ws = [WeakMem.Val.num n0, WeakMem.Val.num n1]
    | x, x_3, x_4 => True
TermSound.VRR : (Nat → Nat) → TermSound.TEnv → CoreLang.Ty → TermSound.Shp → List TermRun.Word → TermSound.Res → Prop
  eq_def ∀ (ρ : Nat → Nat) (η : TermSound.TEnv) (x : CoreLang.Ty) (x_1 : TermSound.Shp) (x_2 : List TermRun.Word)
  (x_3 : TermSound.Res),
  TermSound.VRR ρ η x x_1 x_2 x_3 =
    match x, x_1, x_2, x_3 with
    | CoreLang.Ty.unit, x, x_4, R => R = TermSound.Res.zero
    | CoreLang.Ty.intS i, x, ws, R =>
      List.map (fun x => x.val) ws = [WeakMem.Val.num (CoreTyping.ev ρ i)] ∧ R = TermSound.Res.zero
    | CoreLang.Ty.int, x, ws, R => (∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n]) ∧ R = TermSound.Res.zero
    | CoreLang.Ty.opq o, x, ws, R =>
      TermSound.VR ρ η (CoreLang.Ty.opq o) (TermSound.Shp.leaf (some 1)) ws ∧ R = TermSound.Res.zero
    | CoreLang.Ty.tid, x, ws, R =>
      TermSound.VR ρ η CoreLang.Ty.tid (TermSound.Shp.leaf (some 1)) ws ∧ R = TermSound.Res.zero
    | CoreLang.Ty.fix sg w, rt, ws, R =>
      (rt = TermSound.Shp.fix sg w ∧ ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n] ∧ n < TermRun.pw2 w) ∧
        R = TermSound.Res.zero
    | CoreLang.Ty.lanes sg w, rt, ws, R =>
      (rt = TermSound.Shp.lanes sg w ∧ ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n]) ∧ R = TermSound.Res.zero
    | t.refine ψ, rt, ws, R => TermSound.VRR ρ η t rt ws R ∧ TermSound.IniR ρ (η.selIni t) ψ ws
    | b.iex, rt, ws, R => ∃ n rb w, rt = TermSound.Shp.iex n rb ∧ TermSound.VRR (TermSound.vcons w ρ) η b rb ws R
    | a.prod b, rt, ws, R =>
      ∃ ra rb,
        rt = ra.prod rb ∧
          ∃ Ra Rb,
            R = Ra + Rb ∧
              (ra.wc = none → R = TermSound.Res.zero) ∧
                ∀ (n : Nat),
                  ra.wc = some n →
                    n ≤ ws.length ∧
                      TermSound.VRR ρ η a ra (List.take n ws) Ra ∧ TermSound.VRR ρ η b rb (List.drop n ws) Rb
    | a.sum b, rt, ws, R =>
      ∃ n Ra Rb,
        rt = TermSound.Shp.sum n (Ra.sum Rb) ∧
          (((Ra.sum Rb).tier = CoreLang.Tier.boxed ∧
              ∃ x s B,
                List.map (fun x => x.val) ws = [WeakMem.Val.loc x 0] ∧
                  x.offset = 0 ∧
                    η.box x.label = some B ∧
                      n = some (s + 1) ∧
                        (s ≤ B.tag ∧ (TermRun.armsOf (Ra.sum Rb))[B.tag - s]?.bind TermSound.wc = some B.pay.length) ∧
                          ((η.seen x.label ∧ ∃ f, η.pfr x.label = some f ∧ B.fr ≤ f) ∨
                              ∀ (w : TermRun.Word), w ∈ ws → B.fr ≤ w.view) ∧
                            (B.tag - s < (TermRun.armsOf Ra).length →
                                (TermRun.isBoxed Ra = true →
                                    TermSound.VRR ρ η a (TermSound.vsh s (TermSound.shp Ra)) ws R) ∧
                                  (TermRun.isBoxed Ra = false →
                                    ∃ q,
                                      TermRun.ofArm Ra (B.tag - s) B.pay = some q ∧
                                        TermSound.ShapeC Ra q ∧
                                          TermSound.wc Ra ≠ none ∧
                                            TermSound.Valid Ra q ∧ TermSound.VRR ρ η.unsee a (TermSound.shp Ra) q R)) ∧
                              ((TermRun.armsOf Ra).length ≤ B.tag - s →
                                (TermRun.isBoxed Rb = true →
                                    TermSound.VRR ρ η b
                                      (TermSound.vsh (s + (TermRun.armsOf Ra).length) (TermSound.shp Rb)) ws R) ∧
                                  (TermRun.isBoxed Rb = false →
                                    ∃ q,
                                      TermRun.ofArm Rb (B.tag - s - (TermRun.armsOf Ra).length) B.pay = some q ∧
                                        TermSound.ShapeC Rb q ∧
                                          TermSound.wc Rb ≠ none ∧
                                            TermSound.Valid Rb q ∧ TermSound.VRR ρ η.unsee b (TermSound.shp Rb) q R))) ∨
            ∃ j p,
              TermRun.decodeArm (Ra.sum Rb).tier (TermRun.armsOf (Ra.sum Rb)) ws = some (j, p) ∧
                (j < (TermRun.armsOf Ra).length →
                    ∃ q, TermRun.ofArm Ra j p = some q ∧ TermSound.VRR ρ η a (TermSound.shp Ra) q R) ∧
                  ((TermRun.armsOf Ra).length ≤ j →
                    ∃ q,
                      TermRun.ofArm Rb (j - (TermRun.armsOf Ra).length) p = some q ∧
                        TermSound.VRR ρ η b (TermSound.shp Rb) q R))
    | CoreLang.Ty.tvar k, rt, ws, R => ∃ ws', TermSound.WLe ws' ws ∧ η.rrs k rt ws' R
    | σ.code τ k, x, ws, R =>
      (∃ i d,
          List.map (fun x => x.val) ws = [WeakMem.Val.num i] ∧
            η.prog[i]? = some d ∧
              List.find? (fun x => x.name == d.name) η.prog = some d ∧
                d.tparams = [] ∧
                  d.idx = [] ∧
                    TermClose.cl ρ η.wit 0 0 σ = TermSound.clD d.argTy ∧
                      TermClose.cl ρ η.wit 0 0 τ = TermSound.clD d.retTy) ∧
        R = TermSound.Res.zero
    | CoreLang.Ty.ptr μ u, x, ws, R =>
      (∃ a c,
          List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
            a.offset = 0 ∧
              η.sto a.label = some c ∧
                TermSound.Fd c.ty (TermClose.cl ρ η.wit 0 0 u) ∧
                  TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧ TermRun.accs c.ty ≠ none) ∧
        R = TermSound.ownW μ ws
    | CoreLang.Ty.loc μ u, x, ws, R =>
      ((List.map (fun x => x.val) ws = [WeakMem.Val.null] ∧
            ∃ e, TermSound.Fd (CoreLang.Ty.array (CoreLang.Idx.lit 0) e) (TermClose.cl ρ η.wit 0 0 u)) ∨
          ∃ a c U,
            List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
              η.sto a.label = some c ∧
                TermSound.AtOff c.ty a.offset U ∧
                  TermSound.Fd U (TermClose.cl ρ η.wit 0 0 u) ∧
                    TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧
                      TermSound.IniAt (TermSound.modeIni μ η.ini η.rini) c a U) ∧
        R = TermSound.ownW μ ws
    | CoreLang.Ty.ptrOpt μ u, x, ws, R =>
      (List.map (fun x => x.val) ws = [WeakMem.Val.null] ∨
          ∃ a c,
            List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0] ∧
              a.offset = 0 ∧
                η.sto a.label = some c ∧
                  TermSound.Fd c.ty (TermClose.cl ρ η.wit 0 0 u) ∧
                    TermSound.ModeOK η.tid η.pfr η.seen μ a.label c ws ∧ TermRun.accs c.ty ≠ none) ∧
        R = TermSound.ownW μ ws
    | CoreLang.Ty.void, x, x_4, x_5 => False
    | CoreLang.Ty.tex κ b, rt, ws, R =>
      ∃ n rb s,
        rt = TermSound.Shp.tex n rb ∧ TermClose.Clo s ∧ TermSound.WitOK κ s ∧ TermSound.VRR ρ (η.push0 s) b rb ws R
    | b.mu, rt, ws, R => TermSound.VRR ρ (η.push0 (TermClose.cl ρ η.wit 0 0 b.mu)) b rt ws R
    | b.iall, rt, ws, R => ∀ (n : Nat), TermSound.VRR (TermSound.vcons n ρ) η b rt ws R
    | CoreLang.Ty.word r P, x, ws, R =>
      (CoreTerm.pwf P = true ∧
          (P.ghost = false → ∃ n, List.map (fun x => x.val) ws = [WeakMem.Val.num n] ∧ TermSound.InP P n)) ∧
        (TermSound.WordHolds P (List.map (fun x => x.val) ws) ∧
            R = TermSound.Res.one (TermSound.Key.rg (CoreTyping.ev ρ r)) + TermSound.wdW P (CoreTyping.ev ρ r) ∨
          ¬TermSound.WordHolds P (List.map (fun x => x.val) ws) ∧ R = TermSound.wdW P (CoreTyping.ev ρ r))
    | CoreLang.Ty.word2 r P, x, ws, R =>
      (CoreTerm.pwf2 P = true ∧
          ∃ n0 n1, List.map (fun x => x.val) ws = [WeakMem.Val.num n0, WeakMem.Val.num n1] ∧ TermSound.InP2 P n0 n1) ∧
        (TermSound.Word2Holds P (List.map (fun x => x.val) ws) ∧
            R =
              TermSound.Res.one (TermSound.Key.rg (CoreTyping.ev ρ r)) +
                TermSound.Res.one (TermSound.Key.wd (CoreTyping.ev ρ r)) ∨
          ¬TermSound.Word2Holds P (List.map (fun x => x.val) ws) ∧
            R = TermSound.Res.one (TermSound.Key.wd (CoreTyping.ev ρ r)))
    | CoreLang.Ty.epair r P, x, ws, R =>
      (CoreTerm.pwfE P = true ∧ ∃ n0 n1, List.map (fun x => x.val) ws = [WeakMem.Val.num n0, WeakMem.Val.num n1]) ∧
        R = TermSound.Res.zero
    | CoreLang.Ty.cap s r, x, x_4, R => R = TermSound.Res.zero
    | CoreLang.Ty.tok r st, x, x_4, R =>
      R = TermSound.Res.one (TermSound.Key.rg (CoreTyping.ev ρ r)) + TermSound.tokWd (CoreTyping.ev ρ r) st
    | CoreLang.Ty.tok2 r P i0 i1, x, x_4, R => R = TermSound.Res.one (TermSound.Key.rg (CoreTyping.ev ρ r))
    | CoreLang.Ty.tokE r P e, x, x_4, R =>
      R = TermSound.Res.one (TermSound.Key.ep (CoreTyping.ev ρ r) (CoreTyping.ev ρ e))
    | CoreLang.Ty.share r, x, x_4, R => R = TermSound.Res.one (TermSound.Key.sh (CoreTyping.ev ρ r))
    | CoreLang.Ty.array i τ, x, x_4, R => R = TermSound.Res.zero
TermSound.Valid : CoreLang.Ty → List TermRun.Word → Prop
  eq_def ∀ (x : CoreLang.Ty) (x_1 : List TermRun.Word),
  TermSound.Valid x x_1 =
    match x, x_1 with
    | CoreLang.Ty.ptr μ τ, ws => ∃ a, List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0]
    | CoreLang.Ty.ptrOpt μ τ, ws =>
      List.map (fun x => x.val) ws = [WeakMem.Val.null] ∨ ∃ a, List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0]
    | CoreLang.Ty.loc μ τ, ws =>
      List.map (fun x => x.val) ws = [WeakMem.Val.null] ∨ ∃ a, List.map (fun x => x.val) ws = [WeakMem.Val.loc a 0]
    | a.prod b, ws =>
      ∀ (n : Nat), TermSound.wc a = some n → TermSound.Valid a (List.take n ws) ∧ TermSound.Valid b (List.drop n ws)
    | a.sum b, ws => TermSound.SumV (TermSound.Valid a) (TermSound.Valid b) a b ws
    | CoreLang.Ty.void, x => False
    | t.refine ψ, ws => TermSound.Valid t ws
    | t.iall, ws => TermSound.Valid t ws
    | t.iex, ws => TermSound.Valid t ws
    | t.mu, ws => TermSound.Valid t ws
    | CoreLang.Ty.tex κ t, ws => TermSound.Valid t ws
    | x, x_2 => True
TermSound.WLe : List TermRun.Word → List TermRun.Word → Prop
  eq_def ∀ (x x_1 : List TermRun.Word),
  TermSound.WLe x x_1 =
    match x, x_1 with
    | [], [] => True
    | w :: ws, w' :: ws' => TermSound.WLe1 w w' ∧ TermSound.WLe ws ws'
    | x, x_2 => False
TermSound.WLe1 : TermRun.Word → TermRun.Word → Prop
  := fun w w' => w.val = w'.val ∧ ∀ (a : WeakMem.Address) (g : Nat), w.val = WeakMem.Val.loc a g → w.view ≤ w'.view
TermSound.WitOK : CoreLang.Kind → CoreLang.Ty → Prop
  := fun κ s =>
  s.arrFree = true ∧
    (κ.share = true → s.coupledOK = true) ∧
      (κ.copy = true → s.affine = false) ∧
        (κ.noOwn = true → s.noOwn = true) ∧
          (κ.share = true → TermSound.rgnFree s = true) ∧ TermTv.noIallV s = true ∧ TermTv.mg s = true
TermSound.Word2Holds : CoreLang.Proto2 → List WeakMem.Val → Prop
  := fun P vs =>
  ∃ n0 n1 c cl k0 k1,
    vs = [WeakMem.Val.num n0, WeakMem.Val.num n1] ∧
      P.clauses[c]? = some cl ∧
        cl.holds = true ∧ n0 = cl.s0 * k0 + cl.o0 ∧ n1 = cl.s1 * k1 + cl.o1 ∧ CoreTyping.Rel2Holds cl.rel k0 k1
TermSound.WordHolds : CoreLang.Proto → List WeakMem.Val → Prop
  := fun P vs =>
  P.ghost = false ∧
    ∃ n c cl k, vs = [WeakMem.Val.num n] ∧ P.clauses[c]? = some cl ∧ cl.holds = true ∧ n = cl.scale * k + cl.off
TermSound.anyB : (Nat → Bool) → Nat → Bool
  := fun m d => (List.range d).any m
TermSound.clD : CoreLang.Ty → CoreLang.Ty
  := fun A => TermClose.cl (fun x => 0) (fun x => CoreLang.Ty.void) 0 0 A
TermSound.etok : CoreLang.Ty → CoreLang.Ty
  eq_def ∀ (x : CoreLang.Ty),
  TermSound.etok x =
    match x with
    | CoreLang.Ty.tok r st => CoreLang.Ty.unit
    | CoreLang.Ty.tok2 r P i0 i1 => CoreLang.Ty.unit
    | CoreLang.Ty.tokE r P e => CoreLang.Ty.unit
    | CoreLang.Ty.share r => CoreLang.Ty.unit
    | a.prod b => (TermSound.etok a).prod (TermSound.etok b)
    | a.sum b => (TermSound.etok a).sum (TermSound.etok b)
    | a.code b k => (TermSound.etok a).code (TermSound.etok b) k
    | CoreLang.Ty.array i t => CoreLang.Ty.array i (TermSound.etok t)
    | CoreLang.Ty.ptr μ t => CoreLang.Ty.ptr μ (TermSound.etok t)
    | CoreLang.Ty.ptrOpt μ t => CoreLang.Ty.ptrOpt μ (TermSound.etok t)
    | CoreLang.Ty.loc μ t => CoreLang.Ty.loc μ (TermSound.etok t)
    | t.refine ψ => (TermSound.etok t).refine ψ
    | t => t
TermSound.idxC : Nat → CoreLang.Idx → CoreLang.Idx
  := fun d i => TermSound.ofMs (fun k => CoreLang.Idx.mentions k i) d
TermSound.instDecidableEqKey : DecidableEq TermSound.Key
  := TermSound.instDecidableEqKey.decEq
TermSound.instDecidableEqKey.decEq : (x x_1 : TermSound.Key) → Decidable (x = x_1)
  := fun x x_1 =>
  match x, x_1 with
  | TermSound.Key.rg a, TermSound.Key.rg b => if h : a = b then h ▸ isTrue ⋯ else isFalse ⋯
  | TermSound.Key.rg n, TermSound.Key.ep n_1 j => isFalse ⋯
  | TermSound.Key.rg n, TermSound.Key.sh n_1 => isFalse ⋯
  | TermSound.Key.rg n, TermSound.Key.wd n_1 => isFalse ⋯
  | TermSound.Key.rg n, TermSound.Key.own l => isFalse ⋯
  | TermSound.Key.ep n j, TermSound.Key.rg n_1 => isFalse ⋯
  | TermSound.Key.ep a a_1, TermSound.Key.ep b b_1 =>
    if h : a = b then h ▸ if h : a_1 = b_1 then h ▸ isTrue ⋯ else isFalse ⋯ else isFalse ⋯
  | TermSound.Key.ep n j, TermSound.Key.sh n_1 => isFalse ⋯
  | TermSound.Key.ep n j, TermSound.Key.wd n_1 => isFalse ⋯
  | TermSound.Key.ep n j, TermSound.Key.own l => isFalse ⋯
  | TermSound.Key.sh n, TermSound.Key.rg n_1 => isFalse ⋯
  | TermSound.Key.sh n, TermSound.Key.ep n_1 j => isFalse ⋯
  | TermSound.Key.sh a, TermSound.Key.sh b => if h : a = b then h ▸ isTrue ⋯ else isFalse ⋯
  | TermSound.Key.sh n, TermSound.Key.wd n_1 => isFalse ⋯
  | TermSound.Key.sh n, TermSound.Key.own l => isFalse ⋯
  | TermSound.Key.wd n, TermSound.Key.rg n_1 => isFalse ⋯
  | TermSound.Key.wd n, TermSound.Key.ep n_1 j => isFalse ⋯
  | TermSound.Key.wd n, TermSound.Key.sh n_1 => isFalse ⋯
  | TermSound.Key.wd a, TermSound.Key.wd b => if h : a = b then h ▸ isTrue ⋯ else isFalse ⋯
  | TermSound.Key.wd n, TermSound.Key.own l => isFalse ⋯
  | TermSound.Key.own l, TermSound.Key.rg n => isFalse ⋯
  | TermSound.Key.own l, TermSound.Key.ep n j => isFalse ⋯
  | TermSound.Key.own l, TermSound.Key.sh n => isFalse ⋯
  | TermSound.Key.own l, TermSound.Key.wd n => isFalse ⋯
  | TermSound.Key.own a, TermSound.Key.own b => if h : a = b then h ▸ isTrue ⋯ else isFalse ⋯
TermSound.lnorm : CoreLang.Ty → CoreLang.Ty
  eq_def ∀ (x : CoreLang.Ty),
  TermSound.lnorm x =
    match x with
    | CoreLang.Ty.intS i => CoreLang.Ty.int
    | a.prod b => (TermSound.lnorm a).prod (TermSound.lnorm b)
    | a.sum b => (TermSound.lnorm a).sum (TermSound.lnorm b)
    | σ.code τ k => CoreLang.Ty.unit.code CoreLang.Ty.unit CoreLang.CodeKind.plain
    | CoreLang.Ty.array i τ => CoreLang.Ty.array (CoreLang.Idx.lit 0) CoreLang.Ty.unit
    | CoreLang.Ty.ptr μ τ => CoreLang.Ty.ptrOpt CoreLang.Mode.pub CoreLang.Ty.unit
    | CoreLang.Ty.ptrOpt μ τ => CoreLang.Ty.ptrOpt CoreLang.Mode.pub CoreLang.Ty.unit
    | CoreLang.Ty.loc μ τ => CoreLang.Ty.ptrOpt CoreLang.Mode.pub CoreLang.Ty.unit
    | CoreLang.Ty.word r P => CoreLang.Ty.word (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tok r st => CoreLang.Ty.unit
    | CoreLang.Ty.word2 r P => CoreLang.Ty.word2 (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tok2 r P i0 i1 => CoreLang.Ty.unit
    | CoreLang.Ty.epair r P => CoreLang.Ty.epair (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tokE r P e => CoreLang.Ty.unit
    | CoreLang.Ty.share r => CoreLang.Ty.unit
    | CoreLang.Ty.cap s r => CoreLang.Ty.unit
    | t.mu => (TermSound.lnorm t).mu
    | CoreLang.Ty.tex κ t => CoreLang.Ty.tex κ (TermSound.lnorm t)
    | t.iall => (TermSound.lnorm t).iall
    | t.iex => (TermSound.lnorm t).iex
    | t.refine ψ => TermSound.lnorm t
    | CoreLang.Ty.tvar k => CoreLang.Ty.tvar 0
    | t => t
TermSound.lyt : Nat → CoreLang.Ty → CoreLang.Ty
  eq_def ∀ (d : Nat) (x : CoreLang.Ty),
  TermSound.lyt d x =
    match x with
    | CoreLang.Ty.int => CoreLang.Ty.intS (CoreLang.Idx.lit 0)
    | CoreLang.Ty.intS i => CoreLang.Ty.intS (TermSound.idxC d i)
    | a.prod b => (TermSound.lyt d a).prod (TermSound.lyt d b)
    | a.sum b => (TermSound.lyt d a).sum (TermSound.lyt d b)
    | a.code b k => (TermSound.lyt d a).code (TermSound.lyt d b) k
    | CoreLang.Ty.array i t => CoreLang.Ty.array (TermSound.idxC d i) (TermSound.lyt d t)
    | CoreLang.Ty.ptr μ t => CoreLang.Ty.ptr (TermSound.modeC d μ) (TermSound.lyt d t)
    | CoreLang.Ty.ptrOpt μ t => CoreLang.Ty.ptrOpt (TermSound.modeC d μ) (TermSound.lyt d t)
    | CoreLang.Ty.loc μ t => CoreLang.Ty.loc (TermSound.modeC d μ) (TermSound.lyt d t)
    | CoreLang.Ty.word r P => CoreLang.Ty.word (TermSound.idxC d r) P
    | CoreLang.Ty.tok r st =>
      CoreLang.Ty.tok (TermSound.idxC d r)
        (Option.map
          (fun x =>
            match x with
            | (P, i) => (P, TermSound.idxC d i))
          st)
    | CoreLang.Ty.word2 r P => CoreLang.Ty.word2 (TermSound.idxC d r) P
    | CoreLang.Ty.tok2 r P i j => CoreLang.Ty.tok2 (TermSound.idxC d r) P (TermSound.idxC d i) (TermSound.idxC d j)
    | CoreLang.Ty.epair r P => CoreLang.Ty.epair (TermSound.idxC d r) P
    | CoreLang.Ty.tokE r P i => CoreLang.Ty.tokE (TermSound.idxC d r) P (TermSound.idxC d i)
    | CoreLang.Ty.share r => CoreLang.Ty.share (TermSound.idxC d r)
    | CoreLang.Ty.cap s r => CoreLang.Ty.cap s (TermSound.idxC d r)
    | t.mu => (TermSound.lyt d t).mu
    | CoreLang.Ty.tex κ t => CoreLang.Ty.tex κ (TermSound.lyt d t)
    | t.iall => (TermSound.lyt (d + 1) t).iall
    | t.iex => (TermSound.lyt (d + 1) t).iex
    | t.refine ψ =>
      if TermSound.anyB (fun k => CoreLang.Pred.mentions k ψ) d = true then
        (TermSound.lyt d t).refine (CoreLang.Pred.ini (TermSound.ofMs (fun k => CoreLang.Pred.mentions k ψ) d))
      else TermSound.lyt d t
    | t => t
TermSound.modeC : Nat → CoreLang.Mode → CoreLang.Mode
  := fun d x =>
  match x with
  | CoreLang.Mode.rgn r => CoreLang.Mode.rgn (TermSound.idxC d r)
  | μ => μ
TermSound.modeIni : CoreLang.Mode → (WeakMem.AllocLabel → Nat) → (WeakMem.AllocLabel → Nat) → WeakMem.AllocLabel → Nat
  := fun x x_1 x_2 =>
  match (motive := CoreLang.Mode → (WeakMem.AllocLabel → Nat) → (WeakMem.AllocLabel → Nat) → WeakMem.AllocLabel → Nat)
    x, x_1, x_2 with
  | CoreLang.Mode.own, ini, rini => fun l => min (ini l) (rini l)
  | x, ini, x_3 => ini
TermSound.ofMs : (Nat → Bool) → Nat → CoreLang.Idx
  eq_def ∀ (m : Nat → Bool) (x : Nat),
  TermSound.ofMs m x =
    match x with
    | 0 => CoreLang.Idx.lit 0
    | n.succ => if m n = true then (CoreLang.Idx.var n).add (TermSound.ofMs m n) else TermSound.ofMs m n
TermSound.ownW : CoreLang.Mode → List TermRun.Word → TermSound.Res
  := fun μ ws =>
  match μ, List.map (fun x => x.val) ws with
  | CoreLang.Mode.own, [WeakMem.Val.loc a tag] => TermSound.Res.one (TermSound.Key.own a.label)
  | x, x_1 => TermSound.Res.zero
TermSound.rf : CoreLang.Ty → CoreLang.Ty
  := fun x =>
  match x with
  | CoreLang.Ty.int => CoreLang.Ty.int
  | t => t.refine CoreLang.Pred.tt
TermSound.rgnFree : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  TermSound.rgnFree x =
    match x with
    | CoreLang.Ty.ptr (CoreLang.Mode.rgn r) τ => false
    | CoreLang.Ty.ptrOpt (CoreLang.Mode.rgn r) τ => false
    | CoreLang.Ty.loc (CoreLang.Mode.rgn r) τ => false
    | a.prod b => TermSound.rgnFree a && TermSound.rgnFree b
    | a.sum b => TermSound.rgnFree a && TermSound.rgnFree b
    | CoreLang.Ty.array i t => TermSound.rgnFree t
    | t.mu => TermSound.rgnFree t
    | CoreLang.Ty.tex κ t => TermSound.rgnFree t
    | t.iall => TermSound.rgnFree t
    | t.iex => TermSound.rgnFree t
    | t.refine ψ => TermSound.rgnFree t
    | x => true
TermSound.shp : CoreLang.Ty → TermSound.Shp
  eq_def ∀ (x : CoreLang.Ty),
  TermSound.shp x =
    match x with
    | a.prod b => (TermSound.shp a).prod (TermSound.shp b)
    | b.iex => TermSound.Shp.iex (TermSound.wc b.iex) (TermSound.shp b)
    | CoreLang.Ty.fix sg w => TermSound.Shp.fix sg w
    | CoreLang.Ty.lanes sg w => TermSound.Shp.lanes sg w
    | t.refine ψ => TermSound.shp t
    | t.mu => TermSound.shp t
    | t.iall => TermSound.shp t
    | a.sum b => TermSound.Shp.sum (TermSound.wc (a.sum b)) (TermSound.lnorm (a.sum b))
    | σ.code τ k => TermSound.Shp.code CoreLang.Ty.unit CoreLang.Ty.unit
    | CoreLang.Ty.tex κ b => TermSound.Shp.tex (TermSound.wc (CoreLang.Ty.tex κ b)) (TermSound.shp b)
    | t => TermSound.Shp.leaf (TermSound.wc t)
TermSound.skel : CoreLang.Ty → CoreLang.Ty
  eq_def ∀ (x : CoreLang.Ty),
  TermSound.skel x =
    match x with
    | CoreLang.Ty.intS i => CoreLang.Ty.int
    | a.prod b => (TermSound.skel a).prod (TermSound.skel b)
    | a.sum b => (TermSound.skel a).sum (TermSound.skel b)
    | a.code b k => (TermSound.skel a).code (TermSound.skel b) k
    | CoreLang.Ty.array i t => CoreLang.Ty.array (CoreLang.Idx.lit 0) (TermSound.skel t)
    | CoreLang.Ty.ptr μ t => CoreLang.Ty.ptr (TermSound.skelM μ) (TermSound.skel t)
    | CoreLang.Ty.ptrOpt μ t => CoreLang.Ty.ptrOpt (TermSound.skelM μ) (TermSound.skel t)
    | CoreLang.Ty.loc μ t => CoreLang.Ty.loc (TermSound.skelM μ) (TermSound.skel t)
    | CoreLang.Ty.word r P => CoreLang.Ty.word (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tok r st => CoreLang.Ty.tok (CoreLang.Idx.lit 0) none
    | CoreLang.Ty.word2 r P => CoreLang.Ty.word2 (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tok2 r P i0 i1 => CoreLang.Ty.tok2 (CoreLang.Idx.lit 0) P (CoreLang.Idx.lit 0) (CoreLang.Idx.lit 0)
    | CoreLang.Ty.epair r P => CoreLang.Ty.epair (CoreLang.Idx.lit 0) P
    | CoreLang.Ty.tokE r P e => CoreLang.Ty.tokE (CoreLang.Idx.lit 0) P (CoreLang.Idx.lit 0)
    | CoreLang.Ty.share r => CoreLang.Ty.share (CoreLang.Idx.lit 0)
    | CoreLang.Ty.cap s r => CoreLang.Ty.cap s (CoreLang.Idx.lit 0)
    | t.mu => (TermSound.skel t).mu
    | CoreLang.Ty.tex κ t => CoreLang.Ty.tex κ (TermSound.skel t)
    | t.iall => (TermSound.skel t).iall
    | t.iex => (TermSound.skel t).iex
    | t.refine ψ => TermSound.rf (TermSound.skel t)
    | t => t
TermSound.skelM : CoreLang.Mode → CoreLang.Mode
  := fun x =>
  match x with
  | CoreLang.Mode.rgn r => CoreLang.Mode.rgn (CoreLang.Idx.lit 0)
  | μ => μ
TermSound.tokWd : Nat → Option (CoreLang.Proto × CoreLang.Idx) → TermSound.Res
  := fun n x =>
  match x with
  | none => TermSound.Res.one (TermSound.Key.wd n)
  | some val => TermSound.Res.zero
TermSound.vcons : Nat → (Nat → Nat) → Nat → Nat
  := fun w ρ => TermSound.vins 0 w ρ
TermSound.vins : Nat → Nat → (Nat → Nat) → Nat → Nat
  := fun c w ρ k => if k < c then ρ k else if k = c then w else ρ (k - 1)
TermSound.vsh : Nat → TermSound.Shp → TermSound.Shp
  eq_def ∀ (s : Nat) (x : TermSound.Shp),
  TermSound.vsh s x =
    match x with
    | TermSound.Shp.sum n R => TermSound.Shp.sum (some (s + 1)) R
    | TermSound.Shp.iex n b => TermSound.Shp.iex n (TermSound.vsh s b)
    | TermSound.Shp.tex n b => TermSound.Shp.tex n (TermSound.vsh s b)
    | r => r
TermSound.wc : CoreLang.Ty → Option Nat
  := fun t => Option.map TermRun.nWords (TermRun.accs t)
TermSound.wdW : CoreLang.Proto → Nat → TermSound.Res
  := fun P n => if P.epoch = true then TermSound.Res.zero else TermSound.Res.one (TermSound.Key.wd n)
TermSpawn.ConstVal : CoreTerm.Prog → CoreLang.Ty → List WeakMem.Val → Prop
  := fun p t vals =>
  TermSound.VRR (fun x => 0) (TermCut.Cut (TermInit.Eg0 p) 0 TermInit.S0) t (TermSound.shp t)
      (List.map TermRun.cst vals) TermSound.Res.zero ∧
    TermSound.Lay [] t t (List.map TermRun.cst vals)

  | TermSpawn.Kid.mk : String → List CoreLang.Ty → CoreLang.Ty → Bool → List WeakMem.Val → TermSpawn.Kid

  := fun self => self.1

  := fun self => self.2

  := fun self => self.3

  := fun self => self.4

TermSpawn.KidOK : CoreTerm.Prog → CoreLang.Ty → TermSpawn.Kid → Prop
  := fun p rt kd =>
  kd.ty.affine = false ∧
    TermClose.tb 0 kd.ty = true ∧
      TermScope.drf kd.ty = true ∧
        kd.ts.all TermScope.drf = true ∧
          kd.ts.all TermScope.argOK = true ∧
            TermScope.wrapOK kd.ty = true ∧
              (∃ r,
                  CoreTerm.check p { vars := [("a", kd.ty, true)], ret := r }
                        (CoreTerm.Expr.tail kd.f [] (CoreTerm.At.var "a") kd.ts) =
                      Except.ok () ∧
                    TermTv.TvOK { vars := [("a", kd.ty, true)], ret := r }) ∧
                (kd.useRes = true ∧
                    (kd.vals = [] ∧ kd.ty = rt ∨ ∃ τ, kd.ty = rt.prod τ ∧ TermSpawn.ConstVal p τ kd.vals) ∨
                  kd.useRes = false ∧ TermSpawn.ConstVal p kd.ty kd.vals)
TermTv.TvOK : CoreTerm.Cx → Prop
  := fun cx =>
  (∀ (e : String × CoreLang.Ty × Bool), e ∈ cx.vars → TermClose.tb (List.length cx.ke) e.snd.fst = true) ∧
    (∀ (tn : CoreTerm.Torn), tn ∈ cx.torn → TermClose.tb (List.length cx.ke) tn.ty = true) ∧
      TermClose.tb (List.length cx.ke) cx.ret = true ∧
        (∀ (e : String × CoreLang.Ty × Bool), e ∈ cx.vars → cx.opaqueOK e.snd.fst = true) ∧
          (∀ (e : String × CoreLang.Ty × Bool), e ∈ cx.vars → TermTv.mg e.snd.fst = true) ∧
            (∀ (tn : CoreTerm.Torn), tn ∈ cx.torn → TermTv.mg tn.ty = true) ∧
              TermTv.mg cx.ret = true ∧ TermTv.noIallV cx.ret = true
TermTv.mg : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  TermTv.mg x =
    match x with
    | t.mu => !CoreLang.Ty.unguarded 0 t && TermTv.mg t
    | a.prod b => TermTv.mg a && TermTv.mg b
    | a.sum b => TermTv.mg a && TermTv.mg b
    | a.code b k => TermTv.mg a && TermTv.mg b
    | CoreLang.Ty.array i t => TermTv.mg t
    | CoreLang.Ty.ptr μ t => TermTv.mg t
    | CoreLang.Ty.ptrOpt μ t => TermTv.mg t
    | CoreLang.Ty.loc μ t => TermTv.mg t
    | CoreLang.Ty.tex κ t => TermTv.mg t
    | t.iall => TermTv.mg t
    | t.iex => TermTv.mg t
    | t.refine ψ => TermTv.mg t
    | x => true
TermTv.noIallV : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  TermTv.noIallV x =
    match x with
    | τ.iall => false
    | a.prod b => TermTv.noIallV a && TermTv.noIallV b
    | a.sum b => TermTv.noIallV a && TermTv.noIallV b
    | CoreLang.Ty.array i t => TermTv.noIallV t
    | t.mu => TermTv.noIallV t
    | CoreLang.Ty.tex κ t => TermTv.noIallV t
    | t.iex => TermTv.noIallV t
    | t.refine ψ => TermTv.noIallV t
    | x => true
