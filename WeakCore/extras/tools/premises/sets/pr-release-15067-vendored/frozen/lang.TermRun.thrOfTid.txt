TermRun.thrOfTid : TermRun.Cfg → TermRun.Word → Option Nat
  := fun c w =>
  match w.val with
  | WeakMem.Val.loc { label := l, offset := 0 } 0 =>
    List.find? (fun j => (c.ths[j]?.bind fun x => x.tidLbl) == some l) (List.range c.ths.size)
  | x => none
