CoreTerm.restrictedDeclOK : CoreTerm.Prog → CoreTerm.Decl → Bool
  := fun p d =>
  match d.kind with
  | CoreTerm.DeclKind.restricted fuel stack ro loop =>
    (CoreTerm.noShadow d &&
          match loop, CoreTerm.selfBound d d.body with
          | none, CoreTerm.Tail.noTail => true
          | some val, CoreTerm.Tail.bounded q j => true
          | x, x_1 => false) &&
        CoreTerm.restrictedOK ro d.name (fun f => Option.map (fun x => x.kind) (List.find? (fun x => x.name == f) p))
          d.arg
          (match d.argTy with
          | σ.code τ k => some k
          | x => none)
          d.body &&
      decide (CoreTerm.fuelDecl p d ≤ fuel)
  | x => true
