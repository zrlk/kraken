CoreLang.zeroArmP : (CoreLang.Ty → Bool) → CoreLang.IEnv → CoreLang.KEnv → CoreLang.Ty → Option Bool
  eq_def ∀ (sel : CoreLang.Ty → Bool) (env : CoreLang.IEnv) (ke : CoreLang.KEnv) (x : CoreLang.Ty),
  CoreLang.zeroArmP sel env ke x =
    match x with
    | a.sum b => CoreLang.zeroArmP sel env ke a <|> CoreLang.zeroArmP sel env ke b
    | CoreLang.Ty.void => none
    | t.refine ψ => Option.map (fun x => x && CoreLang.Pred.zeroHolds env ψ) (CoreLang.zeroArmP sel env ke t)
    | t.iall => CoreLang.zeroArmP sel (none :: env) ke t
    | t.iex => CoreLang.zeroArmP sel (some 0 :: env) ke t
    | t.mu => CoreLang.zeroArmP sel env (none :: ke) t
    | CoreLang.Ty.tex κ t => Option.map (fun x => false) (CoreLang.zeroArmP sel env (none :: ke) t)
    | t => if sel t = true then some (CoreLang.zeroableIn env ke t) else none
CoreLang.zeroableIn : CoreLang.IEnv → CoreLang.KEnv → CoreLang.Ty → Bool
  eq_def ∀ (env : CoreLang.IEnv) (ke : CoreLang.KEnv) (x : CoreLang.Ty),
  CoreLang.zeroableIn env ke x =
    match x with
    | CoreLang.Ty.unit => true
    | CoreLang.Ty.void => false
    | CoreLang.Ty.int => true
    | CoreLang.Ty.intS i => CoreLang.Idx.evalIn env i == some 0
    | CoreLang.Ty.fix sg w => true
    | CoreLang.Ty.lanes sg w => true
    | a.prod b => CoreLang.zeroableIn env ke a && CoreLang.zeroableIn env ke b
    | a.sum b =>
      match (a.sum b).tier with
      | CoreLang.Tier.niche =>
        (CoreLang.zeroArmP CoreLang.Ty.payloadFree env ke a <|> CoreLang.zeroArmP CoreLang.Ty.payloadFree env ke b) ==
          some true
      | CoreLang.Tier.wide =>
        (CoreLang.zeroArmP (fun x => true) env ke a <|> CoreLang.zeroArmP (fun x => true) env ke b) == some true
      | CoreLang.Tier.boxed => false
    | σ.code τ k => false
    | CoreLang.Ty.tid => false
    | CoreLang.Ty.opq o => false
    | CoreLang.Ty.array i t => CoreLang.zeroableIn env ke t
    | CoreLang.Ty.ptr μ τ => false
    | CoreLang.Ty.ptrOpt μ τ => true
    | CoreLang.Ty.loc μ (CoreLang.Ty.array i τ) => CoreLang.Idx.evalIn env i == some 0
    | CoreLang.Ty.loc μ τ => false
    | CoreLang.Ty.word r P => false
    | CoreLang.Ty.tok r st => false
    | CoreLang.Ty.word2 r P => false
    | CoreLang.Ty.tok2 r P i0 i1 => false
    | CoreLang.Ty.epair r P => false
    | CoreLang.Ty.tokE r P e => false
    | CoreLang.Ty.share r => false
    | CoreLang.Ty.cap s r => false
    | CoreLang.Ty.tvar k => ke.ask k false fun x => x.zero
    | t.mu => CoreLang.zeroableIn env (none :: ke) t
    | CoreLang.Ty.tex κ τ => false
    | t.iall => CoreLang.zeroableIn (none :: env) ke t
    | t.iex => CoreLang.zeroableIn (some 0 :: env) ke t
    | t.refine ψ => CoreLang.zeroableIn env ke t && CoreLang.Pred.zeroHolds env ψ

CoreTerm.checkOp : CoreTerm.Prog → CoreTerm.Cx → String → CoreTerm.Op → CoreTerm.M CoreTerm.Cx
  := fun p cx x x_1 =>
  match x_1 with
  | CoreTerm.Op.atom a => do
    let __x ← cx.use a
    match __x with
      | (t, cx) => cx.bind x t
  | CoreTerm.Op.pair a b => do
    let __x ← cx.use a
    match __x with
      | (t, cx) => do
        let __x ← cx.use b
        match __x with
          | (u, cx) => cx.bind x (t.prod u)
  | CoreTerm.Op.inl τ a =>
    have __do_jp := fun __r => do
      let __x ← cx.use a
      match __x with
        | (t, cx) => cx.bind x (t.sum τ);
    if (!CoreLang.Ty.wf (List.length cx.ke) 0 τ) = true then do
      let __r ← throw "inl: the other arm must be closed"
      __do_jp __r
    else __do_jp ()
  | CoreTerm.Op.inr τ a =>
    have __do_jp := fun __r => do
      let __x ← cx.use a
      match __x with
        | (t, cx) => cx.bind x (τ.sum t);
    if (!CoreLang.Ty.wf (List.length cx.ke) 0 τ) = true then do
      let __r ← throw "inr: the other arm must be closed"
      __do_jp __r
    else __do_jp ()
  | CoreTerm.Op.add a b => do
    let t ← cx.peek a
    let u ← cx.peek b
    match CoreTerm.intOf t, CoreTerm.intOf u with
      | some (some i), some (some j) => cx.bind x (CoreLang.Ty.intS (i.add j))
      | some val, some val_1 => cx.bind x CoreLang.Ty.int
      | x, x_2 => throw "add needs integers"
  | CoreTerm.Op.sub a b => do
    let t ← cx.peek a
    let u ← cx.peek b
    match CoreTerm.intOf t, CoreTerm.intOf u with
      | some (some i), some (some j) =>
        match cx.norm i, cx.norm j with
        | some ni, some nj =>
          if nj.leB ni = true then cx.bind x (CoreLang.Ty.intS (ni.diff nj).toIdx)
          else throw "sub: the difference is not provably nonnegative"
        | x, x_2 => throw "sub: index outside the decidable fragment"
      | x, x_2 => throw "sub needs integers of known value"
  | CoreTerm.Op.new a => do
    let __x ← cx.use a
    match __x with
      | (t, cx) =>
        have __do_jp := fun __r => cx.bind x (CoreLang.Ty.ptr CoreLang.Mode.own t);
        if t.secondClass = true then do
          let __r ← throw "new: a second-class capability cannot be stored"
          __do_jp __r
        else __do_jp ()
  | CoreTerm.Op.publish a => do
    let __x ← cx.use a
    match __x with
      | (t, cx) =>
        match t with
        | CoreLang.Ty.ptr CoreLang.Mode.own u =>
          if CoreLang.Ty.sharedOKIn cx.ke (CoreLang.Ty.ptr CoreLang.Mode.pub u) = true then
            cx.bind x (CoreLang.Ty.ptr CoreLang.Mode.pub u)
          else throw "publish: the target is not shareable (rows 16, 17)"
        | (CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini k) =>
          if CoreLang.Ty.sharedOKIn cx.ke (CoreLang.Ty.loc CoreLang.Mode.pub (CoreLang.Ty.array n u)) = true then
            cx.bind x ((CoreLang.Ty.loc CoreLang.Mode.pub (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini k))
          else throw "publish: the target is not shareable (rows 16, 17)"
        | x => throw "publish needs an own reference"
  | CoreTerm.Op.intoRgn rn a => do
    let r ← cx.idx rn
    let __x ← cx.use a
    match __x with
      | (t, cx) =>
        match t with
        | CoreLang.Ty.ptr CoreLang.Mode.own u =>
          if CoreTerm.rgnCellOK cx.ke u = true then cx.bind x (CoreLang.Ty.ptr (CoreLang.Mode.rgn r) u)
          else throw "intoRgn: the target is not shareable (rows 16, 17): build it inside the region"
        | (CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini k) =>
          if CoreTerm.rgnCellOK cx.ke (CoreLang.Ty.array n u) = true then
            cx.bind x ((CoreLang.Ty.loc (CoreLang.Mode.rgn r) (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini k))
          else throw "intoRgn: the target is not shareable (rows 16, 17): build it inside the region"
        | x => throw "intoRgn needs an own reference or an own buffer"
  | CoreTerm.Op.newBuf n τ =>
    have __do_jp := fun __r =>
      have __do_jp := fun __r =>
        have __do_jp := fun __r =>
          have __do_jp := fun __r => do
            let k ← CoreTerm.intIdx cx n
            have ini : CoreLang.Idx := if CoreLang.zeroableIn [] cx.ke τ = true then k else CoreLang.Idx.lit 0
            cx.bind x ((CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array k τ)).refine (CoreLang.Pred.ini ini));
          if (!τ.arrFree) = true then do
            let __r ← throw "newBuf: a slot cannot hold an array by value (an array lives behind a Loc)"
            __do_jp __r
          else __do_jp ();
        if (!cx.opaqueOK τ) = true then do
          let __r ← throw "newBuf: the slots' layout depends on an unpacked type"
          __do_jp __r
        else __do_jp ();
      if τ.secondClass = true then do
        let __r ← throw "newBuf: a second-class capability cannot be stored"
        __do_jp __r
      else __do_jp ();
    if (!CoreLang.Ty.wf (List.length cx.ke) cx.inames.length τ) = true then do
      let __r ← throw "newBuf: the element type is not well formed"
      __do_jp __r
    else __do_jp ()
  | CoreTerm.Op.idx b i => do
    let t ← cx.peek b
    have __do_jp : CoreLang.Mode × CoreLang.Idx × CoreLang.Ty × CoreLang.Idx → CoreTerm.M CoreTerm.Cx := fun __x =>
      match __x with
      | (μ, n, u, k) => do
        let j ← CoreTerm.intIdx cx i
        have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
          have __do_jp := fun __r =>
            have __do_jp := fun __r =>
              have __do_jp := fun cx => cx.bind x (CoreLang.Ty.loc μ u);
              if (μ == CoreLang.Mode.own) = true then do
                let cx ← (fun x => x.snd) <$> cx.use b
                __do_jp cx
              else do
                let cx ← pure cx
                __do_jp cx;
            if (!cx.opaqueOK u) = true then do
              let __r ← throw "idx: the slots' layout depends on an unpacked type"
              __do_jp __r
            else __do_jp ();
          if (!cx.ltI j k) = true then do
            let __r ← throw "idx: the slot is not provably initialised"
            __do_jp __r
          else __do_jp ()
        if (!cx.ltI j n) = true then do
            let __r ← throw "idx: the index is not provably below the capacity"
            __do_jp __r
          else __do_jp ()
    match CoreTerm.bufOf t with
      | some c => do
        let __x ← pure c
        __do_jp __x
      | none => do
        let __x ← throw "idx needs a buffer"
        __do_jp __x
  | CoreTerm.Op.bufInit b i v => do
    let __x ← cx.use b
    match __x with
      | (t, cx) =>
        have __do_jp := fun __x =>
          match __x with
          | (n, u, k) => do
            let j ← CoreTerm.intIdx cx i
            have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
              have __do_jp := fun __r => do
                let __x ← cx.use v
                match __x with
                  | (tv, cx) =>
                    have __do_jp := fun __r =>
                      cx.bind x
                        ((CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array n u)).refine
                          (CoreLang.Pred.ini (k.add (CoreLang.Idx.lit 1))));
                    if (!CoreTerm.sub cx tv u) = true then do
                      let __r ← throw "bufInit: the value does not have the element type"
                      __do_jp __r
                    else __do_jp ();
              if (!cx.ltI k n) = true then do
                let __r ← throw "bufInit: the buffer is full"
                __do_jp __r
              else __do_jp ()
            if (!cx.eqI j k) = true then do
                let __r ← throw "bufInit: the slot is not the next uninitialised one"
                __do_jp __r
              else __do_jp ();
        match CoreTerm.bufOf t with
        | some (CoreLang.Mode.own, n, u, k) => do
          let __x ← pure (n, u, k)
          __do_jp __x
        | some val => do
          let __x ← throw "bufInit needs an own buffer"
          __do_jp __x
        | none => do
          let __x ← throw "bufInit needs a buffer"
          __do_jp __x
  | CoreTerm.Op.null μ τ =>
    have t := CoreLang.Ty.ptrOpt μ τ;
    have __do_jp := fun __r => cx.bind x t;
    if (!CoreLang.Ty.wf (List.length cx.ke) cx.inames.length t) = true then do
      let __r ← throw "null: the target is not well formed"
      __do_jp __r
    else __do_jp ()
  | CoreTerm.Op.fold a τ =>
    have __do_jp := fun __r => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          if CoreTerm.foldOk cx t τ = true then cx.bind x τ
          else throw "fold: the reference's target is not one unrolling of the μ-type";
    if (!CoreLang.Ty.wf (List.length cx.ke) cx.inames.length τ) = true then do
      let __r ← throw "fold: the target is not well formed"
      __do_jp __r
    else __do_jp ()
  | CoreTerm.Op.fld i a => do
    let t ← cx.peek a
    have __do_jp : CoreLang.Mode × CoreLang.Ty → CoreTerm.M CoreTerm.Cx := fun __x =>
      match __x with
      | (μ, u) =>
        have __do_jp := fun __x =>
          match __x with
          | (c1, c2) =>
            have __do_jp := fun __r =>
              have __do_jp := fun cx => cx.bind x (CoreLang.Ty.loc μ (if (i == 0) = true then c1 else c2));
              if (μ == CoreLang.Mode.own) = true then do
                let cx ← (fun x => x.snd) <$> cx.use a
                __do_jp cx
              else do
                let cx ← pure cx
                __do_jp cx;
            if (!(i == 0 || cx.opaqueOK c1)) = true then do
              let __r ← throw "fld: the field's offset depends on an unpacked type"
              __do_jp __r
            else __do_jp ();
        match CoreTerm.unfold1 u with
        | c1.prod c2 => do
          let __x ← pure (c1, c2)
          __do_jp __x
        | x => do
          let __x ← throw "fld needs a reference to a pair"
          __do_jp __x
    match CoreTerm.cellOf t with
      | some c => do
        let __x ← pure c
        __do_jp __x
      | none => do
        let __x ← throw "fld needs a reference"
        __do_jp __x
  | CoreTerm.Op.load acq a => do
    let t ← cx.peek a
    match CoreTerm.wordOf t, a with
      | some (fst, P), CoreTerm.At.var w =>
        have __do_jp := fun __r =>
          have cx := cx.push x;
          have n := CoreLang.Idx.var 0;
          have cx :=
            { inames := cx.inames, facts := cx.facts, vars := cx.vars, saws := (w, n, acq) :: cx.saws,
              torn :=
                List.map
                  (fun tn =>
                    if (tn.w == w && tn.φ == CoreTerm.Phase.fenced) = true then
                      { x := tn.x, ty := tn.ty, w := tn.w, s := tn.s, φ := CoreTerm.Phase.rechecked n }
                    else tn)
                  cx.torn,
              unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq, lent := cx.lent, ret := cx.ret };
          cx.bind x (CoreLang.Ty.intS n);
        if P.ghost = true then do
          let __r ← throw "a ghost word has no runtime value to load"
          __do_jp __r
        else __do_jp ()
      | some val, x => throw "a protocol word must be named by a variable"
      | none, x_2 =>
        match CoreTerm.cellOf t with
        | none => throw "load needs a reference"
        | some (CoreLang.Mode.own, u) =>
          if CoreTerm.affine cx.ke u = true then throw "load would duplicate an affine value" else cx.bind x u
        | some (CoreLang.Mode.pub, u) =>
          if CoreTerm.affine cx.ke u = true then throw "load would duplicate an affine value"
          else
            if (CoreLang.Ty.sensitiveIn cx.ke u && !acq) = true then
              throw "load: a view-sensitive type needs acq (§3.8)"
            else cx.bind x u
        | some (CoreLang.Mode.rgn r, u) =>
          have u' := CoreTerm.adapt r u;
          if (cx.hasTok r || cx.hasShare r) = true then
            if CoreTerm.affine cx.ke u' = true then throw "load would duplicate an affine value" else cx.bind x u'
          else
            if cx.hasCap r false = true then
              if (CoreLang.Ty.sensitiveIn cx.ke u' && !acq) = true then
                throw "load under a capability: a view-sensitive type needs acq (§3.8)"
              else
                if CoreTerm.affine cx.ke u' = true then throw "load would duplicate an affine value" else cx.bind x u'
            else do
              let __x ← CoreTerm.tornLicence cx r
              match __x with
                | (w, s) =>
                  have __do_jp := fun __r =>
                    pure
                      { inames := cx.inames, facts := cx.facts, vars := cx.vars, saws := cx.saws,
                        torn := { x := x, ty := u', w := w, s := s, φ := CoreTerm.Phase.opn } :: cx.torn,
                        unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq, lent := cx.lent, ret := cx.ret };
                  if ((cx.vars.any fun x_3 => x_3.fst == x) || cx.torn.any fun x_3 => x_3.x == x) = true then do
                    let __r ← throw (toString x ++ toString " is already bound (no shadowing)")
                    __do_jp __r
                  else __do_jp ()
  | CoreTerm.Op.store rel a v tok => do
    let t ← cx.peek a
    match CoreTerm.wordOf t with
      | some (r, P) =>
        have __do_jp := fun tk => do
          let __x ← cx.use tk
          match __x with
            | (tt, cx) =>
              have __do_jp := fun __x =>
                match __x with
                | (r', i) => do
                  let j ← CoreTerm.intIdx cx v
                  let __x ← CoreTerm.classifyM cx P i
                  match __x with
                    | (c, k) => do
                      let __x ← CoreTerm.classifyM cx P j
                      match __x with
                        | (c', k') => do
                          let cl ← CoreTerm.clauseM P c
                          let cl' ← CoreTerm.clauseM P c'
                          have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
                            have __do_jp := fun __r =>
                              if cl'.holds = true then
                                have __do_jp := fun __r => cx.bind x CoreLang.Ty.unit;
                                if (!rel && !P.ghost) = true then do
                                  let __r ← throw "giving the region back needs rel"
                                  __do_jp __r
                                else __do_jp ()
                              else
                                if P.ticket = true then
                                  have __do_jp := fun __r => cx.bind x CoreLang.Ty.unit;
                                  if (!rel) = true then do
                                    let __r ← throw "handing the lock on needs rel"
                                    __do_jp __r
                                  else __do_jp ()
                                else cx.bind x (CoreLang.Ty.tok r' (some (P, j)));
                            if (!CoreTerm.pstep P c k c' k') = true then do
                              let __r ← throw "not a transition of the protocol"
                              __do_jp __r
                            else __do_jp ()
                          if cl.holds = true then do
                              let __r ← throw "store from a shared state: use cas"
                              __do_jp __r
                            else __do_jp ();
              match tt with
              | CoreLang.Ty.tok r' (some (Q, i)) =>
                if (!cx.eqI r r') = true then do
                  let __x ← throw "the token is for another region"
                  __do_jp __x
                else
                  if (Q != P) = true then do
                    let __x ← throw "the token is for another protocol"
                    __do_jp __x
                  else do
                    let __x ← pure (r', i)
                    __do_jp __x
              | x => do
                let __x ← throw "a store to a protocol word needs a taken token"
                __do_jp __x;
        match tok with
        | some tk => do
          let tk ← pure tk
          __do_jp tk
        | none => do
          let tk ← throw "a store to a protocol word needs the region's token"
          __do_jp tk
      | none =>
        match CoreTerm.cellOf t with
        | none => throw "store needs a reference"
        | some (μ, u) =>
          have __do_jp := fun __r => do
            let __x ← cx.use v
            match __x with
              | (tv, cx) =>
                match μ with
                | CoreLang.Mode.own =>
                  if CoreTerm.sub cx tv u = true then cx.bind x CoreLang.Ty.unit
                  else throw "store: the value does not have the cell's type"
                | CoreLang.Mode.pub =>
                  if (!CoreTerm.sub cx tv u) = true then throw "store: the value does not have the cell's type"
                  else
                    if (CoreLang.Ty.sensitiveIn cx.ke u && !rel) = true then
                      throw "store: a view-sensitive type needs rel (§3.8)"
                    else cx.bind x CoreLang.Ty.unit
                | CoreLang.Mode.rgn r =>
                  if cx.hasTok r = true then
                    if (!cx.sealedR r) = true then
                      throw "store into a region taken from an observable state before fence st (§4.4 rule 7)"
                    else
                      if (CoreTerm.sub cx tv u || CoreTerm.sub cx tv (CoreTerm.adapt r u)) = true then
                        cx.bind x CoreLang.Ty.unit
                      else throw "store: the value does not have the cell's type"
                  else
                    if cx.hasCap r true = true then
                      if (CoreLang.Ty.sensitiveIn cx.ke u && !rel) = true then
                        throw "store under a capability: a view-sensitive type needs rel (§3.8)"
                      else
                        if (CoreTerm.sub cx tv u || CoreTerm.sub cx tv (CoreTerm.adapt r u)) = true then
                          cx.bind x CoreLang.Ty.unit
                        else throw "store: the value does not have the cell's type"
                    else throw "store into a region without its token";
          if (!CoreLang.Ty.wordFreeIn cx.ke u) = true then do
            let __r ← throw CoreTerm.wordCellMsg
            __do_jp __r
          else __do_jp ()
  | CoreTerm.Op.fence f =>
    have cx :=
      if (f != CoreTerm.Fence.st) = true then
        { inames := cx.inames, facts := cx.facts, vars := cx.vars,
          saws :=
            List.map
              (fun x =>
                match x with
                | (w, s, snd) => (w, s, true))
              cx.saws,
          torn :=
            List.map (fun tn => { x := tn.x, ty := tn.ty, w := tn.w, s := tn.s, φ := CoreTerm.phaseFence tn.φ })
              cx.torn,
          unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq, lent := cx.lent, ret := cx.ret }
      else cx;
    have cx :=
      if (f != CoreTerm.Fence.ld) = true then
        { inames := cx.inames, facts := cx.facts, vars := cx.vars, saws := cx.saws, torn := cx.torn, ke := cx.ke,
          opq := cx.opq, lent := cx.lent, ret := cx.ret }
      else cx;
    cx.bind x CoreLang.Ty.unit
  | CoreTerm.Op.mkWord rn P v tk => do
    let r ← cx.idx rn
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => do
      let __x ← cx.use tk
      match __x with
        | (tt, cx) =>
          have __do_jp := fun __r => do
            let j ← CoreTerm.intIdx cx v
            let __x ← CoreTerm.classifyM cx P j
            match __x with
              | (c, snd) => do
                let cl ← CoreTerm.clauseM P c
                if P.ghost = true then
                    have __do_jp := fun __r =>
                      cx.bind x ((CoreLang.Ty.word r P).prod (CoreLang.Ty.tok r (some (P, j))));
                    if cl.holds = true then do
                      let __r ← throw "a ghost word is born in a state that does not hold its region"
                      __do_jp __r
                    else __do_jp ()
                  else
                    have __do_jp := fun __r => cx.bind x (CoreLang.Ty.word r P);
                    if (!cl.holds) = true then do
                      let __r ← throw "a word is born in a state that holds its region"
                      __do_jp __r
                    else __do_jp ();
          match tt with
          | CoreLang.Ty.tok r' none =>
            if (!cx.eqI r r') = true then do
              let __r ← throw "the token is for another region"
              __do_jp __r
            else __do_jp ()
          | CoreLang.Ty.tok r (some val) => do
            let __r ← throw "a word is born from a fresh region token, not a taken one"
            __do_jp __r
          | x => do
            let __r ← throw "mkWord needs a region token"
            __do_jp __r
    if (!CoreTerm.pwf P) = true then do
        let __r ←
          throw
              "the protocol is not well formed (states overlap, a move names no state, a lending state does not hold, a robust/norepeat/epoch claim fails, or a recovery target is not held -> shared)"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.mkWord2 rn P v0 v1 tk => do
    let r ← cx.idx rn
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => do
      let __x ← cx.use tk
      match __x with
        | (tt, cx) =>
          have __do_jp := fun __r => do
            let j0 ← CoreTerm.intIdx cx v0
            let j1 ← CoreTerm.intIdx cx v1
            let __x ← CoreTerm.classify2M cx P j0 j1
            match __x with
              | (c, fst, snd) => do
                let cl ← CoreTerm.clause2M P c
                have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x (CoreLang.Ty.word2 r P)
                if (!cl.holds) = true then do
                    let __r ← throw "a pair is born in a state that holds its region"
                    __do_jp __r
                  else __do_jp ();
          match tt with
          | CoreLang.Ty.tok r' none =>
            if (!cx.eqI r r') = true then do
              let __r ← throw "the token is for another region"
              __do_jp __r
            else __do_jp ()
          | CoreLang.Ty.tok r (some val) => do
            let __r ← throw "a pair is born from a fresh region token, not a taken one"
            __do_jp __r
          | x => do
            let __r ← throw "mkWord2 needs a region token"
            __do_jp __r
    if (!CoreTerm.pwf2 P) = true then do
        let __r ←
          throw
              "the pair protocol is not well formed (states overlap, a move names no state, or a robust/epoch claim fails)"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.ldp acq a n0 n1 => do
    let t ← cx.peek a
    if CoreTerm.pairRef t = true then
        match a with
        | CoreTerm.At.var x_2 =>
          have cx := (cx.push n0).push n1;
          cx.bind x ((CoreLang.Ty.intS (CoreLang.Idx.var 1)).prod (CoreLang.Ty.intS (CoreLang.Idx.var 0)))
        | x => throw "a protocol pair must be named by a variable"
      else throw "ldp needs a reference to a protocol pair"
  | CoreTerm.Op.stp rel a v0 v1 tok => do
    let t ← cx.peek a
    match CoreTerm.word2Of t with
      | none => throw "stp needs a reference to a protocol pair"
      | some (r, P) => do
        let __x ← cx.use tok
        match __x with
          | (tt, cx) =>
            have __do_jp := fun __x =>
              match __x with
              | (r', i0, i1) => do
                let j0 ← CoreTerm.intIdx cx v0
                let j1 ← CoreTerm.intIdx cx v1
                let __x ← CoreTerm.classify2M cx P i0 i1
                match __x with
                  | (c, k0, k1) => do
                    let __x ← CoreTerm.classify2M cx P j0 j1
                    match __x with
                      | (c', k0', k1') => do
                        let cl ← CoreTerm.clause2M P c
                        let cl' ← CoreTerm.clause2M P c'
                        have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
                          have __do_jp := fun __r =>
                            if cl'.holds = true then
                              have __do_jp := fun __r => cx.bind x CoreLang.Ty.unit;
                              if (!rel) = true then do
                                let __r ← throw "giving the region back needs rel"
                                __do_jp __r
                              else __do_jp ()

                          if (!CoreTerm.pstep2 P c k0 k1 c' k0' k1') = true then do
                            let __r ← throw "not a transition of the pair protocol"
                            __do_jp __r
                          else __do_jp ()
                        if cl.holds = true then do
                            let __r ← throw "stp from a shared state: use casp"
                            __do_jp __r
                          else __do_jp ();
            match tt with
            | CoreLang.Ty.tok2 r' Q i0 i1 =>
              if (!cx.eqI r r') = true then do
                let __x ← throw "the token is for another region"
                __do_jp __x
              else
                if (Q != P) = true then do
                  let __x ← throw "the token is for another protocol"
                  __do_jp __x
                else do
                  let __x ← pure (r', i0, i1)
                  __do_jp __x
            | x => do
              let __x ← throw "a store to a protocol pair needs its taken token"
              __do_jp __x
  | CoreTerm.Op.mkEPair rn P v0 v1 => do
    let r ← cx.idx rn
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => do
      let _ ← CoreTerm.intIdx cx v0
      let _ ← CoreTerm.intIdx cx v1
      cx.bind x (CoreLang.Ty.epair r P)
    if (!CoreTerm.pwfE P) = true then do
        let __r ← throw "mkEPair needs a well-formed epoch protocol"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.mkTicket rn N P v tk => do
    let r ← cx.idx rn
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
      have __do_jp := fun __r => do
        let __x ← cx.use tk
        match __x with
          | (tt, cx) =>
            have __do_jp := fun __r => do
              let _ ← CoreTerm.intIdx cx v
              cx.bind x ((CoreLang.Ty.word r N).prod (CoreLang.Ty.word r P));
            match tt with
            | CoreLang.Ty.tok r' none =>
              if (!cx.eqI r r') = true then do
                let __r ← throw "the token is for another region"
                __do_jp __r
              else __do_jp ()
            | CoreLang.Ty.tok r (some val) => do
              let __r ← throw "a ticket lock is born from a fresh region token, not a taken one"
              __do_jp __r
            | x => do
              let __r ← throw "mkTicket needs a region token"
              __do_jp __r;
      if (!(CoreTerm.pwf P && P.ticket)) = true then do
        let __r ← throw "mkTicket needs a ticket protocol for its owner"
        __do_jp __r
      else __do_jp ()
    if (!(CoreTerm.pwf N && N.epoch)) = true then do
        let __r ← throw "mkTicket needs an epoch counter for its dispenser"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.claim a tk => do
    let t ← cx.peek a
    match CoreTerm.wordOf t, a with
      | some (r, P), CoreTerm.At.var w =>
        have __do_jp := fun __r => do
          let __x ← cx.use tk
          match __x with
            | (tt, cx) =>

              | CoreLang.Ty.tokE r' P j =>

                  let j ← throw "the ticket is for another region"
                  __do_jp j
                else do
                  let j ← pure j
                  __do_jp j
              | x => do
                let j ← throw "claim needs a ticket (the dispenser's epoch token)"
                __do_jp j;
        if (!P.ticket) = true then do
          let __r ← throw "claim needs the owner word of a ticket lock"
          __do_jp __r
        else __do_jp ()
      | some val, x => throw "a protocol word must be named by a variable"
      | none, x => throw "claim needs a reference to a protocol word"
  | CoreTerm.Op.fetch o acq rel a v => do
    let tw ← cx.peek a
    match CoreTerm.wordOf tw with
      | none => do
        let u ← CoreTerm.casCellTy cx tw
        have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => do
          let __x ← cx.use v
          match __x with
            | (tv, cx) =>
              have __do_jp := fun __r => cx.bind x CoreLang.Ty.int;
              if (!CoreTerm.sub cx tv CoreLang.Ty.int) = true then do
                let __r ← throw "fetch: the operand is not an Int"
                __do_jp __r
              else __do_jp ()
        if (u != CoreLang.Ty.int) = true then do
            let __r ← throw "fetch needs an Int cell or a protocol word"
            __do_jp __r
          else __do_jp ()
      | some (r, P) =>
        have __do_jp := fun __r => do
          let j ← CoreTerm.intIdx cx v
          if P.epoch = true then
              have __do_jp := fun __r =>
                have __do_jp := fun __r => cx.bind x (CoreTerm.fetchEpochTy r P);
                if (!cx.eqI j (CoreLang.Idx.lit 1)) = true then do
                  let __r ← throw "an epoch counter advances by 1"
                  __do_jp __r
                else __do_jp ();
              if (!acq) = true then do
                let __r ← throw "entering an epoch needs acq"
                __do_jp __r
              else __do_jp ()
            else do
              let d ← cx.constIdx j
              have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r =>
                have __do_jp := fun __r => cx.bind x CoreLang.Ty.int;
                if (!CoreTerm.fetchOK P d) = true then do
                  let __r ←
                    throw "fetch: not a +d move between shared states from every family of the protocol (or it lends)"
                  __do_jp __r
                else __do_jp ()
              if (!cx.eqI j (CoreLang.Idx.lit d)) = true then do
                  let __r ← throw "fetch on a protocol word: the operand must be a known constant"
                  __do_jp __r
                else __do_jp ();
        if (o != CoreTerm.FetchOp.add) = true then do
          let __r ← throw "a fetch on a protocol word must be add: a protocol's moves are increments"
          __do_jp __r
        else __do_jp ()
  | CoreTerm.Op.pack a wn τ => do
    let w ← cx.idx wn
    have __do_jp : CoreLang.Ty → CoreTerm.M CoreTerm.Cx := fun body =>
      have __do_jp := fun __r => do
        let __x ← cx.use a
        match __x with
          | (t, cx) => do
            CoreTerm.noLeak cx t
            if CoreTerm.sub cx t (CoreTerm.instT [w] 0 body) = true then cx.bind x τ
              else throw "pack: the value does not match the target at the witness";
      if (!CoreLang.Ty.wf (List.length cx.ke) cx.inames.length τ) = true then do
        let __r ← throw "pack: the target must be well formed in the index scope"
        __do_jp __r
      else __do_jp ()

      | b.iex => do
        let body ← pure b
        __do_jp body
      | x => do
        let body ← throw "pack targets an index existential"
        __do_jp body
  | CoreTerm.Op.packT a σ τ =>
    match τ with
    | CoreLang.Ty.tex κ body =>
      have __do_jp := fun __r =>
        have __do_jp := fun __r =>
          have __do_jp := fun __r => do
            let __x ← cx.use a
            match __x with
              | (t, cx) => do
                CoreTerm.noLeak cx t
                if CoreTerm.sub cx t (CoreLang.Ty.substT 0 σ body) = true then cx.bind x τ

            let __r ← throw "pack: the witness does not have the variable's kind"
            __do_jp __r
          else __do_jp ();

          let __r ← throw "pack: the target must be closed"
          __do_jp __r
        else __do_jp ();

        let __r ← throw "pack: the package's variable must occur only under a reference"
        __do_jp __r
      else __do_jp ()

  | CoreTerm.Op.validate y =>
    match List.find? (fun x => x.x == y) cx.torn with
    | none => throw (toString y ++ toString " is not a torn read")
    | some tn =>
      match tn.φ with
      | CoreTerm.Phase.rechecked v =>

      | CoreTerm.Phase.fenced => throw "validate: the word was not re-read after the fence"
      | CoreTerm.Phase.opn => throw "validate: no fence ld between the torn read and the re-check"

    let d ← p.find f
    let __x ← CoreTerm.instantiate cx d is ts

      | (at_, rt) => do
        let __x ← cx.use a
        match __x with
          | (t, cx) => do
            CoreTerm.noLeak cx t
            have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x rt
            if (!CoreTerm.sub cx t at_) = true then do
                let __r ← throw (toString "call " ++ toString f ++ toString ": argument type mismatch")
                __do_jp __r
              else __do_jp ()
  | CoreTerm.Op.borrow c a => do
    let y ← CoreTerm.varOf a
    let t ← cx.peek a
    let r ← CoreTerm.lendOf cx c t
    let __x ← cx.use a
    match __x with
      | (fst, c₁) =>
        { inames := c₁.inames, facts := c₁.facts, vars := c₁.vars, saws := c₁.saws, torn := c₁.torn,

  | CoreTerm.Op.cvt sg w a => do
    let ta ← cx.peek a
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x (CoreLang.Ty.fix sg w)
    if (!CoreTerm.cvtSrc ta) = true then do
        let __r ← throw "cvt needs an integer"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.toInt a => do
    let ta ← cx.peek a
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x CoreLang.Ty.int
    if (!CoreTerm.toIntSrc ta) = true then do
        let __r ← throw "toInt needs an unsigned fix or a lanes word (a signed one: cvt it, or test fcmp lt first)"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.asLanes sg w a => do
    let ta ← cx.peek a
    have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x (CoreLang.Ty.lanes sg w)
    if (CoreTerm.intOf ta).isNone = true then do
        let __r ← throw "asLanes needs an Int"
        __do_jp __r
      else __do_jp ()
  | CoreTerm.Op.fop o a b => do
    let ta ← cx.peek a
    let tb ← cx.peek b
    match CoreTerm.fixOf ta with
      | none => throw "fop needs fixed-width integers"
      | some p =>
        have __do_jp := fun __r => cx.bind x (CoreLang.Ty.fix p.fst p.snd);
        if (CoreTerm.fixOf tb != some p) = true then do
          let __r ← throw "fop: the operands have different fixed-width types"
          __do_jp __r
        else __do_jp ()
  | CoreTerm.Op.fcmp o a b => do
    let ta ← cx.peek a
    let tb ← cx.peek b
    match CoreTerm.fixOf ta with
      | none => throw "fcmp needs fixed-width integers"
      | some p =>
        have __do_jp := fun __r => cx.bind x CoreLang.Ty.int;
        if (CoreTerm.fixOf tb != some p) = true then do
          let __r ← throw "fcmp: the operands have different fixed-width types"
          __do_jp __r
        else __do_jp ()
  | CoreTerm.Op.fop1 o a => do
    let ta ← cx.peek a
    match CoreTerm.fixOf ta with
      | none => throw "fop1 needs a fixed-width integer"
      | some p => cx.bind x (CoreLang.Ty.fix p.fst p.snd)
  | CoreTerm.Op.lane a j => do
    let ta ← cx.peek a
    match CoreTerm.lanesOf ta with
      | none => throw "lane needs a lanes word"
      | some p => do
        let tj ← cx.peek j
        have __do_jp : Unit → CoreTerm.M CoreTerm.Cx := fun __r => cx.bind x (CoreLang.Ty.fix p.fst p.snd)
        if (CoreTerm.intOf tj).isNone = true then do
            let __r ← throw "lane: the lane number must be an Int"
            __do_jp __r
          else __do_jp ()
  | CoreTerm.Op.setLane a j v => do

TermClose.tb : Nat → CoreLang.Ty → Bool
  eq_def ∀ (c : Nat) (x : CoreLang.Ty),
  TermClose.tb c x =
    match x with
    | CoreLang.Ty.tvar k => decide (k < c)
    | a.prod b => TermClose.tb c a && TermClose.tb c b
    | a.sum b => TermClose.tb c a && TermClose.tb c b
    | a.code b k => TermClose.tb c a && TermClose.tb c b
    | CoreLang.Ty.array i t => TermClose.tb c t
    | CoreLang.Ty.ptr μ t => TermClose.tb c t
    | CoreLang.Ty.ptrOpt μ t => TermClose.tb c t
    | CoreLang.Ty.loc μ t => TermClose.tb c t
    | t.mu => TermClose.tb (c + 1) t
    | CoreLang.Ty.tex κ t => TermClose.tb (c + 1) t
    | t.iall => TermClose.tb c t
    | t.iex => TermClose.tb c t
    | t.refine ψ => TermClose.tb c t
    | x => true
TermScope.argOK : CoreLang.Ty → Bool
  := fun t => TermScope.wrapOK t && !TermScope.rcar t
TermScope.drf : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  TermScope.drf x =
    match x with
    | a.prod b => TermScope.drf a && TermScope.drf b
    | a.sum b => TermScope.drf a && TermScope.drf b
    | a.code b k => TermScope.drf a && TermScope.drf b
    | CoreLang.Ty.array i t => TermScope.drf t
    | CoreLang.Ty.ptr μ t => TermScope.drfM μ && TermScope.drf t
    | CoreLang.Ty.ptrOpt μ t => TermScope.drfM μ && TermScope.drf t
    | CoreLang.Ty.loc μ t => TermScope.drfM μ && TermScope.drf t
    | t.mu => TermScope.drf t
    | CoreLang.Ty.tex κ t => TermScope.drf t
    | τ.iall => false
    | t.iex => TermScope.drf t
    | t.refine ψ => TermScope.drf t
    | x => true
TermScope.drfM : CoreLang.Mode → Bool
  := fun x =>
  match x with
  | CoreLang.Mode.rgn r => false
  | x => true
TermScope.rcar : CoreLang.Ty → Bool
  := fun t => TermScope.rcarE [] t
TermScope.rcarE : List Bool → CoreLang.Ty → Bool
  eq_def ∀ (env : List Bool) (x : CoreLang.Ty),
  TermScope.rcarE env x =
    match x with
    | CoreLang.Ty.word r P => true
    | CoreLang.Ty.word2 r P => true
    | CoreLang.Ty.tok r st => true
    | CoreLang.Ty.tok2 r P i0 i1 => true
    | CoreLang.Ty.tokE r P e => true
    | CoreLang.Ty.share r => true
    | a.prod b => TermScope.rcarE env a || TermScope.rcarE env b
    | a.sum b => TermScope.rcarE env a || TermScope.rcarE env b
    | CoreLang.Ty.tvar k => env.getD k false
    | t.mu => TermScope.rcarE (TermScope.rcarE (false :: env) t :: env) t
    | CoreLang.Ty.tex κ t => TermScope.rcarE (false :: env) t
    | t.iall => TermScope.rcarE env t
    | t.iex => TermScope.rcarE env t
    | t.refine ψ => TermScope.rcarE env t
    | x => false
TermScope.wrapOK : CoreLang.Ty → Bool
  := fun t => TermScope.wrapOKE [] t
TermScope.wrapOKE : List Bool → CoreLang.Ty → Bool
  eq_def ∀ (env : List Bool) (x : CoreLang.Ty),
  TermScope.wrapOKE env x =
    match x with
    | a.sum b => !TermScope.rcarE env a && !TermScope.rcarE env b && TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | t.iex => !TermScope.rcarE env t && TermScope.wrapOKE env t
    | t.refine ψ => !TermScope.rcarE env t && TermScope.wrapOKE env t
    | CoreLang.Ty.tex κ t => !TermScope.rcarE (false :: env) t && TermScope.wrapOKE (false :: env) t
    | t.mu => TermScope.wrapOKE (TermScope.rcarE (false :: env) t :: env) t
    | a.prod b => TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | a.code b k => TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | CoreLang.Ty.array i t => TermScope.wrapOKE env t
    | CoreLang.Ty.ptr μ t => TermScope.wrapOKE env t
    | CoreLang.Ty.ptrOpt μ t => TermScope.wrapOKE env t
    | CoreLang.Ty.loc μ t => TermScope.wrapOKE env t
    | t.iall => TermScope.wrapOKE env t
    | x => true
