TermRun.curCfg : TermRun.Cfg → Nat → TermRun.TTh → String → CoreTerm.Expr → TermRun.Cfg
  := fun c i th x e =>
  match th.tidLbl with
  | some l =>
    { mem := c.mem, pairs := c.pairs,
      ths :=
        c.ths.set! i
          { term := e,
            env := (x, CoreLang.Ty.tid, [TermRun.cst (WeakMem.Val.loc { label := l, offset := 0 } 0)]) :: th.env,
            σs := th.σs, stack := th.stack, vs := th.vs, promises := th.promises, nalloc := th.nalloc,
            undefs := th.undefs, tidLbl := th.tidLbl, permit := th.permit },
      lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace }
  | none =>
    have l := TermRun.freshLabel c.lbase c.n i th.nalloc;
    { mem := c.mem, pairs := c.pairs,
      ths :=
        c.ths.set! i
          { term := e,
            env := (x, CoreLang.Ty.tid, [TermRun.cst (WeakMem.Val.loc { label := l, offset := 0 } 0)]) :: th.env,
            σs := th.σs, stack := th.stack, vs := th.vs, promises := th.promises, nalloc := th.nalloc + 1,
            undefs := th.undefs, tidLbl := some l, permit := th.permit },
      lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace }
