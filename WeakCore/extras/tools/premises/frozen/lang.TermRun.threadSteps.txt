TermRun.threadSteps : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → List TermRun.Cfg
  := fun p k arch c i =>
  match c.ths[i]? with
  | none => []
  | some th =>
    List.map
      (fun x =>
        match x with
        | (mem, pairs, th') =>
          { mem := mem, pairs := pairs, ths := c.ths.set! i th', lbase := c.lbase, boxFence := c.boxFence, cap := c.cap,
            trace := c.trace })
      (TermRun.tstep (TermRun.gOf p k arch c i) c.mem c.pairs th)
