CoreLang.zeroArmP : (CoreLang.Ty → Bool) → CoreLang.IEnv → CoreLang.KEnv → CoreLang.Ty → Option Bool
  eq_def ∀ (sel : CoreLang.Ty → Bool) (env : CoreLang.IEnv) (ke : CoreLang.KEnv) (x : CoreLang.Ty),
  CoreLang.zeroArmP sel env ke x =
    match x with
    | a.sum b => CoreLang.zeroArmP sel env ke a <|> CoreLang.zeroArmP sel env ke b
    | CoreLang.Ty.void => none
    | t.refine ψ => Option.map (fun x => x && CoreLang.Pred.zeroHolds env ψ) (CoreLang.zeroArmP sel env ke t)
    | t.iall => CoreLang.zeroArmP sel (none :: env) ke t
    | t.iex => CoreLang.zeroArmP sel (some 0 :: env) ke t
    | t.mu => CoreLang.zeroArmP sel env (none :: ke) t
    | CoreLang.Ty.tex κ t => Option.map (fun x => false) (CoreLang.zeroArmP sel env (none :: ke) t)
    | t => if sel t = true then some (CoreLang.zeroableIn env ke t) else none
CoreLang.zeroableIn : CoreLang.IEnv → CoreLang.KEnv → CoreLang.Ty → Bool
  eq_def ∀ (env : CoreLang.IEnv) (ke : CoreLang.KEnv) (x : CoreLang.Ty),
  CoreLang.zeroableIn env ke x =
    match x with
    | CoreLang.Ty.unit => true
    | CoreLang.Ty.void => false
    | CoreLang.Ty.int => true
    | CoreLang.Ty.intS i => CoreLang.Idx.evalIn env i == some 0
    | CoreLang.Ty.fix sg w => true
    | CoreLang.Ty.lanes sg w => true
    | a.prod b => CoreLang.zeroableIn env ke a && CoreLang.zeroableIn env ke b
    | a.sum b =>
      match (a.sum b).tier with
      | CoreLang.Tier.niche =>
        (CoreLang.zeroArmP CoreLang.Ty.payloadFree env ke a <|> CoreLang.zeroArmP CoreLang.Ty.payloadFree env ke b) ==
          some true
      | CoreLang.Tier.wide =>
        (CoreLang.zeroArmP (fun x => true) env ke a <|> CoreLang.zeroArmP (fun x => true) env ke b) == some true
      | CoreLang.Tier.boxed => false
    | σ.code τ => false
    | CoreLang.Ty.array i t => CoreLang.zeroableIn env ke t
    | CoreLang.Ty.ptr μ τ => false
    | CoreLang.Ty.ptrOpt μ τ => true
    | CoreLang.Ty.loc μ (CoreLang.Ty.array i τ) => CoreLang.Idx.evalIn env i == some 0
    | CoreLang.Ty.loc μ τ => false
    | CoreLang.Ty.word r P => false
    | CoreLang.Ty.tok r st => false
    | CoreLang.Ty.word2 r P => false
    | CoreLang.Ty.tok2 r P i0 i1 => false
    | CoreLang.Ty.epair r P => false
    | CoreLang.Ty.tokE r P e => false
    | CoreLang.Ty.share r => false
    | CoreLang.Ty.cap s r => false
    | CoreLang.Ty.tvar k => ke.ask k false fun x => x.zero
    | t.mu => CoreLang.zeroableIn env (none :: ke) t
    | CoreLang.Ty.tex κ τ => false
    | t.iall => CoreLang.zeroableIn (none :: env) ke t
    | t.iex => CoreLang.zeroableIn (some 0 :: env) ke t
    | t.refine ψ => CoreLang.zeroableIn env ke t && CoreLang.Pred.zeroHolds env ψ

TermRun.evalOp : TermRun.G →
  TermRun.Env → List CoreLang.Ty → TermRun.St → CoreTerm.Op → List (TermRun.St × CoreLang.Ty × List TermRun.Word)
  := fun g env σs s x =>
  match x with
  | CoreTerm.Op.atom a =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (t, ws) => (s, t, ws))
        (TermRun.evalAt env a))
  | CoreTerm.Op.pair a b =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← TermRun.evalAt env b
          match __x with
            | (u, us) => pure (s, t.prod u, ws ++ us)
  | CoreTerm.Op.inl τ a =>
    match TermRun.evalAt env a with
    | some (t, ws) =>
      have S := t.sum (TermRun.close σs τ);
      List.flatMap
        (fun x =>
          match x with
          | (s1, j, p) =>
            List.map
              (fun x =>
                match x with
                | (s2, q) => (s2, S, q))
              (TermRun.injM g s1 S j p))
        (TermRun.projM g s t ws)
    | none => []
  | CoreTerm.Op.inr τ a =>
    match TermRun.evalAt env a with
    | some (t, ws) =>
      have τ' := TermRun.close σs τ;
      have S := τ'.sum t;
      List.flatMap
        (fun x =>
          match x with
          | (s1, j, p) =>
            List.map
              (fun x =>
                match x with
                | (s2, q) => (s2, S, q))
              (TermRun.injM g s1 S ((TermRun.armsOf τ').length + j) p))
        (TermRun.projM g s t ws)
    | none => []
  | CoreTerm.Op.add a b =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (fst, ws) => do
          let __x ← TermRun.evalAt env b
          match __x with
            | (fst, us) =>
              match ws, us with
              | [x], [y] => pure (s, CoreLang.Ty.int, [TermRun.addW x y])
              | x, x_1 => none
  | CoreTerm.Op.sub a b =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (fst, ws) => do
          let __x ← TermRun.evalAt env b
          match __x with
            | (fst, us) =>
              match ws, us with
              | [x], [y] => pure (s, CoreLang.Ty.int, [TermRun.subW x y])
              | x, x_1 => none
  | CoreTerm.Op.new a =>
    match TermRun.evalAt env a with
    | none => []
    | some (t, ws) =>
      match TermRun.accs t with
      | none => []
      | some us =>
        have lbl := TermRun.freshLabel g.lbase g.n g.i s.nalloc;
        List.map
          (fun x =>
            match x with
            | (s', pw) => (s', CoreLang.Ty.ptr CoreLang.Mode.own t, [pw]))
          (TermRun.alloc g.k g.arch lbl us ws s)
  | CoreTerm.Op.null μ τ => [(s, CoreLang.Ty.ptrOpt μ (TermRun.close σs τ), [TermRun.cst WeakMem.Val.null])]
  | CoreTerm.Op.fold a τ =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (fst, ws) => (s, TermRun.close σs τ, ws))
        (TermRun.evalAt env a))
  | CoreTerm.Op.publish a =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) =>
          match t with
          | CoreLang.Ty.ptr μ u => pure (TermRun.fenceS s WeakMem.Fence.st, CoreLang.Ty.ptr CoreLang.Mode.pub u, ws)
          | (CoreLang.Ty.loc μ (CoreLang.Ty.array n u)).refine ψ =>
            pure
              (TermRun.fenceS s WeakMem.Fence.st, (CoreLang.Ty.loc CoreLang.Mode.pub (CoreLang.Ty.array n u)).refine ψ,
                ws)
          | x => none
  | CoreTerm.Op.intoRgn r a =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) =>
          match t with
          | CoreLang.Ty.ptr μ u =>
            pure (TermRun.fenceS s WeakMem.Fence.st, CoreLang.Ty.ptr (CoreLang.Mode.rgn (CoreLang.Idx.lit 0)) u, ws)
          | (CoreLang.Ty.loc μ (CoreLang.Ty.array n u)).refine ψ =>
            pure
              (TermRun.fenceS s WeakMem.Fence.st,
                (CoreLang.Ty.loc (CoreLang.Mode.rgn (CoreLang.Idx.lit 0)) (CoreLang.Ty.array n u)).refine ψ, ws)
          | x => none
  | CoreTerm.Op.fld i a =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) =>
          have __do_jp := fun pw => do
            let __x ← CoreTerm.cellOf t
            match __x with
              | (μ, u) =>
                have __do_jp := fun __x =>
                  match __x with
                  | (c1, c2) =>
                    have __do_jp := fun off =>
                      match pw.val with
                      | WeakMem.Val.loc b g =>
                        pure
                          (s, CoreLang.Ty.loc μ (if (i == 0) = true then c1 else c2),
                            [{ val := WeakMem.Val.loc (TermRun.offs b off) g, view := pw.view, rts := pw.rts }])
                      | x => none;
                    if (i == 0) = true then do
                      let off ← some 0
                      __do_jp off
                    else do
                      let off ← TermRun.nWords <$> TermRun.accs c1
                      __do_jp off;
                match CoreTerm.unfold1 u with
                | c1.prod c2 => do
                  let __x ← some (c1, c2)
                  __do_jp __x
                | x => do
                  let __x ← none
                  __do_jp __x;
          match ws with
          | [w] => do
            let pw ← some w
            __do_jp pw
          | x => do
            let pw ← none
            __do_jp pw
  | CoreTerm.Op.load acq a =>
    match TermRun.evalAt env a with
    | some (t, [pw]) =>
      match CoreTerm.cellOf t, TermRun.addrOf pw with
      | some (fst, u), some b =>
        have rt := if (CoreTerm.wordOf t).isSome = true then CoreLang.Ty.int else u;
        match TermRun.accs u with
        | some us =>
          List.map
            (fun x =>
              match x with
              | (s', ws) => (s', rt, ws))
            (TermRun.loadUnits g.arch pw b acq us 0 s)
        | none => []
      | x, x_1 => []
    | x => []
  | CoreTerm.Op.store rel a v tok =>
    match TermRun.evalAt env a, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, ws) =>
      match CoreTerm.cellOf t, TermRun.addrOf pw with
      | some (fst, u), some b =>
        match TermRun.accs u with
        | some us => List.map (fun x => (x, CoreLang.Ty.unit, [])) (TermRun.storeUnits g.k g.arch pw b rel us ws 0 s)
        | none => []
      | x, x_1 => []
    | x, x_1 => []
  | CoreTerm.Op.fence f => [(TermRun.fenceS s (TermRun.toWFence f), CoreLang.Ty.unit, [])]
  | CoreTerm.Op.mkWord r P v tok =>
    if P.ghost = true then
      [(s, (CoreLang.Ty.word (CoreLang.Idx.lit 0) P).prod (CoreLang.Ty.tok (CoreLang.Idx.lit 0) none), [])]
    else
      TermRun.ol
        (Option.map
          (fun x =>
            match x with
            | (fst, ws) => (s, CoreLang.Ty.word (CoreLang.Idx.lit 0) P, ws))
          (TermRun.evalAt env v))
  | CoreTerm.Op.mkWord2 r P v0 v1 tok =>
    TermRun.ol do
      let __x ← TermRun.evalAt env v0
      match __x with
        | (fst, w0) => do
          let __x ← TermRun.evalAt env v1
          match __x with
            | (fst, w1) => pure (s, CoreLang.Ty.word2 (CoreLang.Idx.lit 0) P, w0 ++ w1)
  | CoreTerm.Op.ldp acq a n0 n1 =>
    match TermRun.evalAt env a with
    | some (t, [pw]) =>
      if CoreTerm.pairRef t = true then
        match TermRun.addrOf pw with
        | some b =>
          List.map
            (fun x =>
              match x with
              | (s', ws) => (s', CoreLang.Ty.int.prod CoreLang.Ty.int, ws))
            (TermRun.loadUnits g.arch pw b acq [TermRun.Acc.two] 0 s)
        | none => []
      else []
    | x => []
  | CoreTerm.Op.stp rel a v0 v1 tok =>
    match TermRun.evalAt env a, TermRun.evalAt env v0, TermRun.evalAt env v1 with
    | some (t, [pw]), some (fst, w0), some (fst_1, w1) =>
      match CoreTerm.word2Of t, TermRun.addrOf pw with
      | some val, some b =>
        List.map (fun x => (x, CoreLang.Ty.unit, []))
          (TermRun.storeUnits g.k g.arch pw b rel [TermRun.Acc.two] (w0 ++ w1) 0 s)
      | x, x_1 => []
    | x, x_1, x_2 => []
  | CoreTerm.Op.mkEPair r P v0 v1 =>
    TermRun.ol do
      let __x ← TermRun.evalAt env v0
      match __x with
        | (fst, w0) => do
          let __x ← TermRun.evalAt env v1
          match __x with
            | (fst, w1) => pure (s, CoreLang.Ty.epair (CoreLang.Idx.lit 0) P, w0 ++ w1)
  | CoreTerm.Op.mkTicket r N P v tok =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (fst, ws) =>
            (s, (CoreLang.Ty.word (CoreLang.Idx.lit 0) N).prod (CoreLang.Ty.word (CoreLang.Idx.lit 0) P), ws ++ ws))
        (TermRun.evalAt env v))
  | CoreTerm.Op.claim a tok => [(s, CoreLang.Ty.unit, [])]
  | CoreTerm.Op.fetch o acq rel a v =>
    match TermRun.evalAt env a, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, [nw]) =>
      match TermRun.addrOf pw with
      | some b =>
        List.map
          (fun x =>
            match x with
            | (mem', th') =>
              have rd := th'.regs 1;
              ({ mem := mem', pairs := s.pairs, vs := th'.vs, promises := th'.promises, nalloc := s.nalloc,
                  undefs := s.undefs + TermRun.countUndef [rd] },
                TermRun.fetchRtTy t, [rd]))
          (Promises.stepRmw g.k g.arch s.mem s.pairs (TermRun.mth s pw nw nw) 1 b o.kind (TermRun.dataE 1 nw) [0] acq
            rel)
      | none => []
    | x, x_1 => []
  | CoreTerm.Op.pack a w τ =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (fst, ws) => (s, TermRun.close σs τ, ws))
        (TermRun.evalAt env a))
  | CoreTerm.Op.packT a σ τ =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (fst, ws) => (s, TermRun.close σs τ, ws))
        (TermRun.evalAt env a))
  | CoreTerm.Op.validate y =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (t, ws) => (s, t, ws))
        (TermRun.lookupE env y))
  | CoreTerm.Op.call f is a ts => []
  | CoreTerm.Op.spawn f a T ts => []
  | CoreTerm.Op.borrow c t => [(s, CoreLang.Ty.cap c (CoreLang.Idx.lit 0), [])]
  | CoreTerm.Op.freeze t => [(s, CoreLang.Ty.cap CoreLang.Cap.frz (CoreLang.Idx.lit 0), [])]
  | CoreTerm.Op.cvt sg w a =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let x ← TermRun.numOf ws
          pure (s, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.cvtN t w x))])
  | CoreTerm.Op.toInt a =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env a
      let x ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num x)])
  | CoreTerm.Op.asLanes sg w a =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env a
      let x ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.lanes sg w, [TermRun.cst (WeakMem.Val.num (x % TermRun.W64))])
  | CoreTerm.Op.fop o a b =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← CoreTerm.fixOf t
          match __x with
            | (sg, w) => do
              let x ← TermRun.numOf ws
              let __do_lift ← TermRun.evalAt env b
              let y ← TermRun.numOf __do_lift.snd
              pure (s, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.fopN sg w o x y))])
  | CoreTerm.Op.fcmp o a b =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← CoreTerm.fixOf t
          match __x with
            | (sg, w) => do
              let x ← TermRun.numOf ws
              let __do_lift ← TermRun.evalAt env b
              let y ← TermRun.numOf __do_lift.snd
              pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (TermRun.fcmpN sg w o x y))])
  | CoreTerm.Op.fop1 o a =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← CoreTerm.fixOf t
          match __x with
            | (sg, w) => do
              let x ← TermRun.numOf ws
              pure (s, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.fop1N sg w o x))])
  | CoreTerm.Op.lane a j =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← CoreTerm.lanesOf t
          match __x with
            | (sg, w) => do
              let x ← TermRun.numOf ws
              let __do_lift ← TermRun.evalAt env j
              let jj ← TermRun.numOf __do_lift.snd
              pure (s, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.laneN w x (jj % w.lanes)))])
  | CoreTerm.Op.setLane a j v =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (t, ws) => do
          let __x ← CoreTerm.lanesOf t
          match __x with
            | (sg, w) => do
              let x ← TermRun.numOf ws
              let __do_lift ← TermRun.evalAt env j
              let jj ← TermRun.numOf __do_lift.snd
              let __do_lift ← TermRun.evalAt env v
              let y ← TermRun.numOf __do_lift.snd
              pure (s, CoreLang.Ty.lanes sg w, [TermRun.cst (WeakMem.Val.num (TermRun.setLaneN w x (jj % w.lanes) y))])
  | CoreTerm.Op.ldl b i =>
    match TermRun.evalAt env b, TermRun.evalAt env i with
    | some (t, [pw]), some (fst, iws) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match CoreTerm.lanesOf u with
        | some (sg, w) =>
          List.filterMap
            (fun x =>
              match x with
              | (s1, old) =>
                Option.map
                  (fun x =>
                    (s1, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.laneN w x (j % w.lanes)))]))
                  (TermRun.numOf old))
            (TermRun.loadUnits g.arch pw (TermRun.offs a (j / w.lanes)) false [TermRun.Acc.one] 0 s)
        | none => []
      | x, x_1, x_2 => []
    | x, x_1 => []
  | CoreTerm.Op.ldlw b wi jl =>
    match TermRun.evalAt env b, TermRun.evalAt env wi, TermRun.evalAt env jl with
    | some (t, [pw]), some (fst, iws), some (fst_1, jws) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.numOf jws, TermRun.addrOf pw with
      | some (fst, u), some i, some j, some a =>
        match CoreTerm.lanesOf u with
        | some (sg, w) =>
          List.filterMap
            (fun x =>
              match x with
              | (s1, old) =>
                Option.map
                  (fun x =>
                    (s1, CoreLang.Ty.fix sg w, [TermRun.cst (WeakMem.Val.num (TermRun.laneN w x (j % w.lanes)))]))
                  (TermRun.numOf old))
            (TermRun.loadUnits g.arch pw (TermRun.offs a i) false [TermRun.Acc.one] 0 s)
        | none => []
      | x, x_1, x_2, x_3 => []
    | x, x_1, x_2 => []
  | CoreTerm.Op.stlw b wi jl v =>
    match TermRun.evalAt env b, TermRun.evalAt env wi, TermRun.evalAt env jl, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, iws), some (fst_1, jws), some (fst_2, vs) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.numOf jws, TermRun.numOf vs, TermRun.addrOf pw with
      | some (fst, u), some i, some j, some y, some a =>
        match CoreTerm.lanesOf u with
        | some (fst, w) =>
          List.flatMap
            (fun x =>
              match x with
              | (s1, old) =>
                match TermRun.numOf old with
                | some x =>
                  List.map (fun s2 => (s2, CoreLang.Ty.unit, []))
                    (TermRun.storeUnits g.k g.arch pw (TermRun.offs a i) false [TermRun.Acc.one]
                      [TermRun.cst (WeakMem.Val.num (TermRun.setLaneN w x (j % w.lanes) y))] 0 s1)
                | none => [])
            (TermRun.loadUnits g.arch pw (TermRun.offs a i) false [TermRun.Acc.one] 0 s)
        | none => []
      | x, x_1, x_2, x_3, x_4 => []
    | x, x_1, x_2, x_3 => []
  | CoreTerm.Op.stl b i v =>
    match TermRun.evalAt env b, TermRun.evalAt env i, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, iws), some (fst_1, vs) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.numOf vs, TermRun.addrOf pw with
      | some (fst, u), some j, some y, some a =>
        match CoreTerm.lanesOf u with
        | some (fst, w) =>
          have a' := TermRun.offs a (j / w.lanes);
          List.flatMap
            (fun x =>
              match x with
              | (s1, old) =>
                match TermRun.numOf old with
                | some x =>
                  List.map (fun s2 => (s2, CoreLang.Ty.unit, []))
                    (TermRun.storeUnits g.k g.arch pw a' false [TermRun.Acc.one]
                      [TermRun.cst (WeakMem.Val.num (TermRun.setLaneN w x (j % w.lanes) y))] 0 s1)
                | none => [])
            (TermRun.loadUnits g.arch pw a' false [TermRun.Acc.one] 0 s)
        | none => []
      | x, x_1, x_2, x_3 => []
    | x, x_1, x_2 => []
  | CoreTerm.Op.endBorrow t => [(s, CoreLang.Ty.unit, [])]
  | CoreTerm.Op.xchg a v =>
    match TermRun.evalAt env a, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, ws) =>
      match CoreTerm.cellOf t, TermRun.addrOf pw with
      | some (fst, u), some b =>
        match TermRun.accs u with
        | some us =>
          List.flatMap
            (fun x =>
              match x with
              | (s1, old) => List.map (fun s2 => (s2, u, old)) (TermRun.storeUnits g.k g.arch pw b false us ws 0 s1))
            (TermRun.loadUnits g.arch pw b false us 0 s)
        | none => []
      | x, x_1 => []
    | x, x_1 => []
  | CoreTerm.Op.fnptr f =>
    match List.findIdx? (fun x => x.name == f) g.p, List.find? (fun x => x.name == f) g.p with
    | some i, some d => [(s, d.argTy.code d.retTy, [TermRun.cst (WeakMem.Val.num i)])]
    | x, x_1 => []
  | CoreTerm.Op.callc f a => []
  | CoreTerm.Op.rem a n =>
    TermRun.ol do
      let __x ← TermRun.evalAt env a
      match __x with
        | (fst, ws) => do
          let __x ← TermRun.evalAt env n
          match __x with
            | (fst, ns) => do
              let x ← TermRun.numOf ws
              let m ← TermRun.numOf ns
              if (m == 0) = true then none else pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (x % m))])
  | CoreTerm.Op.nameI a =>
    TermRun.ol
      (Option.map
        (fun x =>
          match x with
          | (t, ws) => (s, t, ws))
        (TermRun.evalAt env a))
  | CoreTerm.Op.xchgAt b i v =>
    match TermRun.evalAt env b, TermRun.evalAt env i, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, iws), some (fst_1, xs) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match TermRun.accs u with
        | some us =>
          have a' := TermRun.offs a (j * TermRun.nWords us);
          List.flatMap
            (fun x =>
              match x with
              | (s1, old) => List.map (fun s2 => (s2, u, old)) (TermRun.storeUnits g.k g.arch pw a' false us xs 0 s1))
            (TermRun.loadUnits g.arch pw a' false us 0 s)
        | none => []
      | x, x_1, x_2 => []
    | x, x_1, x_2 => []
  | CoreTerm.Op.loadAt b i =>
    match TermRun.evalAt env b, TermRun.evalAt env i with
    | some (t, [pw]), some (fst, iws) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match TermRun.accs u with
        | some us =>
          List.map
            (fun x =>
              match x with
              | (s1, old) => (s1, u, old))
            (TermRun.loadUnits g.arch pw (TermRun.offs a (j * TermRun.nWords us)) false us 0 s)
        | none => []
      | x, x_1, x_2 => []
    | x, x_1 => []
  | CoreTerm.Op.bits o a b =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env a
      let x ← TermRun.numOf __do_lift.snd
      let __do_lift ← TermRun.evalAt env b
      let y ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (TermRun.bitN o x y))])
  | CoreTerm.Op.bit1 o a =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env a
      let x ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (TermRun.bit1N o x))])
  | CoreTerm.Op.vmatch lo hi b =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env lo
      let x ← TermRun.numOf __do_lift.snd
      let __do_lift ← TermRun.evalAt env hi
      let y ← TermRun.numOf __do_lift.snd
      let __do_lift ← TermRun.evalAt env b
      let z ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (TermRun.vmatchN x y z))])
  | CoreTerm.Op.band a m =>
    TermRun.ol do
      let __do_lift ← TermRun.evalAt env a
      let x ← TermRun.numOf __do_lift.snd
      let __do_lift ← TermRun.evalAt env m
      let y ← TermRun.numOf __do_lift.snd
      pure (s, CoreLang.Ty.int, [TermRun.cst (WeakMem.Val.num (x % TermRun.W64 &&& y))])
  | CoreTerm.Op.zero τ =>
    TermRun.ol
      (have τ' := TermRun.close σs τ;
      do
      let z ← TermRun.zeroWords τ'
      pure (s, τ', List.map TermRun.cst z))
  | CoreTerm.Op.xchgAtF b i p v =>
    match TermRun.evalAt env b, TermRun.evalAt env i, TermRun.evalAt env v with
    | some (t, [pw]), some (fst, iws), some (fst_1, xs) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match TermRun.accs u, TermRun.compOff u p with
        | some us, some (c, o) =>
          match TermRun.accs c with
          | some cs =>
            have a' := TermRun.offs a (j * TermRun.nWords us + o);
            List.flatMap
              (fun x =>
                match x with
                | (s1, old) => List.map (fun s2 => (s2, c, old)) (TermRun.storeUnits g.k g.arch pw a' false cs xs 0 s1))
              (TermRun.loadUnits g.arch pw a' false cs 0 s)
          | none => []
        | x, x_1 => []
      | x, x_1, x_2 => []
    | x, x_1, x_2 => []
  | CoreTerm.Op.loadAtF b i p =>
    match TermRun.evalAt env b, TermRun.evalAt env i with
    | some (t, [pw]), some (fst, iws) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match TermRun.accs u, TermRun.compOff u p with
        | some us, some (c, o) =>
          match TermRun.accs c with
          | some cs =>
            List.map
              (fun x =>
                match x with
                | (s1, old) => (s1, c, old))
              (TermRun.loadUnits g.arch pw (TermRun.offs a (j * TermRun.nWords us + o)) false cs 0 s)
          | none => []
        | x, x_1 => []
      | x, x_1, x_2 => []
    | x, x_1 => []
  | CoreTerm.Op.newBuf n τ =>
    TermRun.ol do
      let __x ← TermRun.evalAt env n
      match __x with
        | (fst, ws) => do
          let k ← TermRun.numOf ws
          have τ' : CoreLang.Ty := TermRun.close σs τ
          let us ← TermRun.accs τ'
          have w : Nat := TermRun.nWords us
          have z : Option (List WeakMem.Val) :=
            if CoreLang.zeroableIn [] [] τ' = true then TermRun.zeroWords τ' else none
          have lbl : WeakMem.AllocLabel := TermRun.freshLabel g.lbase g.n g.i s.nalloc
          match TermRun.allocBuf lbl (k * w) z s with
            | (s', pw) =>
              have ini := if z.isSome = true then k else 0;
              pure
                (s',
                  (CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array (CoreLang.Idx.lit k) τ')).refine
                    (CoreLang.Pred.ini (CoreLang.Idx.lit ini)),
                  [pw])
  | CoreTerm.Op.idx b i =>
    TermRun.ol do
      let __x ← TermRun.evalAt env b
      match __x with
        | (t, ws) =>
          have __do_jp := fun pw => do
            let __x ← TermRun.evalAt env i
            match __x with
              | (fst, iws) => do
                let j ← TermRun.numOf iws
                let __x ← TermRun.bufElem t
                match __x with
                  | (μ, u) => do
                    let w ← TermRun.nWords <$> TermRun.accs u
                    match pw.val with
                      | WeakMem.Val.loc a gg =>
                        pure
                          (s, CoreLang.Ty.loc μ u,
                            [{ val := WeakMem.Val.loc (TermRun.offs a (j * w)) gg, view := pw.view, rts := pw.rts }])
                      | x => none;
          match ws with
          | [w] => do
            let pw ← some w
            __do_jp pw
          | x => do
            let pw ← none
            __do_jp pw
  | CoreTerm.Op.bufInit b i x =>
    match TermRun.evalAt env b, TermRun.evalAt env i, TermRun.evalAt env x with
    | some (t, [pw]), some (fst, iws), some (fst_1, xs) =>
      match TermRun.bufElem t, TermRun.numOf iws, TermRun.addrOf pw with
      | some (fst, u), some j, some a =>
        match TermRun.accs u, t with
        | some us, (CoreLang.Ty.loc μ (CoreLang.Ty.array n τ)).refine ψ =>
          List.map
            (fun s' =>
              (s', (CoreLang.Ty.loc μ (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini (CoreLang.Idx.lit (j + 1))),
                [pw]))
            (TermRun.storeUnits g.k g.arch pw (TermRun.offs a (j * TermRun.nWords us)) false us xs 0 s)
        | x, x_1 => []
      | x, x_1, x_2 => []
    | x, x_1, x_2 => []
