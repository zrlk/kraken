TermRun.atChoiceCutB : TermRun.Bounds → TermRun.TTh → Bool
TermRun.atSpawnB : TermRun.TTh → Bool
TermRun.isDone : TermRun.TTh → Bool
TermRun.nodeTrunc : Bool → Bool → TermRun.Bounds → TermRun.Cfg → TermRun.Trunc
  := fun promOn spawnOff b c =>
  { maxP := c.ths.any fun th => !TermRun.isDone th && (!promOn || decide (b.maxP ≤ th.promises.length)),
    spawn := spawnOff && c.ths.any TermRun.atSpawnB, io := c.ths.any (TermRun.atChoiceCutB b) }
