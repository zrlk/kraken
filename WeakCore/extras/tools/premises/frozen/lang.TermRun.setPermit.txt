TermRun.setPermit : TermRun.Cfg → Nat → WeakMem.Ts → TermRun.Cfg
  := fun c j v =>
  match c.ths[j]? with
  | some th =>
    { mem := c.mem, pairs := c.pairs,
      ths :=
        c.ths.set! j
          { term := th.term, env := th.env, σs := th.σs, stack := th.stack, vs := th.vs, promises := th.promises,
            nalloc := th.nalloc, undefs := th.undefs, tidLbl := th.tidLbl, permit := some (max (th.permit.getD 0) v) },
      lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace }
  | none => c
