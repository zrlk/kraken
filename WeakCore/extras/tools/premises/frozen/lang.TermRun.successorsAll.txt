TermRun.successorsAll : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Bounds → TermRun.Cfg → List TermRun.Cfg
  := fun p k arch b c =>
  List.flatMap
    (fun i =>
      have ts := List.filter (fun c' => TermRun.certB p k arch b.solo c' i) (TermRun.threadSteps p k arch c i);
      have ps :=
        match c.ths[i]? with
        | some th =>
          if (k.promises && decide (th.promises.length < b.maxP) && !TermRun.isDone th) = true then
            List.filter (fun c' => TermRun.certB p k arch b.solo c' i)
              (List.flatMap
                (fun x =>
                  match x with
                  | (a, v) => List.map (fun w => TermRun.promise c i a v w) (List.range (Array.size c.mem)))
                (TermRun.soloWrites p k arch i b.solo c).eraseDups)
          else []
        | none => [];
      ts ++ ps ++ TermRun.extSuccs p k arch b c i ++ TermRun.wakeSuccs p k arch b c i)
    (List.range c.ths.size)
