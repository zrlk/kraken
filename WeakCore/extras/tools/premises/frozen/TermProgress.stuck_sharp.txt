TermProgress.ExtBy : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun p k arch c i c' =>
  ∃ x n a e th rt ws rs,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e ∧
        (TermRun.ExtNeedsNoProm n th → th.promises = []) ∧
          TermRun.ExtEnabled n th ∧
            TermRun.evalAt th.env a = some (rt, ws) ∧
              n.Admits i ws c.trace rs ∧
                TermRun.Certified p k arch (TermRun.extCfg c i th x n e ws rs) i ∧
                  c' = TermRun.wake (TermRun.extCfg c i th x n e ws rs) i (TermRun.relView th.vs) n ws
TermProgress.MovesBy : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → Nat → TermRun.Cfg → Nat → Prop
  := fun p k arch cap c i =>
  (∃ c', c' ∈ TermRun.threadSteps p k arch c i ∧ TermRun.Certified p k arch c' i) ∨
    (∃ a v w, i < c.ths.size ∧ w < Array.size c.mem ∧ TermRun.Certified p k arch (TermRun.promise c i a v w) i) ∨
      (∃ c', TermProgress.SpawnBy cap c i c') ∨ ∃ c', TermProgress.ExtBy p k arch c i c'
TermProgress.SpawnBy : Nat → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun cap c i c' =>
  ∃ x f a T ts e th rt ws,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.spawn f a T ts) e ∧
        th.promises = [] ∧
          c.ths.size < cap ∧ TermRun.evalAt th.env a = some (rt, ws) ∧ c' = TermSpawnStep.spawnCfg c i th x e f ts T ws
TermProgress.StuckWhy : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → Nat → TermRun.Cfg → Nat → TermRun.TTh → Prop
  | TermProgress.StuckWhy.uncert : ∀ {p : CoreTerm.Prog} {k : Promises.Knobs} {arch : WeakMem.Arch} {cap : Nat} {c : TermRun.Cfg} {i : Nat}
  {th : TermRun.TTh}, th.promises ≠ [] → ¬TermRun.Certified p k arch c i → TermProgress.StuckWhy p k arch cap c i th
  | TermProgress.StuckWhy.atCap : ∀ {p : CoreTerm.Prog} {k : Promises.Knobs} {arch : WeakMem.Arch} {cap : Nat} {c : TermRun.Cfg} {i : Nat}
  {th : TermRun.TTh},
  TermSpawnStep.AtSite th → th.promises = [] → cap ≤ c.ths.size → TermProgress.StuckWhy p k arch cap c i th
  | TermProgress.StuckWhy.parked : ∀ {p : CoreTerm.Prog} {k : Promises.Knobs} {arch : WeakMem.Arch} {cap : Nat} {c : TermRun.Cfg} {i : Nat}
  {th : TermRun.TTh},
  (∃ x a e, th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext CoreTerm.ExtName.park a) e) →
    th.permit = none →
      (∀ (a : WeakMem.Address) (v : WeakMem.Val) (w : Nat),
          w < Array.size c.mem → ¬TermRun.Certified p k arch (TermRun.promise c i a v w) i) →
        TermProgress.StuckWhy p k arch cap c i th

TermSpawn.bumpW : WeakMem.Ts → TermRun.Word → TermRun.Word
  := fun V w => { val := w.val, view := max w.view V, rts := max w.rts V }
TermSpawnStep.AtSite : TermRun.TTh → Prop
  := fun th => ∃ x f a T ts e, th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.spawn f a T ts) e
TermSpawnStep.spawnCfg : TermRun.Cfg →
  Nat → TermRun.TTh → String → CoreTerm.Expr → String → List CoreLang.Ty → CoreLang.Ty → List TermRun.Word → TermRun.Cfg
  := fun c i th x e f ts T ws =>
  TermSpawnStep.spawnT c i (TermSpawnStep.spawnPar c i th x e) (TermSpawnStep.spawnKid c i th f ts T ws)
TermSpawnStep.spawnKid : TermRun.Cfg → Nat → TermRun.TTh → String → List CoreLang.Ty → CoreLang.Ty → List TermRun.Word → TermRun.TTh
  := fun c i th f ts T ws =>
  { term := CoreTerm.Expr.tail f [] (CoreTerm.At.var "a") ts,
    env := [("a", T, List.map (TermSpawn.bumpW (TermSpawnStep.spawnS c th).vs.vRelSt) ws)],
    vs :=
      have __src := (TermSpawnStep.spawnS c th).vs;
      { coh := __src.coh, vRd := __src.vRd, vWr := __src.vWr, vCtrl := __src.vCtrl, vPreBase := __src.vPreBase,
        vRelSt := __src.vRelSt, fwd := __src.fwd },
    tidLbl := some (TermSpawnStep.tidLabel c i th) }
TermSpawnStep.spawnPar : TermRun.Cfg → Nat → TermRun.TTh → String → CoreTerm.Expr → TermRun.TTh
  := fun c i th x e =>
  have __src := TermRun.withS th (TermSpawnStep.spawnS c th);
  { term := e,
    env :=
      (x, CoreLang.Ty.tid, [TermRun.cst (WeakMem.Val.loc { label := TermSpawnStep.tidLabel c i th, offset := 0 } 0)]) ::
        th.env,
    σs := __src.σs, stack := __src.stack, vs := __src.vs, promises := __src.promises,
    nalloc := th.nalloc + if th.tidLbl = none then 2 else 1, undefs := __src.undefs,
    tidLbl := th.tidLbl.orElse fun x => some (TermRun.freshLabel c.lbase c.n i (th.nalloc + 1)),
    permit := __src.permit }
TermSpawnStep.spawnS : TermRun.Cfg → TermRun.TTh → TermRun.St
  := fun c th => TermRun.fenceS (TermRun.stOf c.mem c.pairs th) WeakMem.Fence.sy
TermSpawnStep.spawnT : TermRun.Cfg → Nat → TermRun.TTh → TermRun.TTh → TermRun.Cfg
  := fun c i par kid =>
  { mem := c.mem, pairs := c.pairs, ths := (c.ths.set! i par).push kid, lbase := c.lbase, boxFence := c.boxFence,
    cap := c.cap, trace := c.trace }
TermSpawnStep.tidLabel : TermRun.Cfg → Nat → TermRun.TTh → WeakMem.AllocLabel
  := fun c i th => TermRun.freshLabel c.lbase c.n i th.nalloc
