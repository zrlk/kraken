TermRun.lastMmioW : List TermRun.Event → Nat → Nat
  := fun tr r =>
  (List.findSome?
        (fun e =>
          match e with
          | TermRun.Event.mmioW r' w mo v => if r' = r then some v else none
          | x => none)
        tr.reverse).getD
    0
