CoreTerm.devCellOK : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  CoreTerm.devCellOK x =
    match x with
    | t.refine ψ => CoreTerm.devCellOK t
    | CoreLang.Ty.word r P => true
    | CoreLang.Ty.word2 r P => true
    | CoreLang.Ty.epair r P => true
    | t => CoreLang.bitsIn t
