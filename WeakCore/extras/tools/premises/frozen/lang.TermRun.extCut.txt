TermRun.extCut : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Bounds → TermRun.Cfg → Nat → Bool
  := fun p k arch b c i =>
  match c.ths[i]? with
  | some th =>
    match th.term with
    | CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e =>
      if (decide (TermRun.ExtNeedsNoProm n th) && !th.promises.isEmpty || !decide (TermRun.ExtEnabled n th)) = true then
        false
      else
        match TermRun.evalAt th.env a with
        | some (fst, ws) =>
          (TermRun.extRs b n i ws c.trace).any fun rs =>
            !TermRun.certB p k arch b.solo (TermRun.extCfg c i th x n e ws rs) i &&
              TermRun.certCut p k arch b.solo (TermRun.extCfg c i th x n e ws rs) i
        | none => false
    | x => false
  | none => false
