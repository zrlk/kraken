TermRun.Step : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → TermRun.Cfg → Prop
  | TermRun.Step.thread : ∀ {p : CoreTerm.Prog} {k : Promises.Knobs} {arch : WeakMem.Arch} {c c' : TermRun.Cfg} (i : Nat),
  c' ∈ TermRun.threadSteps p k arch c i → TermRun.Certified p k arch c' i → TermRun.Step p k arch c c'
  | TermRun.Step.prom : ∀ {p : CoreTerm.Prog} {k : Promises.Knobs} {arch : WeakMem.Arch} {c : TermRun.Cfg} (i : Nat) (a : WeakMem.Address)
  (v : WeakMem.Val) (w : WeakMem.Ts),
  i < c.ths.size →
    w < Array.size c.mem →
      TermRun.Certified p k arch (TermRun.promise c i a v w) i → TermRun.Step p k arch c (TermRun.promise c i a v w)
