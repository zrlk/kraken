CoreTerm.ExtName : Type
CoreTerm.ExtName.ok : CoreTerm.ExtName → Bool
  := fun x =>
  match x with
  | CoreTerm.ExtName.mmioR w acc mo => acc.canRead && mo != CoreLang.Mo.rel
  | CoreTerm.ExtName.mmioW w acc mo => acc.canWrite && mo != CoreLang.Mo.acq
  | CoreTerm.ExtName.regAt n w acc st tk k =>
    decide (k < n) && !tk.contains k && decide (w.bits / 8 ≤ st) && decide (st % (w.bits / 8) = 0)
  | x => true
