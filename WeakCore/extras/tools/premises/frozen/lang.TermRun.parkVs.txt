TermRun.TTh : Type
TermRun.TTh.permit : TermRun.TTh → Option WeakMem.Ts
TermRun.TTh.vs : TermRun.TTh → WeakMem.VState
TermRun.parkVs : Nat → TermRun.TTh → WeakMem.VState
  := fun n th =>
  match th.permit with
  | some v =>
    have __src := th.vs;
    { coh := __src.coh, vRd := max th.vs.vRd (min v n), vWr := __src.vWr, vCtrl := __src.vCtrl,
      vPreBase := max th.vs.vPreBase (min v n), vRelSt := __src.vRelSt, mine := __src.mine, fwd := __src.fwd }
  | none => th.vs
