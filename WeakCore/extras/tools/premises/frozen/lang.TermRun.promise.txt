TermRun.promise : TermRun.Cfg → Nat → WeakMem.Address → WeakMem.Val → WeakMem.Ts → TermRun.Cfg
  := fun c i a v w =>
  match c.ths[i]? with
  | none => c
  | some th =>
    { mem := Array.push c.mem { loc := a, val := v, view := w }, pairs := c.pairs,
      ths :=
        c.ths.set! i
          { term := th.term, env := th.env, σs := th.σs, stack := th.stack, vs := th.vs,
            promises := th.promises ++ [Array.size c.mem], nalloc := th.nalloc, undefs := th.undefs,
            tidLbl := th.tidLbl, permit := th.permit },
      lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace }
