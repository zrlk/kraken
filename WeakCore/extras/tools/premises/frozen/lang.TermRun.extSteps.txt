CoreLang.Ty : Type
CoreTerm.At : Type
CoreTerm.Expr : Type
CoreTerm.ExtName : Type
TermRun.Cfg : Type
TermRun.Cfg.ths : TermRun.Cfg → Array TermRun.TTh
TermRun.ExtNeedsNoProm : CoreTerm.ExtName → TermRun.TTh → Prop
TermRun.TTh : Type
TermRun.TTh.env : TermRun.TTh → TermRun.Env
TermRun.TTh.term : TermRun.TTh → CoreTerm.Expr
TermRun.Word : Type
TermRun.evalAt : TermRun.Env → CoreTerm.At → Option (CoreLang.Ty × List TermRun.Word)
TermRun.extCfg : TermRun.Cfg → Nat → TermRun.TTh → String → CoreTerm.ExtName → CoreTerm.Expr → List TermRun.Word → TermRun.Cfg
TermRun.extSteps : TermRun.Cfg → Nat → List TermRun.Cfg
  := fun c i =>
  match c.ths[i]? with
  | some th =>
    match th.term with
    | CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e =>
      if TermRun.ExtNeedsNoProm n th then []
      else
        match TermRun.evalAt th.env a with
        | some (fst, ws) => [TermRun.extCfg c i th x n e ws]
        | none => []
    | x => []
  | none => []
TermRun.instDecidableExtNeedsNoProm : (n : CoreTerm.ExtName) → (th : TermRun.TTh) → Decidable (TermRun.ExtNeedsNoProm n th)
