CoreTerm.ExtName.events : CoreTerm.ExtName → List TermRun.Word → List TermRun.Event
  := fun x x_1 =>
  match x, x_1 with
  | CoreTerm.ExtName.emit, [w] =>
    match w.val with
    | WeakMem.Val.num b => [TermRun.Event.out b]
    | x => []
  | CoreTerm.ExtName.exitEv, [w] =>
    match w.val with
    | WeakMem.Val.num v => [TermRun.Event.exit (TermRun.exitCode v)]
    | x => []
  | x, x_2 => []
