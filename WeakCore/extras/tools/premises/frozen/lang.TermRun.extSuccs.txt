TermRun.extSuccs : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Bounds → TermRun.Cfg → Nat → List TermRun.Cfg
  := fun p k arch b c i =>
  match c.ths[i]? with
  | some th =>
    match th.term with
    | CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e =>
      if (decide (TermRun.ExtNeedsNoProm n th) && !th.promises.isEmpty || !decide (TermRun.ExtEnabled n th)) = true then
        []
      else
        match TermRun.evalAt th.env a with
        | some (fst, ws) =>
          List.map (fun rs => TermRun.wake (TermRun.extCfg c i th x n e ws rs) i (TermRun.relView th.vs) n ws)
            (List.filter (fun rs => TermRun.certB p k arch b.solo (TermRun.extCfg c i th x n e ws rs) i)
              (TermRun.extRs b n i ws c.trace))
        | none => []
    | x => []
  | none => []
