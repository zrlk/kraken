TermLiveCert.Cert.nodeOK : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → List Nat → List Nat → TermLiveCert.Cert → Nat → Bool
  := fun p k arch X V ct v =>
  have c := ct.nd v;
  (TermLiveCert.pfB c && decide (ct.rk v < ct.just.size) &&
        (List.range c.ths.size).all fun i =>
          decide ((TermLiveCert.movesPF p k arch V c i).length = (ct.tg v i).length) &&
            (TermLiveCert.Cert.mt p k arch V ct v i).all fun x =>
              match x with
              | (e, w) =>
                decide (w < ct.nodes.size) && decide (TermRun.cfgKey e = TermRun.cfgKey (ct.nodes[w]?.getD default)) &&
                    (!(TermLiveCert.Cert.quiet p k arch X V ct v && TermLiveCert.Cert.quiet p k arch X V ct w) ||
                      decide (ct.rk w ≤ ct.rk v) &&
                        (!decide (ct.rk w = ct.rk v) ||
                          decide (ct.js v ≠ TermLiveCert.Just.noSpin) &&
                            decide (ct.js v ≠ TermLiveCert.Just.unfair i))) &&
                  (!TermLiveCert.Cert.quiet p k arch X V ct v ||
                    match ct.js v with
                    | TermLiveCert.Just.noSpin => true
                    | TermLiveCert.Just.unfair i => !X.contains i && !(TermLiveCert.movesPF p k arch V c i).isEmpty)) &&
      decide ((TermLiveCert.envsPF p k arch c).length = (ct.te v).length) &&
    (TermLiveCert.Cert.me p k arch ct v).all fun x =>
      match x with
      | (e, w) =>
        decide (w < ct.nodes.size) && decide (TermRun.cfgKey e = TermRun.cfgKey (ct.nodes[w]?.getD default)) &&
            (!(TermLiveCert.Cert.quiet p k arch X V ct v && TermLiveCert.Cert.quiet p k arch X V ct w) ||
              decide (ct.rk w ≤ ct.rk v)) &&
          (!TermLiveCert.Cert.quiet p k arch X V ct v ||
            match ct.js v with
            | TermLiveCert.Just.noSpin => true
            | TermLiveCert.Just.unfair i => !X.contains i && !(TermLiveCert.movesPF p k arch V c i).isEmpty)

TermLiveCert.FairS : (sys : TermLiveMach.LSys) → TermLiveMach.LRun sys.L → (Nat → Option Nat) → Prop
  | TermLiveCert.FairS.mk : ∀ {sys : TermLiveMach.LSys} {r : TermLiveMach.LRun sys.L} {D : Nat → Option Nat},
  (∀ (i : Nat),
      TermLiveCert.LiveS r D i →
        (TermLiveMach.Ev fun n => ∃ c', sys.S (r.cfg n) (sys.thr i) c') →
          TermLiveMach.Often fun n => r.lbl n = some (sys.thr i)) →
    (∀ (i : Nat),
        TermLiveCert.LiveS r D i →
          (TermLiveMach.Often fun n => ∃ c', sys.S (r.cfg n) (sys.thr i) c' ∧ TermLiveMach.Progress (r.cfg n) c' i) →
            TermLiveMach.Often fun n =>
              r.lbl n = some (sys.thr i) ∧ TermLiveMach.Progress (r.cfg n) (r.cfg (n + 1)) i) →
      TermLiveCert.FairS sys r D

TermLiveCert.PromiseFreeRun : {p : CoreTerm.Prog} →
  {k : Promises.Knobs} → {arch : WeakMem.Arch} → TermLiveMach.LRun (TermLiveL3.machineE p k arch).L → Prop
  := fun {p} {k} {arch} r => ∀ (n i : Nat), r.lbl n ≠ some (i, TermLiveL3.Lbl.prom)
TermLiveCert.ScriptedRun : {p : CoreTerm.Prog} →
  {k : Promises.Knobs} → {arch : WeakMem.Arch} → List Nat → TermLiveMach.LRun (TermLiveL3.machineE p k arch).L → Prop
  := fun {p} {k} {arch} V r =>
  ∀ (n i : Nat) (th : TermRun.TTh) (x : String) (w : CoreLang.Wd) (a : CoreTerm.At) (e : CoreTerm.Expr),
    r.lbl n = some (i, TermLiveL3.Lbl.thr) →
      (r.cfg n).ths[i]? = some th →
        th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext (CoreTerm.ExtName.devChoice w) a) e →
          ∃ v,
            v ∈ V ∧
              ((r.cfg (n + 1)).ths[i]?.bind fun x => List.head? x.env) =
                some (x, (CoreTerm.ExtName.devChoice w).sig.snd, [TermRun.cst (WeakMem.Val.num v)])
TermLiveCert.canon : TermRun.Cfg → TermRun.Cfg
  := fun c =>
  { mem := c.mem, pairs := c.pairs,
    ths :=
      Array.map
        (fun th =>
          { term := th.term, env := th.env, σs := th.σs, stack := th.stack,
            vs :=
              have __src := th.vs;
              { coh := fun b => if b ∈ TermLiveCert.locs c then th.vs.coh b else 0, vRd := __src.vRd, vWr := __src.vWr,
                vCtrl := __src.vCtrl, vPreBase := __src.vPreBase, vRelSt := __src.vRelSt, mine := __src.mine,
                fwd := __src.fwd },
            promises := th.promises, nalloc := th.nalloc, undefs := th.undefs, tidLbl := th.tidLbl, permit := th.permit,
            tys := th.tys })
        c.ths,
    lbase := c.lbase, boxFence := c.boxFence, cap := c.cap, trace := c.trace }

      if h : a = b then h ▸ if h : a_1 = b_1 then h ▸ isTrue ⋯ else isFalse ⋯ else isFalse ⋯

      if h : a = b then h ▸ if h : a_1 = b_1 then h ▸ isTrue ⋯ else isFalse ⋯ else isFalse ⋯
  | isFalse h => isFalse ⋯

TermLiveCert.locs : TermRun.Cfg → List WeakMem.Address

TermLiveCert.movesPF : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → List Nat → TermRun.Cfg → Nat → List TermRun.Cfg
  := fun p k arch V c i =>
  TermRun.threadSteps p k arch c i ++ TermRun.extSuccs p k arch { solo := 0, io := V } c i ++
    TermSpawnSuccs.spawnSuccsAt c i
TermLiveCert.noStuckB : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → List Nat → TermLiveCert.Cert → Bool
  := fun p k arch V ct =>
  (List.range ct.nodes.size).all fun v =>
    (ct.nd v).final ||
      (List.range (ct.nd v).ths.size).any fun i => !(TermLiveCert.movesPF p k arch V (ct.nd v) i).isEmpty
TermLiveCert.pfB : TermRun.Cfg → Bool
  := fun c => c.ths.all fun th => th.promises.isEmpty
TermLiveCert.progB : TermRun.Cfg → TermRun.Cfg → Nat → Bool
  := fun c c' i =>
  decide (Array.size c.mem < Array.size c'.mem) ||
    (TermLiveCert.locs c').any fun b => decide (TermLiveMach.cohOf c i b < TermLiveMach.cohOf c' i b)
TermLiveCert.wakesPF : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → List TermRun.Cfg
  := fun p k arch c i => TermRun.wakeSuccs p k arch { solo := 0 } c i

TermLiveL3.SpuriousWake : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun p k arch c i c' =>
  ∃ x a e th rt ws,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext CoreTerm.ExtName.park a) e ∧
        th.permit = none ∧
          TermRun.evalAt th.env a = some (rt, ws) ∧
            TermRun.Certified p k arch (TermRun.extCfg c i th x CoreTerm.ExtName.park e ws []) i ∧
              c' = TermRun.extCfg c i th x CoreTerm.ExtName.park e ws []
TermLiveL3.machineE : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermLiveMach.LSys
  := fun p k arch =>
  { L := Nat × TermLiveL3.Lbl, S := TermLiveL3.mstepE p k arch, actor := fun l => some l.fst,
    thr := fun i => (i, TermLiveL3.Lbl.thr) }
TermLiveL3.mstepE : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat × TermLiveL3.Lbl → TermRun.Cfg → Prop
  := fun p k arch c x x_1 =>
  match x, x_1 with
  | (i, TermLiveL3.Lbl.thr), c' =>
    c' ∈ TermRun.threadSteps p k arch c i ∧ TermRun.Certified p k arch c' i ∨
      TermProgress.SpawnBy c.cap c i c' ∨ TermProgress.ExtBy p k arch c i c'
  | (i, TermLiveL3.Lbl.prom), c' =>
    ∃ a v w,
      i < c.ths.size ∧
        w < Array.size c.mem ∧ TermRun.Certified p k arch (TermRun.promise c i a v w) i ∧ c' = TermRun.promise c i a v w
  | (i, TermLiveL3.Lbl.env), c' => TermLiveL3.SpuriousWake p k arch c i c'

TermProgress.ExtBy : CoreTerm.Prog → Promises.Knobs → WeakMem.Arch → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun p k arch c i c' =>
  ∃ x n a e th rt ws rs,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.ext n a) e ∧
        (TermRun.ExtNeedsNoProm n th → th.promises = []) ∧
          TermRun.ExtEnabled n th ∧
            TermRun.evalAt th.env a = some (rt, ws) ∧
              n.Admits i ws c.trace rs ∧
                TermRun.Certified p k arch (TermRun.extCfg c i th x n e ws rs) i ∧
                  c' = TermRun.wake (TermRun.extCfg c i th x n e ws rs) i (TermRun.relView th.vs) n ws
TermProgress.SpawnBy : Nat → TermRun.Cfg → Nat → TermRun.Cfg → Prop
  := fun cap c i c' =>
  ∃ x f a T ts e th rt ws,
    c.ths[i]? = some th ∧
      th.term = CoreTerm.Expr.let_ x (CoreTerm.Op.spawn f a T ts) e ∧
        th.promises = [] ∧
          c.ths.size < cap ∧ TermRun.evalAt th.env a = some (rt, ws) ∧ c' = TermSpawnStep.spawnCfg c i th x e f ts T ws

TermSpawn.bumpW : WeakMem.Ts → TermRun.Word → TermRun.Word
  := fun V w => { val := w.val, view := max w.view V, rts := max w.rts V }
TermSpawnStep.spawnCfg : TermRun.Cfg →
  Nat → TermRun.TTh → String → CoreTerm.Expr → String → List CoreLang.Ty → CoreLang.Ty → List TermRun.Word → TermRun.Cfg
  := fun c i th x e f ts T ws =>
  TermSpawnStep.spawnT c i (TermSpawnStep.spawnPar c i th x e) (TermSpawnStep.spawnKid c i th f ts T ws)
TermSpawnStep.spawnKid : TermRun.Cfg → Nat → TermRun.TTh → String → List CoreLang.Ty → CoreLang.Ty → List TermRun.Word → TermRun.TTh
  := fun c i th f ts T ws =>
  { term := CoreTerm.Expr.tail f [] (CoreTerm.At.var "a") ts,
    env := [("a", T, List.map (TermSpawn.bumpW (TermSpawnStep.spawnS c th).vs.vRelSt) ws)],
    vs :=
      have __src := (TermSpawnStep.spawnS c th).vs;
      { coh := __src.coh, vRd := __src.vRd, vWr := __src.vWr, vCtrl := __src.vCtrl, vPreBase := __src.vPreBase,
        vRelSt := __src.vRelSt, fwd := __src.fwd },
    tidLbl := some (TermSpawnStep.tidLabel c i th) }
TermSpawnStep.spawnPar : TermRun.Cfg → Nat → TermRun.TTh → String → CoreTerm.Expr → TermRun.TTh
  := fun c i th x e =>
  have __src := TermRun.withS th (TermSpawnStep.spawnS c th);
  { term := e,
    env :=
      (x, CoreLang.Ty.tid, [TermRun.cst (WeakMem.Val.loc { label := TermSpawnStep.tidLabel c i th, offset := 0 } 0)]) ::
        th.env,
    σs := __src.σs, stack := __src.stack, vs := __src.vs, promises := __src.promises,
    nalloc := th.nalloc + if th.tidLbl = none then 2 else 1, undefs := __src.undefs,
    tidLbl := th.tidLbl.orElse fun x => some (TermRun.freshLabel c.lbase c.n i (th.nalloc + 1)), permit := __src.permit,
    tys := __src.tys }
TermSpawnStep.spawnS : TermRun.Cfg → TermRun.TTh → TermRun.St
  := fun c th => TermRun.fenceS (TermRun.stOf c.mem c.pairs th) WeakMem.Fence.sy
TermSpawnStep.spawnT : TermRun.Cfg → Nat → TermRun.TTh → TermRun.TTh → TermRun.Cfg
  := fun c i par kid =>
  { mem := c.mem, pairs := c.pairs, ths := (c.ths.set! i par).push kid, lbase := c.lbase, boxFence := c.boxFence,
    cap := c.cap, trace := c.trace }
TermSpawnStep.tidLabel : TermRun.Cfg → Nat → TermRun.TTh → WeakMem.AllocLabel
  := fun c i th => TermRun.freshLabel c.lbase c.n i th.nalloc
TermSpawnSuccs.spawnSuccsAt : TermRun.Cfg → Nat → List TermRun.Cfg
  := fun c i =>
  match c.ths[i]? with
  | some th =>
    match th.term with
    | CoreTerm.Expr.let_ x (CoreTerm.Op.spawn f a T ts) e =>
      if (th.promises.isEmpty && decide (c.ths.size < c.cap)) = true then
        match TermRun.evalAt th.env a with
        | some (fst, ws) => [TermSpawnStep.spawnCfg c i th x e f ts T ws]
        | none => []
      else []
    | x => []
  | none => []
