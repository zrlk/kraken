TermScope.AllDrf : List CoreLang.Ty → Prop
  := fun σs => ∀ (σ : CoreLang.Ty), σ ∈ σs → TermScope.drf σ = true
TermScope.EnvDRF : TermRun.Env → CoreTerm.Expr → Prop
  := fun env t =>
  ∀ (x : String),
    TermScope.freeB x t = true →
      ∀ (rt : CoreLang.Ty) (ws : List TermRun.Word), TermRun.lookupE env x = some (rt, ws) → TermScope.drf rt = true
TermScope.FrDRF : TermRun.Frame → Prop
  | TermScope.FrDRF.mk : ∀ {f : TermRun.Frame},
  TermScope.allOK f.body = true →
    TermScope.AllDrf f.σs →
      TermScope.drf f.ty = true →
        (∀ (x : String),
            TermScope.freeB x f.body = true →
              x ≠ f.x →
                ∀ (rt : CoreLang.Ty) (ws : List TermRun.Word),
                  TermRun.lookupE f.env x = some (rt, ws) → TermScope.drf rt = true) →
          TermScope.FrDRF f
TermScope.ProgScopeOK : CoreTerm.Prog → Bool
  := fun p => List.all p fun d => TermScope.drf d.argTy && TermScope.drf d.retTy && TermScope.allOK d.body

  := fun p => List.all p (TermScope.declWrapOK p)

TermScope.allOK : CoreTerm.Expr → Bool
  eq_def ∀ (x : CoreTerm.Expr),
  TermScope.allOK x =
    match x with
    | CoreTerm.Expr.let_ x op e => TermScope.opOK op && TermScope.allOK e
    | CoreTerm.Expr.split a x y e => TermScope.allOK e
    | CoreTerm.Expr.unpack a i x e => TermScope.allOK e
    | CoreTerm.Expr.unpackT a α x e => TermScope.allOK e
    | CoreTerm.Expr.region a r p t e => !TermScope.freeB p e && TermScope.allOK e
    | CoreTerm.Expr.openPub a r p c e => !TermScope.freeB p e && TermScope.allOK e
    | CoreTerm.Expr.case a x e₁ y e₂ => TermScope.allOK e₁ && TermScope.allOK e₂
    | CoreTerm.Expr.ifnull a e₁ x e₂ => TermScope.allOK e₁ && TermScope.allOK e₂
    | CoreTerm.Expr.parity a k e₁ e₂ => TermScope.allOK e₁ && TermScope.allOK e₂
    | CoreTerm.Expr.ifeq a b e₁ e₂ => TermScope.allOK e₁ && TermScope.allOK e₂
    | CoreTerm.Expr.cas acq rel w e v t ok fail => TermScope.allOK ok && TermScope.allOK fail
    | CoreTerm.Expr.casp acq rel w e0 e1 v0 v1 t ok fail => TermScope.allOK ok && TermScope.allOK fail
    | CoreTerm.Expr.ret a => true
    | CoreTerm.Expr.tail f is a ts => ts.all TermScope.drf
TermScope.argOK : CoreLang.Ty → Bool
  := fun t => TermScope.wrapOK t && !TermScope.rcar t
TermScope.checkW : CoreTerm.Prog → CoreTerm.Cx → CoreTerm.Expr → CoreTerm.M Unit
  eq_def ∀ (p : CoreTerm.Prog) (cx : CoreTerm.Cx) (x : CoreTerm.Expr),
  TermScope.checkW p cx x =
    match x with
    | CoreTerm.Expr.let_ x op e =>
      have __do_jp := fun __r => do
        let cx ← CoreTerm.checkOp p cx x op
        have __do_jp : Unit → CoreTerm.M Unit := fun __r => TermScope.checkW p cx e
        match List.find? (fun x_1 => x_1.fst == x) cx.vars with
          | some (fst, t, snd) =>
            if (!TermScope.wrapOK t) = true then do
              let __r ← throw (toString "A1: " ++ toString x ++ toString " has a key-carrying wrapper type")
              __do_jp __r
            else __do_jp ()
          | none => __do_jp ();
      if (!TermScope.opArgOK op) = true then do
        let __r ← throw "A1: a type argument or witness carries a key"
        __do_jp __r
      else __do_jp ()
    | CoreTerm.Expr.split a x y e => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | u.prod v => do
            let cx ← cx.bind x u
            let cx ← cx.bind y v
            TermScope.checkW p cx e
          | x => throw "split needs a pair"
    | CoreTerm.Expr.unpack a i x e => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | b.iex => do
            let cx ← (cx.push i).bind x b
            TermScope.checkW p cx e
          | x => throw "unpack needs an index existential"
    | CoreTerm.Expr.unpackT a α x e => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | CoreLang.Ty.tex κ b =>
            have __do_jp := fun __r => do
              let cx ← (cx.pushT κ).bind x b
              TermScope.checkW p cx e;
            if (!CoreLang.Ty.guardT 0 b) = true then do
              let __r ← throw "unpack: the package's variable must occur only under a reference"
              __do_jp __r
            else __do_jp ()
          | x => throw "unpack at a type needs a type existential"
    | CoreTerm.Expr.region a rn pn tn e => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | CoreLang.Ty.ptr CoreLang.Mode.own u =>
            have __do_jp := fun __r =>
              have cx := cx.push rn;
              do
              let cx ← cx.bind pn (CoreLang.Ty.ptr (CoreLang.Mode.rgn (CoreLang.Idx.var 0)) (CoreLang.Ty.shiftI 0 u))
              let cx ← cx.bind tn (CoreLang.Ty.tok (CoreLang.Idx.var 0) none)
              TermScope.checkW p cx e;
            if (!CoreTerm.rgnCellOK cx.ke u) = true then do
              let __r ←
                throw
                    "region: the cell type is not shareable (rows 16, 17): an own reference inside a region must be moved in with intoRgn"
              __do_jp __r
            else __do_jp ()
          | (CoreLang.Ty.loc CoreLang.Mode.own (CoreLang.Ty.array n u)).refine (CoreLang.Pred.ini k) =>
            have __do_jp := fun __r =>
              have cx := cx.push rn;
              do
              let cx ←
                cx.bind pn
                    ((CoreLang.Ty.loc (CoreLang.Mode.rgn (CoreLang.Idx.var 0))
                          (CoreLang.Ty.shiftI 0 (CoreLang.Ty.array n u))).refine
                      (CoreLang.Pred.shift 0 (CoreLang.Pred.ini k)))
              let cx ← cx.bind tn (CoreLang.Ty.tok (CoreLang.Idx.var 0) none)
              TermScope.checkW p cx e;
            if (!CoreTerm.rgnCellOK cx.ke (CoreLang.Ty.array n u)) = true then do
              let __r ←
                throw
                    "region: the cell type is not shareable (rows 16, 17): an own reference inside a region must be moved in with intoRgn"
              __do_jp __r
            else __do_jp ()
          | x => throw "region needs an own reference or an own buffer"
    | CoreTerm.Expr.openPub a rn pn cn e => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | CoreLang.Ty.ptr CoreLang.Mode.pub u =>
            have cx := cx.push rn;
            do
            let cx ← cx.bind pn (CoreLang.Ty.ptr (CoreLang.Mode.rgn (CoreLang.Idx.var 0)) (CoreLang.Ty.shiftI 0 u))
            let cx ← cx.bind cn (CoreLang.Ty.cap CoreLang.Cap.pub (CoreLang.Idx.var 0))
            TermScope.checkW p cx e
          | x => throw "openPub needs a pub reference"
    | CoreTerm.Expr.case a x e₁ y e₂ => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | u.sum v => do
            let __do_lift ← cx.bind x u
            TermScope.checkW p __do_lift e₁
            let __do_lift ← cx.bind y v
            TermScope.checkW p __do_lift e₂
          | x => throw "case needs a sum"
    | CoreTerm.Expr.ifnull a e₁ y e₂ => do
      let __x ← cx.use a
      match __x with
        | (t, cx) =>
          match t with
          | CoreLang.Ty.ptrOpt μ u => do
            TermScope.checkW p cx e₁
            let __do_lift ← cx.bind y (CoreLang.Ty.ptr μ u)
            TermScope.checkW p __do_lift e₂
          | x => throw "ifnull needs a nullable reference"
    | CoreTerm.Expr.parity a k ev od => do
      let i ← CoreTerm.intIdx cx a
      have __do_jp : Nat → CoreTerm.M Unit := fun v =>
        have cx := cx.push k;
        do
        TermScope.checkW p
            { inames := cx.inames, facts := (v + 1, CoreLang.Idx.smul 2 (CoreLang.Idx.var 0)) :: cx.facts,
              vars := cx.vars, saws := cx.saws, torn := cx.torn, unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq,
              lent := cx.lent, ret := cx.ret }
            ev
        TermScope.checkW p
            { inames := cx.inames,
              facts := (v + 1, (CoreLang.Idx.smul 2 (CoreLang.Idx.var 0)).add (CoreLang.Idx.lit 1)) :: cx.facts,
              vars := cx.vars, saws := cx.saws, torn := cx.torn, unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq,
              lent := cx.lent, ret := cx.ret }
            od
      match (cx.norm i).bind CoreTerm.Lin.asVar? with
        | some v => do
          let v ← pure v
          __do_jp v
        | none => do
          let v ← throw "parity: the index is not a free variable, so there is nothing to refine"
          __do_jp v
    | CoreTerm.Expr.ifeq a b yes no =>
      if CoreTerm.plainEq cx a b = true then do
        TermScope.checkW p cx yes
        TermScope.checkW p cx no
      else do
        let i ← CoreTerm.intIdx cx a
        let j ← CoreTerm.intIdx cx b
        match cx.norm i, cx.norm j with
          | some ni, some nj =>
            have __do_jp := fun __r => TermScope.checkW p (CoreTerm.noCx cx ni nj) no;
            if (ni == nj) = true then do
              let __r ← TermScope.checkW p cx yes
              __do_jp __r
            else
              match ni.asVar?, nj.asVar? with
              | some v, x =>
                have __do_jp := fun __r => do
                  let __r ←
                    TermScope.checkW p
                        { inames := cx.inames, facts := (v, nj.toIdx) :: cx.facts, vars := cx.vars, saws := cx.saws,
                          torn := cx.torn, unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq, lent := cx.lent,
                          ret := cx.ret }
                        yes
                  __do_jp __r;
                if nj.mentions v = true then do
                  let __r ← throw "ifeq: cannot record the equality"
                  __do_jp __r
                else __do_jp ()
              | x, some v =>
                have __do_jp := fun __r => do
                  let __r ←
                    TermScope.checkW p
                        { inames := cx.inames, facts := (v, ni.toIdx) :: cx.facts, vars := cx.vars, saws := cx.saws,
                          torn := cx.torn, unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq, lent := cx.lent,
                          ret := cx.ret }
                        yes
                  __do_jp __r;
                if ni.mentions v = true then do
                  let __r ← throw "ifeq: cannot record the equality"
                  __do_jp __r
                else __do_jp ()
              | x, x_1 =>
                match CoreTerm.zeroVar ni nj with
                | some u => do
                  let __r ←
                    TermScope.checkW p
                        { inames := cx.inames, facts := (u, CoreLang.Idx.lit 0) :: cx.facts, vars := cx.vars,
                          saws := cx.saws, torn := cx.torn, unsealed := cx.unsealed, ke := cx.ke, opq := cx.opq,
                          lent := cx.lent, ret := cx.ret }
                        yes
                  __do_jp __r
                | none => do
                  let __r ← throw "ifeq: cannot record the equality"
                  __do_jp __r
          | x, x_1 => throw "ifeq: index outside the decidable fragment"
    | CoreTerm.Expr.cas acq rel w e v t ok fail => do
      let tw ← cx.peek w
      match CoreTerm.wordOf tw with
        | none => do
          let u ← CoreTerm.casCellTy cx tw
          have __do_jp : Unit → CoreTerm.M Unit := fun __r =>
            have __do_jp := fun __r => do
              let te ← cx.peek e
              let __x ← cx.use v
              match __x with
                | (tv, cx) =>
                  have __do_jp := fun __r =>
                    have __do_jp := fun __r =>
                      have __do_jp := fun __r => do
                        let __do_lift ← cx.bind t CoreLang.Ty.unit
                        TermScope.checkW p __do_lift ok
                        TermScope.checkW p cx fail;
                      if (CoreLang.Ty.sensitiveIn cx.ke u && !(acq && rel)) = true then do
                        let __r ← throw "cas: a view-sensitive type needs acq and rel (§3.8)"
                        __do_jp __r
                      else __do_jp ();
                    if (!CoreTerm.sub cx tv u) = true then do
                      let __r ← throw "cas: the new value does not have the cell's type"
                      __do_jp __r
                    else __do_jp ();
                  if (!CoreTerm.sub cx te u) = true then do
                    let __r ← throw "cas: the expected value does not have the cell's type"
                    __do_jp __r
                  else __do_jp ();
            if CoreTerm.affine cx.ke u = true then do
              let __r ← throw "cas would duplicate an affine value"
              __do_jp __r
            else __do_jp ()
          if (u.rawWords != some 1) = true then do
              let __r ← throw "cas needs a one-word cell (§3.5)"
              __do_jp __r
            else __do_jp ()
        | some (r, P) =>
          if P.epoch = true then
            have __do_jp := fun __r => do
              let i ← CoreTerm.intIdx cx e
              let j ← CoreTerm.intIdx cx v
              let __x ← CoreTerm.classifyM cx P i
              match __x with
                | (c, k) => do
                  let __x ← CoreTerm.classifyM cx P j
                  match __x with
                    | (c', k') =>
                      have __do_jp := fun __r => do
                        let __do_lift ← cx.bind t (CoreLang.Ty.tokE r (CoreLang.ECounter.word P) j)
                        TermScope.checkW p __do_lift ok
                        TermScope.checkW p cx fail;
                      if (!CoreTerm.pstep P c k c' k') = true then do
                        let __r ← throw "not a transition of the protocol"
                        __do_jp __r
                      else __do_jp ();
            if (!acq) = true then do
              let __r ← throw "entering an epoch needs acq"
              __do_jp __r
            else __do_jp ()
          else
            have __do_jp := fun __r => do
              let i ← CoreTerm.intIdx cx e
              let j ← CoreTerm.intIdx cx v
              let __x ← CoreTerm.classifyM cx P i
              match __x with
                | (c, k) => do
                  let __x ← CoreTerm.classifyM cx P j
                  match __x with
                    | (c', k') => do
                      let cl ← CoreTerm.clauseM P c
                      let cl' ← CoreTerm.clauseM P c'
                      have __do_jp : Unit → CoreTerm.M Unit := fun __r =>
                        have __do_jp := fun __r =>
                          have __do_jp := fun __r => TermScope.checkW p cx fail;
                          if cl'.holds = true then do
                            let __do_lift ← CoreTerm.lendCx cx acq rel r (CoreTerm.shareMove P c k c' k') t
                            let __r ← TermScope.checkW p __do_lift ok
                            __do_jp __r
                          else
                            have __do_jp := fun __r =>
                              have __do_jp := fun __r => do
                                let cx' ← cx.bind t (CoreLang.Ty.tok r (some (P, j)))
                                have cx' : CoreTerm.Cx :=
                                  if CoreTerm.monotone P = true then
                                    { inames := cx'.inames, facts := cx'.facts, vars := cx'.vars, saws := cx'.saws,
                                      torn := cx'.torn, unsealed := r :: cx'.unsealed, ke := cx'.ke, opq := cx'.opq,
                                      lent := cx'.lent, ret := cx'.ret }
                                  else cx'
                                let __r ← TermScope.checkW p cx' ok
                                __do_jp __r;
                              if (!acq) = true then do
                                let __r ← throw "taking the region needs acq"
                                __do_jp __r
                              else __do_jp ();
                            if (P.lends.contains c || P.lendsDown.contains c) = true then do
                              let __r ← throw "taking the region from a lending state: shares are out"
                              __do_jp __r
                            else __do_jp ();
                        if (!CoreTerm.pstep P c k c' k') = true then do
                          let __r ← throw "not a transition of the protocol"
                          __do_jp __r
                        else __do_jp ()
                      if (!cl.holds) = true then do
                          let __r ← throw "cas from a held state: the holder stores instead"
                          __do_jp __r
                        else __do_jp ();
            if P.ghost = true then do
              let __r ← throw "cas on a ghost word: it has no runtime value"
              __do_jp __r
            else __do_jp ()
    | CoreTerm.Expr.casp acq rel w e0 e1 v0 v1 t ok fail => do
      let tw ← cx.peek w
      match CoreTerm.word2Of tw with
        | none =>
          match CoreTerm.epairOf tw with
          | none => throw "casp needs a reference to a protocol pair"
          | some (r, P) =>
            have __do_jp := fun __r => do
              let _ ← CoreTerm.intIdx cx e0
              let _ ← CoreTerm.intIdx cx e1
              let j0 ← CoreTerm.intIdx cx v0
              let _ ← CoreTerm.intIdx cx v1
              have __do_jp : Unit → CoreTerm.M Unit := fun __r => do
                let __do_lift ← cx.bind t CoreLang.Ty.unit
                TermScope.checkW p __do_lift ok
                TermScope.checkW p cx fail
              if (!cx.hasTokE r P j0) = true then do
                  let __r ← throw "an epoch-fenced write needs the epoch token of its new epoch"
                  __do_jp __r
                else __do_jp ();
            if (!rel) = true then do
              let __r ← throw "an epoch-fenced write needs rel"
              __do_jp __r
            else __do_jp ()
        | some (r, P) => do
          let i0 ← CoreTerm.intIdx cx e0
          let i1 ← CoreTerm.intIdx cx e1
          let j0 ← CoreTerm.intIdx cx v0
          let j1 ← CoreTerm.intIdx cx v1
          let __x ← CoreTerm.classify2M cx P i0 i1
          match __x with
            | (c, k0, k1) => do
              let __x ← CoreTerm.classify2M cx P j0 j1
              match __x with
                | (c', k0', k1') => do
                  let cl ← CoreTerm.clause2M P c
                  let cl' ← CoreTerm.clause2M P c'
                  have __do_jp : Unit → CoreTerm.M Unit := fun __r =>
                    have __do_jp := fun __r =>
                      have __do_jp := fun __r => TermScope.checkW p cx fail;
                      if cl'.holds = true then
                        if CoreTerm.bump2 P k0 k0' = true then
                          have __do_jp := fun __r => do
                            let __do_lift ← cx.bind t (CoreLang.Ty.tokE r (CoreLang.ECounter.pair P) j0)
                            let __r ← TermScope.checkW p __do_lift ok
                            __do_jp __r;
                          if (!acq) = true then do
                            let __r ← throw "entering an epoch needs acq"
                            __do_jp __r
                          else __do_jp ()
                        else do
                          let __do_lift ← cx.bind t CoreLang.Ty.unit
                          let __r ← TermScope.checkW p __do_lift ok
                          __do_jp __r
                      else
                        have __do_jp := fun __r => do
                          let __do_lift ← cx.bind t (CoreLang.Ty.tok2 r P j0 j1)
                          let __r ← TermScope.checkW p __do_lift ok
                          __do_jp __r;
                        if (!acq) = true then do
                          let __r ← throw "taking the region needs acq"
                          __do_jp __r
                        else __do_jp ();
                    if (!CoreTerm.pstep2 P c k0 k1 c' k0' k1') = true then do
                      let __r ← throw "not a transition of the pair protocol"
                      __do_jp __r
                    else __do_jp ()
                  if (!cl.holds) = true then do
                      let __r ← throw "casp from a held state: the holder stores instead"
                      __do_jp __r
                    else __do_jp ()
    | CoreTerm.Expr.ret a => do
      let __x ← cx.use a
      match __x with
        | (t, cx) => do
          CoreTerm.noLeak cx t
          CoreTerm.noDrop cx
          if (!CoreTerm.sub cx t cx.ret) = true then throw "ret: the value does not have the return type" else pure ()
    | CoreTerm.Expr.tail f is a ts =>
      have __do_jp := fun __r => do
        let d ← p.find f
        let __x ← CoreTerm.instantiate cx d is ts
        match __x with
          | (at_, rt) => do
            let __x ← cx.use a
            match __x with
              | (t, cx) => do
                CoreTerm.noLeak cx t
                CoreTerm.noDrop cx
                have __do_jp : Unit → CoreTerm.M Unit := fun __r =>
                  if (!CoreTerm.sub cx rt cx.ret) = true then
                    throw (toString "tail call " ++ toString f ++ toString ": return type mismatch")
                  else pure ()
                if (!CoreTerm.sub cx t at_) = true then do
                    let __r ← throw (toString "tail call " ++ toString f ++ toString ": argument type mismatch")
                    __do_jp __r
                  else __do_jp ();
      if (!ts.all TermScope.argOK) = true then do
        let __r ← throw "A1: a tail type argument carries a key"
        __do_jp __r
      else __do_jp ()
TermScope.declWrapOK : CoreTerm.Prog → CoreTerm.Decl → Bool
  := fun p d =>
  TermScope.wrapOK d.argTy && TermScope.wrapOK d.retTy &&
    match
      TermScope.checkW p
        { inames := d.idx.reverse, vars := [(d.arg, d.argTy, true)], ke := List.map some d.tparams.reverse,
          ret := d.retTy }
        d.body with
    | Except.ok a => true
    | x => false
TermScope.drf : CoreLang.Ty → Bool
  eq_def ∀ (x : CoreLang.Ty),
  TermScope.drf x =
    match x with
    | a.prod b => TermScope.drf a && TermScope.drf b
    | a.sum b => TermScope.drf a && TermScope.drf b
    | a.code b k => TermScope.drf a && TermScope.drf b
    | CoreLang.Ty.array i t => TermScope.drf t
    | CoreLang.Ty.ptr μ t => TermScope.drfM μ && TermScope.drf t
    | CoreLang.Ty.ptrOpt μ t => TermScope.drfM μ && TermScope.drf t
    | CoreLang.Ty.loc μ t => TermScope.drfM μ && TermScope.drf t
    | t.mu => TermScope.drf t
    | CoreLang.Ty.tex κ t => TermScope.drf t
    | τ.iall => false
    | t.iex => TermScope.drf t
    | t.refine ψ => TermScope.drf t
    | x => true
TermScope.drfM : CoreLang.Mode → Bool
  := fun x =>
  match x with
  | CoreLang.Mode.rgn r => false
  | x => true
TermScope.freeA : String → CoreTerm.At → Bool
  := fun x x_1 =>
  match x_1 with
  | CoreTerm.At.var y => x == y
  | x => false
TermScope.freeB : String → CoreTerm.Expr → Bool
  eq_def ∀ (x : String) (x_1 : CoreTerm.Expr),
  TermScope.freeB x x_1 =
    match x_1 with
    | CoreTerm.Expr.let_ y op e => (TermScope.opAts op).any (TermScope.freeA x) || x != y && TermScope.freeB x e
    | CoreTerm.Expr.split a y z e => TermScope.freeA x a || x != y && x != z && TermScope.freeB x e
    | CoreTerm.Expr.unpack a i y e => TermScope.freeA x a || x != y && TermScope.freeB x e
    | CoreTerm.Expr.unpackT a α y e => TermScope.freeA x a || x != y && TermScope.freeB x e
    | CoreTerm.Expr.region a r p t e => TermScope.freeA x a || x != p && x != t && TermScope.freeB x e
    | CoreTerm.Expr.openPub a r p c e => TermScope.freeA x a || x != p && x != c && TermScope.freeB x e
    | CoreTerm.Expr.case a y e₁ z e₂ =>
      TermScope.freeA x a || x != y && TermScope.freeB x e₁ || x != z && TermScope.freeB x e₂
    | CoreTerm.Expr.ifnull a e₁ y e₂ => TermScope.freeA x a || TermScope.freeB x e₁ || x != y && TermScope.freeB x e₂
    | CoreTerm.Expr.parity a k e₁ e₂ => TermScope.freeA x a || TermScope.freeB x e₁ || TermScope.freeB x e₂
    | CoreTerm.Expr.ifeq a b e₁ e₂ =>
      TermScope.freeA x a || TermScope.freeA x b || TermScope.freeB x e₁ || TermScope.freeB x e₂
    | CoreTerm.Expr.cas acq rel w e v t ok fail =>
      TermScope.freeA x w || TermScope.freeA x e || TermScope.freeA x v || x != t && TermScope.freeB x ok ||
        TermScope.freeB x fail
    | CoreTerm.Expr.casp acq rel w e0 e1 v0 v1 t ok fail =>
      TermScope.freeA x w || TermScope.freeA x e0 || TermScope.freeA x e1 || TermScope.freeA x v0 ||
            TermScope.freeA x v1 ||
          x != t && TermScope.freeB x ok ||
        TermScope.freeB x fail
    | CoreTerm.Expr.ret a => TermScope.freeA x a
    | CoreTerm.Expr.tail f is a ts => TermScope.freeA x a
TermScope.opArgOK : CoreTerm.Op → Bool
  := fun x =>
  match x with
  | CoreTerm.Op.call f is a ts => ts.all TermScope.argOK
  | CoreTerm.Op.spawn f a T ts => TermScope.wrapOK T && ts.all TermScope.argOK
  | CoreTerm.Op.packT a σ τ => TermScope.argOK σ && TermScope.wrapOK τ
  | op => (TermScope.opTys op).all TermScope.wrapOK
TermScope.opAts : CoreTerm.Op → List CoreTerm.At
  := fun x =>
  match x with
  | CoreTerm.Op.atom a => [a]
  | CoreTerm.Op.pair a b => [a, b]
  | CoreTerm.Op.inl other a => [a]
  | CoreTerm.Op.inr other a => [a]
  | CoreTerm.Op.add a b => [a, b]
  | CoreTerm.Op.sub a b => [a, b]
  | CoreTerm.Op.new a T => [a]
  | CoreTerm.Op.null μ target => []
  | CoreTerm.Op.fold a target => [a]
  | CoreTerm.Op.publish a => [a]
  | CoreTerm.Op.intoRgn r a => [a]
  | CoreTerm.Op.newBuf n τ => [n]
  | CoreTerm.Op.idx b i => [b, i]
  | CoreTerm.Op.bufInit b i v => [b, i, v]
  | CoreTerm.Op.fld i a => [a]
  | CoreTerm.Op.load acq a => [a]
  | CoreTerm.Op.store rel a v tok => [a, v] ++ tok.toList
  | CoreTerm.Op.fence f => []
  | CoreTerm.Op.mkWord r P v tok => [v, tok]
  | CoreTerm.Op.pack a w target => [a]
  | CoreTerm.Op.packT a σ target => [a]
  | CoreTerm.Op.validate y => [CoreTerm.At.var y]
  | CoreTerm.Op.call f is a ts => [a]
  | CoreTerm.Op.spawn f a T ts => [a]
  | CoreTerm.Op.ext n a => [a]
  | CoreTerm.Op.borrow s t => [t]
  | CoreTerm.Op.endBorrow t => [t]
  | CoreTerm.Op.xchg a v => [a, v]
  | CoreTerm.Op.fnptr f => []
  | CoreTerm.Op.callc f a => [f, a]
  | CoreTerm.Op.rem a n => [a, n]
  | CoreTerm.Op.nameI a => [a]
  | CoreTerm.Op.xchgAt b i v => [b, i, v]
  | CoreTerm.Op.loadAt b i => [b, i]
  | CoreTerm.Op.bits o a b => [a, b]
  | CoreTerm.Op.bit1 o a => [a]
  | CoreTerm.Op.vmatch lo hi b => [lo, hi, b]
  | CoreTerm.Op.band a m => [a, m]
  | CoreTerm.Op.zero τ => []
  | CoreTerm.Op.loadAtF b i p => [b, i]
  | CoreTerm.Op.xchgAtF b i p v => [b, i, v]
  | CoreTerm.Op.freeze t => [t]
  | CoreTerm.Op.cvt sg w a => [a]
  | CoreTerm.Op.toInt a => [a]
  | CoreTerm.Op.asLanes sg w a => [a]
  | CoreTerm.Op.fop o a b => [a, b]
  | CoreTerm.Op.fcmp o a b => [a, b]
  | CoreTerm.Op.fop1 o a => [a]
  | CoreTerm.Op.lane a j => [a, j]
  | CoreTerm.Op.setLane a j v => [a, j, v]
  | CoreTerm.Op.ldl b i => [b, i]
  | CoreTerm.Op.stl b i v => [b, i, v]
  | CoreTerm.Op.ldlw b wi j => [b, wi, j]
  | CoreTerm.Op.stlw b wi j v => [b, wi, j, v]
  | CoreTerm.Op.mkWord2 r P v0 v1 tok => [v0, v1, tok]
  | CoreTerm.Op.ldp acq a n0 n1 => [a]
  | CoreTerm.Op.stp rel a v0 v1 tok => [a, v0, v1, tok]
  | CoreTerm.Op.mkEPair r P v0 v1 => [v0, v1]
  | CoreTerm.Op.mkTicket r N P v tok => [v, tok]
  | CoreTerm.Op.claim a tok => [a, tok]
  | CoreTerm.Op.fetch o acq rel a v => [a, v]
  | CoreTerm.Op.isTy a τ => [a]
TermScope.opOK : CoreTerm.Op → Bool
  := fun x =>
  match x with
  | CoreTerm.Op.validate x => false
  | CoreTerm.Op.intoRgn r a => false
  | CoreTerm.Op.mkTicket r N P v tok => false
  | CoreTerm.Op.claim a tok => false
  | CoreTerm.Op.null μ τ => TermScope.drfM μ && TermScope.drf τ
  | op => (TermScope.opTys op).all TermScope.drf
TermScope.opTys : CoreTerm.Op → List CoreLang.Ty
  := fun x =>
  match x with
  | CoreTerm.Op.inl τ a => [τ]
  | CoreTerm.Op.inr τ a => [τ]
  | CoreTerm.Op.null μ τ => [τ]
  | CoreTerm.Op.fold a τ => [τ]
  | CoreTerm.Op.newBuf n τ => [τ]
  | CoreTerm.Op.pack a w τ => [τ]
  | CoreTerm.Op.zero τ => [τ]
  | CoreTerm.Op.packT a σ τ => [σ, τ]
  | CoreTerm.Op.call f is a ts => ts
  | CoreTerm.Op.spawn f a T ts => T :: ts
  | CoreTerm.Op.isTy a τ => [τ]
  | x => []
TermScope.rcar : CoreLang.Ty → Bool
  := fun t => TermScope.rcarE [] t
TermScope.rcarE : List Bool → CoreLang.Ty → Bool
  eq_def ∀ (env : List Bool) (x : CoreLang.Ty),
  TermScope.rcarE env x =
    match x with
    | CoreLang.Ty.word r P => true
    | CoreLang.Ty.word2 r P => true
    | CoreLang.Ty.tok r st => true
    | CoreLang.Ty.tok2 r P i0 i1 => true
    | CoreLang.Ty.tokE r P e => true
    | CoreLang.Ty.share r => true
    | a.prod b => TermScope.rcarE env a || TermScope.rcarE env b
    | a.sum b => TermScope.rcarE env a || TermScope.rcarE env b
    | CoreLang.Ty.tvar k => env.getD k false
    | t.mu => TermScope.rcarE (TermScope.rcarE (false :: env) t :: env) t
    | CoreLang.Ty.tex κ t => TermScope.rcarE (false :: env) t
    | t.iall => TermScope.rcarE env t
    | t.iex => TermScope.rcarE env t
    | t.refine ψ => TermScope.rcarE env t
    | x => false
TermScope.wrapOK : CoreLang.Ty → Bool
  := fun t => TermScope.wrapOKE [] t
TermScope.wrapOKE : List Bool → CoreLang.Ty → Bool
  eq_def ∀ (env : List Bool) (x : CoreLang.Ty),
  TermScope.wrapOKE env x =
    match x with
    | a.sum b => !TermScope.rcarE env a && !TermScope.rcarE env b && TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | t.iex => !TermScope.rcarE env t && TermScope.wrapOKE env t
    | t.refine ψ => !TermScope.rcarE env t && TermScope.wrapOKE env t
    | CoreLang.Ty.tex κ t => !TermScope.rcarE (false :: env) t && TermScope.wrapOKE (false :: env) t
    | t.mu => TermScope.wrapOKE (TermScope.rcarE (false :: env) t :: env) t
    | a.prod b => TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | a.code b k => TermScope.wrapOKE env a && TermScope.wrapOKE env b
    | CoreLang.Ty.array i t => TermScope.wrapOKE env t
    | CoreLang.Ty.ptr μ t => TermScope.wrapOKE env t
    | CoreLang.Ty.ptrOpt μ t => TermScope.wrapOKE env t
    | CoreLang.Ty.loc μ t => TermScope.wrapOKE env t
    | t.iall => TermScope.wrapOKE env t
    | x => true
