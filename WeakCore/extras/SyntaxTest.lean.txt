/-
# SyntaxTest — every library program survives print → parse

For each library below, every code block `d` is printed with
`Syntax.showDecl` and read back with `Syntax.parseDecl`; the result must be
`d` again, field by field.  So the concrete syntax can express the whole
library, and printed programs are programs.  Whole programs are also printed
and parsed, and `Scope` (the checker-faithful index scope) is checked to
follow the checker through every accepted block.

Build: LEAN_PATH=build lean SyntaxTest.lean   (after the libraries and Syntax.olean)
-/

/-- (E1) `spawn … : T`, `ThreadId` (compared by `if`) and the trusted externs. -/
def e1Src : String := "
def kid(a : ThreadId * U8) : Unit =
  let (j, b) = a
  let u = emit b
  ret u

def same(t : ThreadId) : Int =
  let me = current ()
  if me = t then ret 1 else ret 0

def par(b : U8) : Int =
  let me = current ()
  let p = (me, b)
  let t = spawn kid p : ThreadId * U8
  let u = exitEv 0
  tail same(t)
"

def e1Prog : Prog := match parseProg e1Src with | .ok p => p | .error _ => []

#guard e1Prog.length == 3
#guard checkProg e1Prog matches .ok _

def corpus : List (String × Prog) :=
  [("mutexLib", mutexLib), ("seqLib", seqLib), ("queueLib Int", queueLib .int),
   ("polyQueueLib", polyQueueLib), ("claimLib Int", claimLib .int), ("pcLib", pcLib),
   ("TermSmoke", TermSmoke.seqProg ++ TermSmoke.qProg ++ TermSmoke.cProg ++ TermSmoke.bProg),
   ("BufSmoke", BufSmoke.mpGood ++ BufSmoke.mpPlain ++ BufSmoke.mpBad ++ BufSmoke.zProg),
   ("Typestate", Typestate.fileLib), ("Caps", Caps.capLib),
   ("RWLock", RWLock.rwLib ++ RWLock.bsemLib),
   ("HashMap", HashMap.hmLib ++ HashMap.instLib ++ HashMap.intLib ++ HashMap.rwProg),
   ("HashMapOwn", HashMapOwn.ohmLib ++ HashMapOwn.oInstLib),
   ("HashMapLF", HashMapLF.lfLib ++ HashMapLF.lfProg),
   ("Freeze", Freeze.frzLib ++ Freeze.frzCLib), ("SubWord", SubWord.swLib),
   ("FlatHashMap portable", FlatHashMap.ownLib false), ("FlatHashMap x64", FlatHashMap.ownLib true),
   ("E1", e1Prog)]

/-- The code blocks that do not round-trip, with what went wrong. -/
def failures : List String :=
  corpus.flatMap fun (_, p) => (p.filter (!roundTrips ·)).map roundTripReport

#guard failures.isEmpty

/-- Whole programs round-trip too, where each block is printed and parsed
knowing the blocks before it (so `Scope` follows the checker through calls). -/
def progFailures : List String :=
  corpus.filterMap fun (n, p) =>
    match parseProg (showProg p) with
    | .ok q => if q.length == p.length && (p.zip q).all (fun (a, b) => sameDecl a b) then none
               else some n
    | .error e => some s!"{n}: {e}"

#guard progFailures.isEmpty

/-- `Scope.branches` follows the checker to the end of every block the
checker accepts (in its own program). -/
def scopeGaps : List String :=
  corpus.flatMap fun (_, p) =>
    (p.filter fun d => accepts (checkDecl p d) && !Scope.covers p d).map (·.name)

#guard scopeGaps.isEmpty

/-- How much was exercised. -/
def blocks : Nat := (corpus.map (·.2.length)).foldl (· + ·) 0
#eval s!"SyntaxTest: {blocks} code blocks round-trip"

/-! ## The restricted-function checker: one accept/refuse per rule

Each test pins one rule with its exact message.  `gsRefuses src msg`: the program parses and `checkProg` refuses it with
a message containing `msg`.  Rule-level tests parse one decl WITHOUT type checking and run a frozen root (or the
checker's restricted block `restrictedPass`) on it directly. -/

def gsChk (s : String) : Except String Unit :=
  match parseProg s with
  | .ok p => checkProg p
  | .error e => .error s!"parse: {e.msg}"

def gsAccepts (s : String) : Bool := match gsChk s with | .ok _ => true | .error _ => false

def gsRefuses (s msg : String) : Bool :=
  match parseProg s with
  | .ok p => match checkProg p with
    | .ok _ => false
    | .error e => (e.splitOn msg).length > 1
  | .error _ => false

/-- The cell count of decl `n` (`cellsDecl` over `checkBinders`'s table), if the body checks and every binder has a width. -/
def gsCells (s n : String) : Option Nat :=
  match parseProg s with
  | .ok p => match p.find? (·.name == n) with
    | some d => match checkBinders p (CoreTyping.declCx d) d.body with
      | .ok tbl => match cellsDecl p d tbl with
        | .ok c => some c
        | .error _ => none
      | .error _ => none
    | none => none
  | .error _ => none

def gsFuel (s n : String) : Option Nat :=
  match parseProg s with
  | .ok p => (p.find? (·.name == n)).map (fuelDecl p)
  | .error _ => none

/-- Rule level: `selfBound` of one parsed decl. -/
def gsSelf (s : String) : Option Tail :=
  match parseDecl s with
  | .ok d => some (selfBound d d.body)
  | .error _ => none

/-- Rule level: `loopCallsOK` of decl `n` in a parsed program, and the count of `m` in its `Decl.allNames`. -/
def gsLoopOK (s n : String) : Option Bool :=
  match parseProg s with
  | .ok p => (p.find? (·.name == n)).map (loopCallsOK p)
  | .error _ => none
def gsCount (s n m : String) : Option Nat :=
  match parseProg s with
  | .ok p => (p.find? (·.name == n)).map fun d => d.allNames.count m
  | .error _ => none

-- an accepted restricted declaration
def gsG (fuel stack : Nat) (loop : String := "") : String :=
  s!"restricted[fuel {fuel}, stack {stack}{loop}] def g(k : Int) : Int =\n  let m = k + 1\n  ret m\n"
#guard gsAccepts (gsG 8 4)
#guard gsFuel (gsG 8 4) "g" == some 2
#guard gsCells (gsG 8 4) "g" == some 2

-- worked example: cellsDecl g = 2, h = 4 (h's own 2 + g's DECLARED stack 2)
def gsH (stack : Nat) : String :=
  s!"restricted[fuel 14, stack {stack}] def h(a : Ptr pub Int) : Int =\n  let x = load a\n  let y = g(x)\n  let z = x + y\n  ret z\n"
#guard gsAccepts (gsG 8 2 ++ gsH 6)
#guard gsCells (gsG 8 2 ++ gsH 6) "h" == some 4
#guard gsRefuses (gsG 8 2 ++ gsH 3) "h: the stack budget is 3 cells but the body needs 4"
#guard gsRefuses (gsG 8 1) "g: the stack budget is 1 cells but the body needs 2"

-- a well-typed batch loop with a SPLIT index (`k` is the second component, `n` index parameter 0)
